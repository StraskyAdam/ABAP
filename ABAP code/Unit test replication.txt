CLASS lcl_repl_contract IMPLEMENTATION.
  METHOD require.
    IF ok <> abap_true.
      RAISE EXCEPTION TYPE zcx_fi_das_error EXPORTING i_text = message.
    ENDIF.
  ENDMETHOD.

  METHOD gate.
    require( ok = xsdbool( refreshed = 14 AND verified = 14 )
             message = 'PRECONDITION: all 14 refresh receipts AND source pairs must pass before Gateway' ).
  ENDMETHOD.

  METHOD receipt.
    require( ok = xsdbool( token IS NOT INITIAL AND actual-token = token
                          AND actual-tabname = tabname AND actual-deleted = abap_true
                          AND actual-success = abap_true )
             message = |PRECONDITION: missing/stale/failed full-refresh receipt for { tabname }| ).
  ENDMETHOD.

  METHOD mode_receipt.
    require( ok = xsdbool( hits = 1 AND desired = observed )
             message = 'PRECONDITION: Gateway did not execute the armed injection exactly once in this context; separate-session topology unsupported' ).
  ENDMETHOD.

  METHOD same_dataset.
    IF source = replica.
      RETURN.
    ENDIF.
    DATA(differences) = dataset_differences(
      source = source replica = replica key_fields = key_fields ).
    require( ok = xsdbool( differences-total = 0 )
      message = |SOURCE MISMATCH { label }: live rows={ lines( source ) }, replicated rows={ lines( replica ) }; total differences={ differences-total }, first { differences-shown }:{ differences-text }| ).
  ENDMETHOD.

  METHOD add_difference.
    result-total = result-total + 1.
    IF result-shown < max_details.
      result-shown = result-shown + 1.
      result-text = result-text && cl_abap_char_utilities=>newline
        && substring( val = detail len = nmin( val1 = strlen( detail ) val2 = 600 ) ).
    ENDIF.
  ENDMETHOD.

  METHOD preview.
    text = |{ value }|.
    IF strlen( text ) > 120.
      text = substring( val = text len = 120 ) && '...'.
    ENDIF.
  ENDMETHOD.

  METHOD collect_fields.
    LOOP AT structure->get_components( ) INTO DATA(component).
      IF component-as_include = abap_true.
        collect_fields( EXPORTING structure = CAST cl_abap_structdescr( component-type )
                        CHANGING names = names ).
      ELSE.
        APPEND component-name TO names.
      ENDIF.
    ENDLOOP.
  ENDMETHOD.

  METHOD require_schema.
    DATA names TYPE stringtab.
    DATA(differences) = VALUE ts_differences( ).
    collect_fields( EXPORTING structure = CAST cl_abap_structdescr(
      cl_abap_typedescr=>describe_by_data( expected ) ) CHANGING names = names ).
    LOOP AT page-rows INTO DATA(row).
      LOOP AT names INTO DATA(name).
        IF NOT line_exists( row-properties[ name = name ] ).
          add_difference(
            EXPORTING detail = |Key { row-ponumber }/{ row-itemno } missing required business property { name }|
            CHANGING result = differences ).
        ENDIF.
      ENDLOOP.
    ENDLOOP.
    require( ok = xsdbool( differences-total = 0 )
      message = |Incomplete Gateway schema { context }: missing={ differences-total }, first { differences-shown }:{ differences-text }| ).
  ENDMETHOD.

  METHOD null_predicate.
    require( ok = xsdbool( fields IS NOT INITIAL AND fields CO 'ABCDEFGHIJKLMNOPQRSTUVWXYZ_0123456789 ' )
      message = 'Invalid null-check projection' ).
    DATA names TYPE stringtab.
    SPLIT fields AT space INTO TABLE names.
    LOOP AT names INTO DATA(name) WHERE table_line IS NOT INITIAL.
      IF predicate IS NOT INITIAL.
        predicate = predicate && ' OR '.
      ENDIF.
      predicate = predicate && name && ' IS NULL'.
    ENDLOOP.
    require( ok = xsdbool( predicate IS NOT INITIAL ) message = 'Empty null-check projection' ).
  ENDMETHOD.

  METHOD index_rows.
    TYPES: BEGIN OF ts_count,
             key TYPE string,
             count TYPE i,
           END OF ts_count.
    DATA counts TYPE HASHED TABLE OF ts_count WITH UNIQUE KEY key.
    DATA fields TYPE stringtab.
    SPLIT key_fields AT space INTO TABLE fields.
    LOOP AT rows ASSIGNING FIELD-SYMBOL(<row>).
      DATA(row_index) = sy-tabix.
      DATA(key) = CONV string( '*' ).
      LOOP AT fields INTO DATA(field) WHERE table_line IS NOT INITIAL.
        ASSIGN COMPONENT field OF STRUCTURE <row> TO FIELD-SYMBOL(<value>).
        require( ok = xsdbool( sy-subrc = 0 ) message = |Unknown dataset key field { field }| ).
        DATA(text) = CONV string( <value> ).
        key = key && |/{ field }={ strlen( text ) }:{ text }|.
      ENDLOOP.
      READ TABLE counts ASSIGNING FIELD-SYMBOL(<count>) WITH TABLE KEY key = key.
      IF sy-subrc <> 0.
        INSERT VALUE #( key = key ) INTO TABLE counts ASSIGNING <count>.
      ENDIF.
      <count>-count = <count>-count + 1.
      INSERT VALUE #( key = key ordinal = <count>-count row_index = row_index ) INTO TABLE result.
    ENDLOOP.
  ENDMETHOD.

  METHOD row_differences.
    IF source = replica.
      RETURN.
    ENDIF.
    DATA(description) = cl_abap_typedescr=>describe_by_data( source ).
    IF description->kind <> cl_abap_typedescr=>kind_struct.
      IF source <> replica.
        add_difference( EXPORTING detail = |{ label } value source=[{ preview( source ) }] replica=[{ preview( replica ) }]|
                        CHANGING result = result ).
      ENDIF.
      RETURN.
    ENDIF.
    DATA(structure) = CAST cl_abap_structdescr( description ).
    LOOP AT structure->get_components( ) INTO DATA(component).
      ASSIGN COMPONENT component-name OF STRUCTURE source TO FIELD-SYMBOL(<source>).
      ASSIGN COMPONENT component-name OF STRUCTURE replica TO FIELD-SYMBOL(<replica>).
      IF <source> <> <replica>.
        add_difference(
          EXPORTING detail = |{ label } field { component-name } source=[{ preview( <source> ) }] replica=[{ preview( <replica> ) }]|
          CHANGING result = result ).
      ENDIF.
    ENDLOOP.
  ENDMETHOD.

  METHOD dataset_differences.
    " Both inputs are sorted on all fields. Ordinals retain repeated keys/rows.
    DATA(source_index) = index_rows( rows = source key_fields = key_fields ).
    DATA(replica_index) = index_rows( rows = replica key_fields = key_fields ).
    LOOP AT source_index INTO DATA(entry).
      READ TABLE replica_index ASSIGNING FIELD-SYMBOL(<match>)
        WITH TABLE KEY key = entry-key ordinal = entry-ordinal.
      IF sy-subrc <> 0.
        add_difference( EXPORTING detail = |Missing replica key { entry-key } occurrence { entry-ordinal }|
                        CHANGING result = result ).
        CONTINUE.
      ENDIF.
      <match>-matched = abap_true.
      READ TABLE source INDEX entry-row_index ASSIGNING FIELD-SYMBOL(<source>).
      READ TABLE replica INDEX <match>-row_index ASSIGNING FIELD-SYMBOL(<replica>).
      row_differences( EXPORTING source = <source> replica = <replica>
                                label = |Key { entry-key } occurrence { entry-ordinal }|
                       CHANGING result = result ).
    ENDLOOP.
    LOOP AT replica_index INTO entry WHERE matched = abap_false.
      add_difference( EXPORTING detail = |Extra replica key { entry-key } occurrence { entry-ordinal }|
                      CHANGING result = result ).
    ENDLOOP.
  ENDMETHOD.

  METHOD whitespace.
    WHILE pos < strlen( json ).
      DATA(char) = json+pos(1).
      IF char <> space AND char <> cl_abap_char_utilities=>horizontal_tab
         AND char <> cl_abap_char_utilities=>newline
         AND char <> cl_abap_char_utilities=>cr_lf(1).
        EXIT.
      ENDIF.
      pos = pos + 1.
    ENDWHILE.
  ENDMETHOD.

  METHOD read_value.
    CLEAR token.
    whitespace( EXPORTING json = json CHANGING pos = pos ).
    DATA: quoted TYPE abap_bool, escaped TYPE abap_bool, stack TYPE string.
    WHILE pos < strlen( json ).
      DATA(char) = json+pos(1).
      IF quoted = abap_true.
        token = token && char.
        IF escaped = abap_true.
          escaped = abap_false.
        ELSEIF char = '\'.
          escaped = abap_true.
        ELSEIF char = '"'.
          quoted = abap_false.
        ENDIF.
      ELSE.
        IF stack IS INITIAL AND ( char = ',' OR char = '}' OR char = ']' ).
          EXIT.
        ENDIF.
        CASE char.
          WHEN '"'. quoted = abap_true.
          WHEN '{'. stack = stack && '}'.
          WHEN '['. stack = stack && ']'.
          WHEN '}' OR ']'.
            DATA(last) = strlen( stack ) - 1.
            require( ok = xsdbool( last >= 0 ) message = 'Malformed JSON nesting' ).
            require( ok = xsdbool( stack+last(1) = char ) message = 'Malformed JSON close' ).
            stack = substring( val = stack len = last ).
        ENDCASE.
        IF char <> space AND char <> cl_abap_char_utilities=>horizontal_tab
           AND char <> cl_abap_char_utilities=>newline
           AND char <> cl_abap_char_utilities=>cr_lf(1).
          token = token && char.
        ENDIF.
      ENDIF.
      pos = pos + 1.
    ENDWHILE.
    require( ok = xsdbool( token IS NOT INITIAL AND quoted = abap_false AND stack IS INITIAL )
             message = 'Malformed/unterminated JSON value' ).
  ENDMETHOD.

  METHOD properties.
    DATA pos TYPE i.
    whitespace( EXPORTING json = json CHANGING pos = pos ).
    require( ok = xsdbool( substring( val = json off = pos len = 1 ) = '{' )
             message = 'Expected JSON object' ).
    pos = pos + 1.
    whitespace( EXPORTING json = json CHANGING pos = pos ).
    WHILE pos < strlen( json ) AND json+pos(1) <> '}'.
      " Business names are literal SEGW names; escaped names fail closed.
      require( ok = xsdbool( json+pos(1) = '"' ) message = 'Expected property name' ).
      pos = pos + 1.
      DATA(start) = pos.
      WHILE pos < strlen( json ) AND json+pos(1) <> '"'.
        require( ok = xsdbool( json+pos(1) <> '\' ) message = 'Escaped property names unsupported' ).
        pos = pos + 1.
      ENDWHILE.
      require( ok = xsdbool( pos < strlen( json ) ) message = 'Unterminated property name' ).
      DATA(name) = substring( val = json off = start len = pos - start ).
      pos = pos + 1.
      whitespace( EXPORTING json = json CHANGING pos = pos ).
      require( ok = xsdbool( pos < strlen( json ) AND json+pos(1) = ':' )
               message = 'Expected property colon' ).
      pos = pos + 1.
      read_value( EXPORTING json = json IMPORTING token = DATA(token) CHANGING pos = pos ).
      INSERT VALUE #( name = name token = token ) INTO TABLE result.
      require( ok = xsdbool( sy-subrc = 0 ) message = |Duplicate JSON property { name }| ).
      whitespace( EXPORTING json = json CHANGING pos = pos ).
      IF pos < strlen( json ) AND json+pos(1) = ','.
        pos = pos + 1.
        whitespace( EXPORTING json = json CHANGING pos = pos ).
        require( ok = xsdbool( pos < strlen( json ) AND json+pos(1) <> '}' )
                 message = 'Trailing JSON comma' ).
      ELSE.
        EXIT.
      ENDIF.
    ENDWHILE.
    require( ok = xsdbool( pos < strlen( json ) AND json+pos(1) = '}' )
             message = 'Expected object close' ).
    pos = pos + 1.
    whitespace( EXPORTING json = json CHANGING pos = pos ).
    require( ok = xsdbool( pos = strlen( json ) ) message = 'Trailing JSON data' ).
  ENDMETHOD.

  METHOD page.
    DATA(envelope) = properties( json ).
    require( ok = xsdbool( line_exists( envelope[ name = 'd' ] ) )
             message = 'Missing OData d envelope' ).
    DATA(body) = properties( envelope[ name = 'd' ]-token ).
    require( ok = xsdbool( line_exists( body[ name = '__count' ] )
                      AND line_exists( body[ name = 'results' ] )
                      AND NOT line_exists( body[ name = '__next' ] ) )
             message = 'Missing count/results or unsupported server-driven paging' ).
    DATA(count_token) = body[ name = '__count' ]-token.
    require( ok = xsdbool( matches( val = count_token regex = '"[0-9]+"' ) )
             message = 'Invalid OData count' ).
    result-count = CONV int8( substring( val = count_token off = 1 len = strlen( count_token ) - 2 ) ).
    DATA(array) = body[ name = 'results' ]-token.
    require( ok = xsdbool( strlen( array ) >= 2 AND array(1) = '[' )
             message = 'Expected results array' ).
    DATA pos TYPE i VALUE 1.
    DATA keys TYPE HASHED TABLE OF string WITH UNIQUE KEY table_line.
    WHILE pos < strlen( array ) AND array+pos(1) <> ']'.
      read_value( EXPORTING json = array IMPORTING token = DATA(row_json) CHANGING pos = pos ).
      DATA(props) = properties( row_json ).
      DELETE props WHERE name = '__metadata'.
      require( ok = xsdbool( line_exists( props[ name = 'PONUMBER' ] )
                        AND line_exists( props[ name = 'ITEMNO' ] )
                        AND line_exists( props[ name = 'COMPANYCODE' ] ) )
               message = 'Missing business key/company property' ).
      DATA(row) = VALUE ts_row( ponumber = props[ name = 'PONUMBER' ]-token
                               itemno = props[ name = 'ITEMNO' ]-token
                               company = props[ name = 'COMPANYCODE' ]-token
                               properties = props ).
      require( ok = xsdbool( matches( val = row-ponumber regex = '"[0-9]{10}"' )
                        AND matches( val = row-itemno regex = '"[0-9]{5}"' ) )
               message = 'Invalid/empty business key' ).
      INSERT row-ponumber && row-itemno INTO TABLE keys.
      require( ok = xsdbool( sy-subrc = 0 ) message = 'Duplicate business key in page' ).
      APPEND row TO result-rows.
      IF pos < strlen( array ) AND array+pos(1) = ','.
        pos = pos + 1.
        require( ok = xsdbool( pos < strlen( array ) AND array+pos(1) <> ']' )
                 message = 'Trailing results comma' ).
      ELSE.
        EXIT.
      ENDIF.
    ENDWHILE.
    require( ok = xsdbool( pos = strlen( array ) - 1 AND array+pos(1) = ']' )
             message = 'Malformed results array' ).
    require( ok = xsdbool( result-count >= lines( result-rows ) )
             message = 'Count smaller than page' ).
  ENDMETHOD.

  METHOD same_page.
    IF source = replica.
      RETURN.
    ENDIF.
    DATA(differences) = page_differences( source = source replica = replica ).
    require( ok = xsdbool( differences-total = 0 )
      message = |Gateway { context }: differences={ differences-total }, first { differences-shown }:{ differences-text }| ).
  ENDMETHOD.

  METHOD page_differences.
    IF source-count <> replica-count.
      add_difference( EXPORTING detail = |Inlinecount source={ source-count } replica={ replica-count }|
                      CHANGING result = result ).
    ENDIF.
    LOOP AT source-rows ASSIGNING FIELD-SYMBOL(<source>).
      DATA(source_position) = sy-tabix.
      READ TABLE replica-rows ASSIGNING FIELD-SYMBOL(<replica>)
        WITH KEY ponumber = <source>-ponumber itemno = <source>-itemno.
      DATA(replica_position) = sy-tabix.
      DATA(label) = |Key { <source>-ponumber }/{ <source>-itemno }|.
      IF sy-subrc <> 0.
        add_difference( EXPORTING detail = |Missing replica { label }| CHANGING result = result ).
        CONTINUE.
      ENDIF.
      IF source_position <> replica_position.
        add_difference( EXPORTING detail = |{ label } position source={ source_position } replica={ replica_position }|
                        CHANGING result = result ).
      ENDIF.
      LOOP AT <source>-properties INTO DATA(property).
        READ TABLE <replica>-properties INTO DATA(other) WITH TABLE KEY name = property-name.
        IF sy-subrc <> 0.
          add_difference(
            EXPORTING detail = |{ label } missing property { property-name } source=[{ preview( property-token ) }]|
            CHANGING result = result ).
        ELSEIF property-token <> other-token.
          add_difference(
            EXPORTING detail = |{ label } field { property-name } source=[{ preview( property-token ) }] replica=[{ preview( other-token ) }]|
            CHANGING result = result ).
        ENDIF.
      ENDLOOP.
      LOOP AT <replica>-properties INTO property.
        IF NOT line_exists( <source>-properties[ name = property-name ] ).
          add_difference(
            EXPORTING detail = |{ label } extra property { property-name } replica=[{ preview( property-token ) }]|
            CHANGING result = result ).
        ENDIF.
      ENDLOOP.
    ENDLOOP.
    LOOP AT replica-rows ASSIGNING <replica>.
      IF NOT line_exists( source-rows[ ponumber = <replica>-ponumber itemno = <replica>-itemno ] ).
        add_difference( EXPORTING detail = |Extra replica key { <replica>-ponumber }/{ <replica>-itemno }|
                        CHANGING result = result ).
      ENDIF.
    ENDLOOP.
  ENDMETHOD.
ENDCLASS.

CLASS ltc_repl_contract IMPLEMENTATION.
  METHOD fixture.
    json = `{"d":{"__count":"1","results":[{"PONUMBER":"8000000001",`
      && `"ITEMNO":"00010","COMPANYCODE":"2028",` && extra && `}]}}`.
  ENDMETHOD.

  METHOD source_multisets.
    TYPES: BEGIN OF ts_value,
             key TYPE i,
             payload TYPE string,
           END OF ts_value,
           tt_value TYPE STANDARD TABLE OF ts_value WITH EMPTY KEY.
    DATA(source) = VALUE tt_value( ( key = 1 payload = 'A' )
                                   ( key = 1 payload = 'B' )
                                   ( key = 1 payload = 'A' ) ).
    DATA(replica) = VALUE tt_value( ( key = 1 payload = 'B' )
                                    ( key = 1 payload = 'A' )
                                    ( key = 1 payload = 'A' ) ).
    SORT source BY key payload.
    SORT replica BY key payload.
    lcl_repl_contract=>same_dataset( source = source replica = replica label = 'positive' ).
    DELETE replica INDEX 1.
    TRY.
        lcl_repl_contract=>same_dataset( source = source replica = replica label = 'multiplicity' ).
        cl_abap_unit_assert=>fail( 'Lost duplicate must fail' ).
      CATCH zcx_fi_das_error.
    ENDTRY.
    replica = source.
    replica[ 3 ]-payload = 'C'.
    TRY.
        lcl_repl_contract=>same_dataset( source = source replica = replica label = 'non-key field' ).
        cl_abap_unit_assert=>fail( 'Changed non-key field must fail' ).
      CATCH zcx_fi_das_error.
    ENDTRY.
  ENDMETHOD.

  METHOD strict_json.
    DATA(source) = lcl_repl_contract=>page( fixture( ) ).
    DATA(replica) = lcl_repl_contract=>page(
      `{"d":{"results":[{"__metadata":{"uri":"different","type":"test"},`
      && `"COMPANYCODE":"2028","AMOUNT":"1.00","ITEMNO":"00010",`
      && `"PONUMBER":"8000000001"}],"__count":"1"}}` ).
    lcl_repl_contract=>same_page( source = source replica = replica ).
    source = lcl_repl_contract=>page( fixture(
      `"DESC":" A;B; ","DATE":"/Date(1793145600000)/","TIME":"PT00H00M01S",`
      && `"N":null,"FLAG":false,"DEC":12345678901234567890.123456789` ) ).
    replica = source.
    lcl_repl_contract=>same_page( source = source replica = replica ).
    cl_abap_unit_assert=>assert_equals(
      act = source-rows[ 1 ]-properties[ name = 'DEC' ]-token
      exp = `12345678901234567890.123456789` ).
    cl_abap_unit_assert=>assert_equals(
      act = source-rows[ 1 ]-properties[ name = 'DESC' ]-token exp = `" A;B; "` ).
  ENDMETHOD.

  METHOD strict_rejections.
    DATA originals TYPE stringtab.
    DATA mutations TYPE stringtab.
    originals = VALUE #( ( `"AMOUNT":"1.00"` ) ( `"DESC":" A "` )

  METHOD null_preflight.
    CASE view.
      WHEN 'ZI_FI_DAS_EKKO' OR 'ZI_FI_DAS_EKPO' OR 'ZI_FI_DAS_EKKN'
        OR 'ZI_FI_DAS_EKBE' OR 'ZI_FI_DAS_T001' OR 'ZI_FI_DAS_LFA1'
        OR 'ZI_FI_DAS_CSKS' OR 'ZI_FI_DAS_CSKT' OR 'ZI_FI_DAS_USR01'
        OR 'ZI_FI_DAS_USR02' OR 'ZI_FI_DAS_USR21' OR 'ZI_FI_DAS_ADRP'
        OR 'ZI_FI_DAS_ADR6' OR 'ZI_FI_DAS_ACCRUAL_EXCL'.
      WHEN OTHERS.
        lcl_repl_contract=>require( ok = abap_false message = 'Unknown null-preflight CDS view' ).
    ENDCASE.
    DATA(condition) = lcl_repl_contract=>null_predicate( fields ).
    DO 2 TIMES.
      DATA(parameters) = COND string( WHEN sy-index = 1
        THEN '( p_use_replication = '' '' )'
        ELSE '( p_use_replication = ''X'' )' ).
      DATA(from_clause) = view && parameters.
      DATA null_count TYPE int8.
      SELECT COUNT( * ) FROM (from_clause) WHERE (condition) INTO @null_count.
      lcl_repl_contract=>require( ok = xsdbool( null_count = 0 )
        message = |PRECONDITION: projected SQL NULL found: { from_clause }, rows={ null_count }; null indicators required before comparison| ).
    ENDDO.
  ENDMETHOD.

  METHOD gateway.
    CLEAR: armed, desired, observed, hits.
    lcl_repl_contract=>gate( refreshed = refreshed verified = verified ).
    lcl_repl_contract=>require( ok = xsdbool( clock( ) = started )
      message = 'PRECONDITION: server/local/UTC/CET/database day or month changed; rerun' ).
    DATA headers TYPE /iwfnd/sutil_property_t.
    headers = VALUE #( ( name = '~request_method' value = 'GET' )
                       ( name = '~request_uri' value = path )
                       ( name = 'accept' value = 'application/json' ) ).
    CLEAR: hits, observed.
    desired = mode.
    armed = abap_true.
    TRY.
        /iwfnd/cl_sutil_client_proxy=>get_instance( )->web_request(
          EXPORTING it_request_header = headers
          IMPORTING ev_status_code = DATA(status) ev_status_text = DATA(status_text)
                    ev_response_body = DATA(body) ).
      CLEANUP.
        CLEAR: armed, desired, observed, hits.
    ENDTRY.
    DATA(receipt_hits) = hits.
    DATA(receipt_mode) = observed.
    CLEAR: armed, desired, observed, hits.
    " This must precede ANY response comparison, including the first probe.
    lcl_repl_contract=>mode_receipt( hits = receipt_hits desired = mode observed = receipt_mode ).
    lcl_repl_contract=>require( ok = xsdbool( clock( ) = started )
      message = 'PRECONDITION: server/local/UTC/CET/database day or month changed; rerun' ).
    lcl_repl_contract=>require( ok = xsdbool( status = 200 )
      message = |Gateway HTTP { status } { status_text }; { path }| ).
    result = lcl_repl_contract=>page( /iwfnd/cl_sutil_xml_helper=>transform_to_string( body ) ).
    IF path CS '/openPOParameters('.
      DATA expected_over TYPE zfi_das_over.
      lcl_repl_contract=>require_schema( page = result expected = expected_over context = path ).
    ELSEIF path CS '/thresholdParameters('.
      DATA expected_under TYPE zfi_das_under.
      lcl_repl_contract=>require_schema( page = result expected = expected_under context = path ).
    ELSEIF path CS '/exclusionPOParameters('.
      DATA expected_exclude TYPE zfi_das_exclude.
      lcl_repl_contract=>require_schema( page = result expected = expected_exclude context = path ).
    ELSE.
      lcl_repl_contract=>require( ok = abap_false message = |Unknown full-result schema: { path }| ).
    ENDIF.
  ENDMETHOD.

  METHOD clock.
    DATA stamp TYPE timestampl.
    DATA db_date TYPE d.
    GET TIME.
    result-server = sy-datum.
    result-local = sy-datlo.
    GET TIME STAMP FIELD stamp.
    CONVERT TIME STAMP stamp TIME ZONE 'UTC' INTO DATE result-utc.
    lcl_repl_contract=>require( ok = xsdbool( sy-subrc = 0 )
      message = 'PRECONDITION: UTC clock conversion failed' ).
    CONVERT TIME STAMP stamp TIME ZONE 'CET' INTO DATE result-cet.
    lcl_repl_contract=>require( ok = xsdbool( sy-subrc = 0 )
      message = 'PRECONDITION: CET clock conversion failed' ).
    EXEC SQL.
      SELECT CURRENT_DATE INTO :db_date FROM DUMMY
    ENDEXEC.
    result-database = db_date.
    lcl_repl_contract=>require(
      ok = xsdbool( result-server IS NOT INITIAL AND result-local IS NOT INITIAL
                    AND result-utc IS NOT INITIAL AND result-cet IS NOT INITIAL
                    AND result-database IS NOT INITIAL )
      message = 'PRECONDITION: unavailable day/month boundary clock' ).
    lcl_repl_contract=>require( ok = xsdbool( result-server = result-database )
      message = 'PRECONDITION: ABAP sy-datum and HANA CURRENT_DATE must be aligned' ).
  ENDMETHOD.

  METHOD scenario.
    DATA: skip TYPE i, total TYPE int8 VALUE -1, prior TYPE string,
          seen TYPE HASHED TABLE OF string WITH UNIQUE KEY table_line,
          filter TYPE string.
    DATA(orderby) = SWITCH string( variant
      WHEN 1 THEN 'PONUMBER asc,ITEMNO asc'
      WHEN 2 THEN 'PONUMBER desc,ITEMNO desc'
      ELSE 'COMPANYCODE asc,PONUMBER asc,ITEMNO asc' ).
    orderby = cl_http_utility=>escape_url( orderby ).
    IF variant = 3.
      DATA(probe) = gateway(
        path = |/sap/opu/odata/sap/ZFI_DAS_SRV/{ base }?$format=json&$inlinecount=allpages&$top=1&$orderby=PONUMBER,ITEMNO|
        mode = abap_false ).
      lcl_repl_contract=>require( ok = xsdbool( probe-rows IS NOT INITIAL )
        message = |PRECONDITION: filter scenario needs data: { base }| ).
      DATA(company) = probe-rows[ 1 ]-company.
      lcl_repl_contract=>require( ok = xsdbool( matches( val = company regex = '"[A-Za-z0-9]{4}"' ) )
        message = 'PRECONDITION: expected four-character company code' ).
      company = substring( val = company off = 1 len = 4 ).
      filter = '&$filter=' && cl_http_utility=>escape_url( |COMPANYCODE eq '{ company }'| ).
    ENDIF.
    " Size one guarantees actual skip/page boundaries even for small scopes.
    DATA(size) = COND i( WHEN variant = 2 THEN 1 ELSE page_size ).
    DO.
      DATA(path) = |/sap/opu/odata/sap/ZFI_DAS_SRV/{ base }?$format=json&$inlinecount=allpages&$orderby={ orderby }&$top={ size }&$skip={ skip }{ filter }|.
      DATA(source) = gateway( path = path mode = abap_false ).
      DATA(replica) = gateway( path = path mode = abap_true ).
      lcl_repl_contract=>same_page( source = source replica = replica context = path ).
      IF total = -1.
        total = source-count.
        lcl_repl_contract=>require(
          ok = xsdbool( total >= 2 OR ( variant = 3 AND total > 0 ) )
          message = |PRECONDITION: nonvacuity needs >=2 rows (filter >=1): { base }| ).
      ENDIF.
      lcl_repl_contract=>require( ok = xsdbool( source-count = total )
        message = |PRECONDITION: count changed during paging: { base }| ).
      DATA(expected_size) = COND int8( WHEN total - skip < size THEN total - skip ELSE size ).
      lcl_repl_contract=>require( ok = xsdbool( lines( source-rows ) = expected_size )
        message = 'Missing/extra/early-empty page' ).
      LOOP AT source-rows INTO DATA(row).
        DATA(key) = row-ponumber && row-itemno.
        INSERT key INTO TABLE seen.
        lcl_repl_contract=>require( ok = xsdbool( sy-subrc = 0 )
          message = |Duplicate key across pages: { key }| ).
        DATA(sort_key) = COND string( WHEN variant = 3 THEN row-company && key ELSE key ).
        IF prior IS NOT INITIAL.
          lcl_repl_contract=>require(
            ok = xsdbool( ( variant = 2 AND prior > sort_key )
                       OR ( variant <> 2 AND prior < sort_key ) )
            message = 'Requested stable page order violated' ).
        ENDIF.
        IF variant = 3.
          lcl_repl_contract=>require( ok = xsdbool( row-company = |"{ company }"| )
            message = 'Generic company filter was not applied' ).
        ENDIF.
        prior = sort_key.
      ENDLOOP.
      skip = skip + lines( source-rows ).
      FREE: source, replica.
      IF skip = total.
        EXIT.
      ENDIF.
    ENDDO.
    lcl_repl_contract=>require( ok = xsdbool( lines( seen ) = total )
      message = 'Final unique-key coverage differs from inlinecount' ).
    " Explicit beyond-end request must be empty with the original total.
    DATA(tail_path) = |/sap/opu/odata/sap/ZFI_DAS_SRV/{ base }?$format=json&$inlinecount=allpages&$orderby={ orderby }&$top={ size }&$skip={ skip }{ filter }|.
    source = gateway( path = tail_path mode = abap_false ).
    replica = gateway( path = tail_path mode = abap_true ).
    lcl_repl_contract=>same_page( source = source replica = replica context = tail_path ).
    lcl_repl_contract=>require( ok = xsdbool( source-rows IS INITIAL AND source-count = total )
      message = 'Beyond-end page/count contract violated' ).
  ENDMETHOD.
ENDCLASS.
