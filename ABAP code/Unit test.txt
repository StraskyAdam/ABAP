" ============================================================================
" XSA vs SEGW functional JSON parity test - paged low-memory implementation
"
" Memory strategy:
" - XSA fixture is indexed once with PO/item and object character offsets only.
" - Raw SEGW JSON is fetched one SKIP/TOP page at a time.
" - JSON properties are read directly from both raw payloads for each matched row.
" - Comparison is strict except for documented, field-specific SEGW mappings of
"   XSA SQL NULL values and a temporary unordered-aggregate compatibility rule.
" - No complete XSA row table, no complete property tree, and no copied JSON
"   objects are held in memory.
"
" Paging relies on the stable ORDER BY implemented by all dashboard flows.
" ============================================================================

  METHOD run_paged_case.

    DATA:
      lv_expected_json   TYPE string,
      lv_actual_json     TYPE string,
      lt_expected_index  TYPE tt_expected_index,
      lt_actual_index    TYPE tt_expected_index,
      lv_skip            TYPE i,
      lv_page_rows       TYPE i,
      lv_total_count     TYPE int8,
      lv_finished        TYPE abap_bool.

    lv_expected_json =
      read_mime_fixture(
        i_fixture = is_case-fixture ).

    lt_expected_index =
      build_expected_index(
        i_json = lv_expected_json ).

    WHILE lv_finished = abap_false.

      CLEAR:
        lv_actual_json,
        lv_total_count,
        lv_page_rows.

      fetch_gateway_page(
        EXPORTING
          is_case       = is_case
          i_skip        = lv_skip
        IMPORTING
          e_json        = lv_actual_json
          e_total_count = lv_total_count ).

      lt_actual_index =
        build_expected_index(
          i_json = lv_actual_json ).

      lv_page_rows = lines( lt_actual_index ).

      compare_actual_page(
        EXPORTING
          i_fixture       = is_case-fixture
          i_expected_json = lv_expected_json
          i_actual_json   = lv_actual_json
          it_actual_index = lt_actual_index
        CHANGING
          ct_expected_index = lt_expected_index ).

*     TODO_PARITY_STRICT_COUNT:
*     Current SEGW implementation can return __count = "0" even with rows.
*     Remove the LV_TOTAL_COUNT > 0 condition after inline count is fixed.
      IF lv_total_count > 0
         AND lv_total_count <> lines( lt_expected_index ).
        add_difference(
          i_text =
              |Inline count differs; Fixture={ is_case-fixture }; |
            && |XSA={ lines( lt_expected_index ) }; |
            && |SEGW={ lv_total_count }| ).
      ENDIF.

      IF lv_page_rows = 0.
        lv_finished = abap_true.
        CONTINUE.
      ENDIF.

      lv_skip = lv_skip + lv_page_rows.

      IF lv_total_count > 0.
        IF lv_skip >= lv_total_count.
          lv_finished = abap_true.
        ENDIF.
      ELSEIF lv_page_rows < mc_page_size.
        lv_finished = abap_true.
      ENDIF.

      FREE lt_actual_index.

    ENDWHILE.

    LOOP AT lt_expected_index
      ASSIGNING FIELD-SYMBOL(<ls_expected>)
      WHERE matched = abap_false.

      add_difference(
        i_text =
            |Missing SEGW raw JSON row; Fixture={ is_case-fixture }; |
          && |PO={ <ls_expected>-ponumber }; |
          && |ITEMNO={ <ls_expected>-itemno }| ).

    ENDLOOP.

    FREE:
      lt_expected_index,
      lt_actual_index,
      lv_expected_json,
      lv_actual_json.

  ENDMETHOD.

  METHOD fetch_gateway_page.

    DATA lv_path TYPE string.

    CLEAR:
      e_json,
      e_total_count.

    lv_path =
      build_resource_path(
        is_case = is_case
        i_skip  = i_skip ).

    e_json =
      execute_gateway_get(
        i_resource_path = lv_path ).

    e_total_count =
      read_inline_count(
        i_json = e_json ).

  ENDMETHOD.

  METHOD build_resource_path.

    DATA:
      lv_email   TYPE string,
      lv_poowner TYPE string,
      lv_pstype  TYPE string.

    lv_email = odata_quote( i_value = is_case-email ).
    lv_poowner = odata_quote( i_value = is_case-poowner ).
    lv_pstype = odata_quote( i_value = is_case-pstype ).

    CASE is_case-flow.
      WHEN zif_fi_das_dashboard=>mc_over.
        rv_path =
            '/sap/opu/odata/sap/ZFI_DAS_SRV/'
          && |openPOParameters(IP_PSTYPE='{ lv_pstype }',|
          && |IP_EMAIL='{ lv_email }',|
          && |IP_POOWNER='{ lv_poowner }')/Results|.

      WHEN zif_fi_das_dashboard=>mc_under.

  METHOD append_filter.

    DATA:
      lt_filters TYPE stringtab,
      lv_filter  TYPE string,
      lv_low     TYPE string,
      lv_high    TYPE string,
      lv_part    TYPE string.

    LOOP AT is_case-ponumber ASSIGNING FIELD-SYMBOL(<ls_ponumber>).
      lv_low = odata_quote( i_value = <ls_ponumber>-low ).
      lv_high = odata_quote( i_value = <ls_ponumber>-high ).

      CASE <ls_ponumber>-option.
        WHEN 'EQ'.
          lv_part = |PONUMBER eq '{ lv_low }'|.
        WHEN 'BT'.
          lv_part =
              |(PONUMBER ge '{ lv_low }' and |
            && |PONUMBER le '{ lv_high }')|.
        WHEN OTHERS.
          CONTINUE.
      ENDCASE.

      IF <ls_ponumber>-sign = 'E'.
        lv_part = |not ({ lv_part })|.
      ENDIF.
      APPEND lv_part TO lt_filters.
    ENDLOOP.

    LOOP AT is_case-itemno ASSIGNING FIELD-SYMBOL(<ls_itemno>).
      lv_low = |{ <ls_itemno>-low ALPHA = IN }|.
      lv_high = |{ <ls_itemno>-high ALPHA = IN }|.
      lv_low = odata_quote( i_value = lv_low ).
      lv_high = odata_quote( i_value = lv_high ).

      CASE <ls_itemno>-option.
        WHEN 'EQ'.
          lv_part = |ITEMNO eq '{ lv_low }'|.
        WHEN 'BT'.
          lv_part =
              |(ITEMNO ge '{ lv_low }' and |
            && |ITEMNO le '{ lv_high }')|.
        WHEN OTHERS.
          CONTINUE.
      ENDCASE.

      IF <ls_itemno>-sign = 'E'.
        lv_part = |not ({ lv_part })|.
      ENDIF.
      APPEND lv_part TO lt_filters.
    ENDLOOP.

    LOOP AT is_case-po_contains ASSIGNING FIELD-SYMBOL(<ls_po_contains>).
      lv_low = odata_quote( i_value = <ls_po_contains>-value ).
      APPEND |substringof('{ lv_low }',PONUMBER)| TO lt_filters.
    ENDLOOP.

    LOOP AT is_case-item_contains ASSIGNING FIELD-SYMBOL(<ls_item_contains>).
      lv_low = odata_quote( i_value = <ls_item_contains>-value ).
      APPEND |substringof('{ lv_low }',ITEMNO)| TO lt_filters.
    ENDLOOP.

    LOOP AT is_case-supplier ASSIGNING FIELD-SYMBOL(<ls_supplier>).
      lv_low = odata_quote( i_value = <ls_supplier>-value ).
      APPEND |substringof('{ lv_low }',SUPPLIERNAME)| TO lt_filters.
    ENDLOOP.

    IF lt_filters IS INITIAL.
      RETURN.
    ENDIF.

    CONCATENATE LINES OF lt_filters
      INTO lv_filter
      SEPARATED BY ' and '.

    cl_http_utility=>escape_url(
      EXPORTING
        unescaped = lv_filter
      RECEIVING
        escaped   = lv_filter ).

    cv_path = cv_path && |&$filter={ lv_filter }|.

  ENDMETHOD.

  METHOD odata_quote.

    rv_value = CONV string( i_value ).
    SHIFT rv_value RIGHT DELETING TRAILING space.

    REPLACE ALL OCCURRENCES OF ''''
      IN rv_value
      WITH ''''''.

  ENDMETHOD.

* values differently from XSA SQL NULL. IS_EQUIVALENT_NULL contains a
* field-specific allowlist. There is intentionally no global rule such
* as NULL = empty string or NULL = zero.
*--------------------------------------------------------------------*
    IF is_equivalent_null(
         i_name           = i_name
         i_expected_token = lv_expected_token
         i_actual_token   = lv_actual_token ) = abap_true.
      RETURN.
    ENDIF.

*--------------------------------------------------------------------*
* Normalize non-NULL CHANGEDTIME values
*
* XSA can omit leading zeroes while SEGW emits two-digit components.
* Both representations describe the same Edm.Time value:
*
*   "PT9H40M11S" = "PT09H40M11S"
*   "PT8H24M4S"  = "PT08H24M04S"
*
* Only CHANGEDTIME enters this normalization. The actual hour, minute,
* and second values remain strict.
*--------------------------------------------------------------------*
    IF i_name = 'CHANGEDTIME'.
      lv_expected_token =
        normalize_odata_time(
          i_token = lv_expected_token ).

      lv_actual_token =
        normalize_odata_time(
          i_token = lv_actual_token ).
    ENDIF.

*--------------------------------------------------------------------*
* TODO_PARITY_STRICT_001 - temporary aggregate-order exception
*
* Aggregate segment order can currently differ between XSA and SEGW.
* Both values are decoded first. Content and multiplicity must still be
* identical after ordering is ignored.
*
* Accepted:
*   XSA="A;B;C", SEGW="C;A;B"
*
* Rejected:
*   XSA="A;B;C", SEGW="A;B"
*   XSA="A;A;B", SEGW="A;B"
*
* Remove this branch after aggregate ordering is deterministic and
* aligned between both implementations.
*--------------------------------------------------------------------*
    IF is_unordered_aggregate_field(
         i_name = i_name ) = abap_true.

      lv_expected_text =
        unquote_json(
          i_token = lv_expected_token ).

      lv_actual_text =
        unquote_json(
          i_token = lv_actual_token ).

      IF equal_aggregate_content(
           i_expected = lv_expected_text
           i_actual   = lv_actual_text ) = abap_false.
        add_difference(
          i_text =
              |Aggregate content differs; Fixture={ i_fixture }; |
            && |PO={ i_ponumber }; ITEMNO={ i_itemno }; |
            && |Field={ i_name }; XSA={ lv_expected_token }; |
            && |SEGW={ lv_actual_token }| ).
      ENDIF.

      RETURN.
    ENDIF.

*--------------------------------------------------------------------*
* Compare decoded JSON string values
*
* Different JSON tokens can represent exactly the same string received
* by the UI. UNQUOTE_JSON removes the outer quotes and decodes supported
* JSON escape sequences.
*
* Accepted examples:
*   XSA="S \u0026 K",              SEGW="S & K"
*   XSA="King\u0027s College",    SEGW="King's College"
*   XSA="\/Date(1677240711333)\/",
*   SEGW="/Date(1677240711333)/"
*
* Rejected examples:
*   XSA="King's College", SEGW="Kings College"
*   XSA="A & B",          SEGW="A and B"
*
* NULL, numbers, booleans, arrays, and objects do not enter this branch
* and remain subject to strict token comparison below.
*--------------------------------------------------------------------*
    lv_expected_len =
      strlen( lv_expected_token ).

    lv_actual_len =
      strlen( lv_actual_token ).

    IF lv_expected_len >= 2
       AND lv_actual_len >= 2
       AND lv_expected_token(1) = '"'
       AND lv_actual_token(1) = '"'.

      lv_expected_text =
        unquote_json(
          i_token = lv_expected_token ).

      lv_actual_text =
        unquote_json(
          i_token = lv_actual_token ).

      IF lv_expected_text <> lv_actual_text.
        add_difference(
          i_text =
              |String value differs; Fixture={ i_fixture }; |
            && |PO={ i_ponumber }; ITEMNO={ i_itemno }; |
            && |Field={ i_name }; XSA={ lv_expected_token }; |
            && |SEGW={ lv_actual_token }| ).
      ENDIF.

      RETURN.
    ENDIF.

*--------------------------------------------------------------------*
* Strict comparison for every remaining JSON token
*--------------------------------------------------------------------*
    IF lv_expected_token <> lv_actual_token.
      add_difference(
        i_text =
            |Raw value differs; Fixture={ i_fixture }; |
          && |PO={ i_ponumber }; ITEMNO={ i_itemno }; |
          && |Field={ i_name }; XSA={ lv_expected_token }; |
          && |SEGW={ lv_actual_token }| ).
    ENDIF.

  ENDMETHOD.

*----------------------------------------------------------------------*
* TODO_PARITY_FUNCTIONAL_NULL_001 - DOCUMENTED SEGW COMPATIBILITY

* XSA preserves database NULL and emits the JSON token null. Classic
* SEGW receives typed ABAP values and emits type-specific initial values:
*
*   initial character value -> ""
*   initial decimal value   -> "0.00"
*   initial TIMS value      -> "PT00H00M00S"
*
* The rules below are deliberately field-specific and one-directional:
*
*   XSA null -> approved SEGW initial representation
*
* Non-NULL XSA values remain strict. These examples are NOT accepted:
*
*   XSA="0.00", SEGW=null
*   XSA=null,   SEGW="1.00"
*   XSA=null,   SEGW="X"
*
* Reassess this allowlist whenever UI consumption or the OData contract
* changes. Do not extend it based on ABAP or EDM type alone.
*----------------------------------------------------------------------*
  METHOD is_equivalent_null.

  rv_result = abap_false.

*--------------------------------------------------------------------*
* TODO_PARITY_FUNCTIONAL_NULL_001
*
* Functional compatibility between XSA SQL NULL values and standard
* SEGW serialization of initial ABAP values.
*
* Background
* ----------
* XSA receives database NULL values directly from SAP HANA and emits:
*
*   null
*
* Classic SEGW receives typed ABAP values. Elementary ABAP value types
* do not preserve the distinction between an absent source value and an
* explicitly supplied initial value.
*
* Standard SEGW therefore serializes initial ABAP values as follows:
*
*   Character value  -> ""
*   Decimal value    -> "0.00"
*   TIMS value       -> "PT00H00M00S"
*
* Scope
* -----
* The rules below are deliberately field-specific. Do not replace them
* with global rules such as:
*
*   null = ""
*   null = "0.00"
*   null = "PT00H00M00S"
*
* NULL and an initial value can have different business meanings for
* other properties.
*
* The exceptions are also one-directional:
*
*   XSA null -> approved SEGW initial representation
*
* A non-NULL XSA value remains subject to strict comparison.
*
* The following examples are therefore NOT accepted:
*
*   XSA="0.00", SEGW=null
*   XSA=null,   SEGW="1.00"
*   XSA=null,   SEGW="X"
*
* Review this allowlist whenever UI consumption or the OData contract
* changes.
*--------------------------------------------------------------------*
  IF i_expected_token <> 'null'.
    RETURN.
  ENDIF.

  CASE i_name.

*--------------------------------------------------------------------*
* Character properties serialized by SEGW as ""
*
* FULL_NAME
*   The current UI usage is display-only. Both XSA NULL and the SEGW
*   empty string produce an empty displayed value and do not change
*   active application control flow.
*
* DECIMAL_FORMAT
*   This exception applies to DECIMAL_FORMAT in the open-PO result.
*   The effective user decimal format is loaded separately from the
*   userDetails service.
*
* DATFM and USERNAME
*   No active UI consumer beyond metadata was identified.
*
* COSTCENTERDESC and MANAGEUNIT
*   These are aggregate/display properties. XSA NULL and SEGW empty
*   string both represent no aggregate value. Non-empty values still go
*   through the unordered aggregate comparison, including duplicates.
*
* NOTIF_24_HR
*   The UI activates the notification indicator only when the value is
*   exactly "X". Both NULL and "" therefore leave it inactive.
*
* REVIWED_BY
*   The property name intentionally follows the existing OData contract.
*   Both XSA NULL and the SEGW empty string render as an empty reviewer.
*   The property remains present in the response, so the UI does not
*   receive an omitted or undefined property.
*
* Accepted examples:
*
*   FULL_NAME:
*     XSA=null, SEGW=""
*
*   DECIMAL_FORMAT:
*     XSA=null, SEGW=""
*
*   DATFM:
*     XSA=null, SEGW=""
*
*   USERNAME:
*     XSA=null, SEGW=""
*
*   NOTIF_24_HR:
*     XSA=null, SEGW=""
*
*   REVIWED_BY:
*     XSA=null, SEGW=""
*
*   COSTCENTERDESC:
*     XSA=null, SEGW=""
*
*   MANAGEUNIT:
*     XSA=null, SEGW=""
*
* Rejected examples:
*
*   FULL_NAME:
*     XSA=null, SEGW="John Smith"
*
*   NOTIF_24_HR:
*     XSA=null, SEGW="X"
*
*   REVIWED_BY:
*     XSA=null, SEGW="USER123"
*--------------------------------------------------------------------*
    WHEN 'COSTCENTERDESC'
      OR 'DATFM'
      OR 'DECIMAL_FORMAT'
      OR 'FULL_NAME'
      OR 'MANAGEUNIT'
      OR 'NOTIF_24_HR'
      OR 'REVIWED_BY'
      OR 'USERNAME'.

      rv_result =
        xsdbool(
          i_actual_token = '""'
        ).

*--------------------------------------------------------------------*
* Decimal properties serialized by SEGW as "0.00"
*
* EDGEWTDAMT
*   Business.controller.js and TBS.controller.js explicitly normalize
*   an XSA NULL value to the string "0.00" before passing the amount to
*   the accrual calculation:
*
*     rData.EDGEWTDAMT === null
*       ? "0.00"
*       : rData.EDGEWTDAMT
*
*   XSA NULL and SEGW "0.00" therefore produce the same calculation
*   input and the same functional UI behavior.
*
* GRVALUE_USD
*   The UI already normalizes an XSA NULL amount to 0.00 before currency
*   formatting. SEGW "0.00" therefore produces the same visible result.
*
* INVVALUE_USD
*   This converted/display amount is initial in SEGW when XSA has no
*   source value. The exception is intentionally limited to INVVALUE_USD.
*   INVVALUE remains strict because it participates in accrual arithmetic.
*
* INVVALUE_CAL
*   No active UI consumer was identified for this calculated helper
*   property. XSA NULL and the initial SEGW decimal value "0.00" therefore
*   have no functional impact in the current UI. The exception remains
*   field-specific and must not be generalized to INVVALUE.
*
* LASTMONTHWRKCOMP
*   XSA NULL displays as an empty value. SEGW "0.00" can display as
*   0.00 in the detail view.
*
*   This small presentation difference is explicitly accepted because
*   custom Gateway JSON serialization solely for this field would add
*   disproportionate implementation and maintenance complexity.
*
*   Backend presence information must remain separate from the numeric
*   value. A value of 0.00 must not be used as proof that the source
*   record existed.
*
* Accepted examples:
*
*   EDGEWTDAMT:
*     XSA=null, SEGW="0.00"
*
*   GRVALUE_USD:
*     XSA=null, SEGW="0.00"
*
*   INVVALUE_USD:
*     XSA=null, SEGW="0.00"
*
*   INVVALUE_CAL:
*     XSA=null, SEGW="0.00"
*
*   LASTMONTHWRKCOMP:
*     XSA=null, SEGW="0.00"
*
* Rejected examples:
*
*   EDGEWTDAMT:
*     XSA=null, SEGW="1.00"
*
*   GRVALUE_USD:
*     XSA=null, SEGW="-1.00"
*
*   LASTMONTHWRKCOMP:
*     XSA=null, SEGW="10.00"
*
* The following direction is also rejected:
*
*   XSA="0.00", SEGW=null
*--------------------------------------------------------------------*
    WHEN 'EDGEWTDAMT'
      OR 'GRVALUE_USD'
      OR 'INVVALUE_CAL'
      OR 'INVVALUE_USD'
      OR 'LASTMONTHWRKCOMP'.

      rv_result =
        xsdbool(
          i_actual_token = '"0.00"'
        ).

*--------------------------------------------------------------------*
* Time property serialized by SEGW as "PT00H00M00S"
*
* CHANGEDTIME
*   Initial ABAP TIMS value 000000 is also a valid midnight value.
*   Standard SEGW therefore serializes it as:
*
*     "PT00H00M00S"
*
*   XSA preserves an absent database value as:
*
*     null
*
*   The active UI primarily uses CHANGED_ON for the complete update
*   timestamp. The SEGW zero-time representation is therefore accepted
*   as a known compatibility deviation.
*
* Accepted example:
*
*   CHANGEDTIME:
*     XSA=null, SEGW="PT00H00M00S"
*
* Rejected examples:
*
*   CHANGEDTIME:
*     XSA=null, SEGW="PT10H15M00S"
*
*   CHANGEDTIME:
*     XSA="PT00H00M00S", SEGW=null
*--------------------------------------------------------------------*
    WHEN 'CHANGEDTIME'.

      rv_result =
        xsdbool(
          i_actual_token = '"PT00H00M00S"'
        ).

  ENDCASE.

ENDMETHOD.


*----------------------------------------------------------------------*
* Normalize an OData V2 Edm.Time duration token
*
* XSA and SEGW can serialize the same time with different zero padding:
*
*   XSA:  "PT9H40M11S"
*   SEGW: "PT09H40M11S"
*
*   XSA:  "PT8H24M4S"
*   SEGW: "PT08H24M04S"
*
* Valid PT...H...M...S values are returned in canonical two-digit form.
* Invalid, unsupported, NULL, or unrelated tokens are returned unchanged,
* allowing the normal strict comparison to report a real difference.
*----------------------------------------------------------------------*
  METHOD normalize_odata_time.

    DATA:
      lv_value       TYPE string,
      lv_remainder   TYPE string,
      lv_hour_text   TYPE string,
      lv_minute_text TYPE string,
      lv_second_text TYPE string,
      lv_hour        TYPE i,
      lv_minute      TYPE i,
      lv_second      TYPE i,
      lv_offset      TYPE i.

    rv_token =
      trim_text(
        i_text = i_token ).

    lv_value =
      unquote_json(
        i_token = rv_token ).

*--------------------------------------------------------------------*
* A supported value must start with PT and contain H, M, and S.
*--------------------------------------------------------------------*
    IF strlen( lv_value ) < 7
       OR lv_value(2) <> 'PT'
       OR lv_value NP 'PT*H*M*S'.
      RETURN.
    ENDIF.

    lv_remainder =
      substring(
        val = lv_value
        off = 2 ).

*--------------------------------------------------------------------*
* Extract hours.
*--------------------------------------------------------------------*
    FIND FIRST OCCURRENCE OF 'H'
      IN lv_remainder
      MATCH OFFSET lv_offset.

    IF sy-subrc <> 0
       OR lv_offset = 0.
      RETURN.
    ENDIF.

    lv_hour_text =
      substring(
        val = lv_remainder
        off = 0
        len = lv_offset ).

    lv_remainder =
      substring(
        val = lv_remainder
        off = lv_offset + 1 ).

*--------------------------------------------------------------------*
* Extract minutes.
*--------------------------------------------------------------------*
    FIND FIRST OCCURRENCE OF 'M'
      IN lv_remainder
      MATCH OFFSET lv_offset.

    IF sy-subrc <> 0
       OR lv_offset = 0.
      RETURN.
    ENDIF.

    lv_minute_text =
      substring(
        val = lv_remainder
        off = 0
        len = lv_offset ).

    lv_remainder =
      substring(
        val = lv_remainder
        off = lv_offset + 1 ).

*--------------------------------------------------------------------*
* Extract seconds and reject any trailing content.
*--------------------------------------------------------------------*
    FIND FIRST OCCURRENCE OF 'S'
      IN lv_remainder
      MATCH OFFSET lv_offset.

    IF sy-subrc <> 0
       OR lv_offset = 0
       OR lv_offset + 1 <> strlen( lv_remainder ).
      RETURN.
    ENDIF.

    lv_second_text =
      substring(
        val = lv_remainder
        off = 0
        len = lv_offset ).

*--------------------------------------------------------------------*
* Reject non-numeric components before conversion.
*--------------------------------------------------------------------*
    IF lv_hour_text CN '0123456789'
       OR lv_minute_text CN '0123456789'
       OR lv_second_text CN '0123456789'.
      RETURN.
    ENDIF.

    lv_hour =
      CONV i( lv_hour_text ).

    lv_minute =
      CONV i( lv_minute_text ).

    lv_second =
      CONV i( lv_second_text ).

*--------------------------------------------------------------------*
* Validate ordinary time-of-day ranges.
*--------------------------------------------------------------------*
    IF lv_hour > 23
       OR lv_minute > 59
       OR lv_second > 59.
      RETURN.
    ENDIF.

*--------------------------------------------------------------------*
* Canonical JSON token with two-digit components.
*--------------------------------------------------------------------*
    rv_token =
        '"PT'
      && |{ lv_hour WIDTH = 2 PAD = '0' ALIGN = RIGHT }|
      && 'H'
      && |{ lv_minute WIDTH = 2 PAD = '0' ALIGN = RIGHT }|
      && 'M'
      && |{ lv_second WIDTH = 2 PAD = '0' ALIGN = RIGHT }|
      && 'S"'.

  ENDMETHOD.

  METHOD equal_aggregate_content.

    DATA:
      lt_expected TYPE tt_string,
      lt_actual   TYPE tt_string.

*--------------------------------------------------------------------*
* Semantic unordered parity:
* - order of all segments is ignored;
* - leading, internal and trailing empty segments are preserved;
* - number of empty segments must still match.
*
* Therefore:
*   ;A;B     = A;B;
*   ;;A;B    = A;B;;
* but:
*   A;;B     <> A;B
*--------------------------------------------------------------------*
    lt_expected =
      split_aggregate_empty(
        i_value = i_expected ).

    lt_actual =
      split_aggregate_empty(
        i_value = i_actual ).

    IF lines( lt_expected ) <>
       lines( lt_actual ).

      FREE:
        lt_expected,
        lt_actual.

      RETURN.

    ENDIF.

    SORT lt_expected
      BY table_line.

    SORT lt_actual
      BY table_line.

    rv_equal =
      xsdbool(
        lt_expected = lt_actual ).

    FREE:
      lt_expected,
      lt_actual.

  ENDMETHOD.

  METHOD split_aggregate_empty.

    DATA:
      lv_length TYPE i,
      lv_start  TYPE i,
      lv_pos    TYPE i,
      lv_token  TYPE string.

*--------------------------------------------------------------------*
* Empty aggregate represents zero aggregate values, not one empty
* account-assignment slot. This follows the existing XSA convention.
*--------------------------------------------------------------------*
    IF i_value IS INITIAL.
      RETURN.
    ENDIF.

    lv_length =
      strlen(
        i_value ).

    WHILE lv_pos < lv_length.

      IF i_value+lv_pos(1) = ';'.

        lv_token =
          substring(
            val = i_value
            off = lv_start
            len = lv_pos - lv_start ).

        APPEND lv_token
          TO rt_tokens.

        CLEAR lv_token.

        lv_start =
          lv_pos + 1.

      ENDIF.

      lv_pos =
        lv_pos + 1.

    ENDWHILE.

*--------------------------------------------------------------------*
* Always append the final segment. If the source ends in ';', this is
* deliberately an empty token.
*--------------------------------------------------------------------*
    lv_token =
      substring(
        val = i_value
        off = lv_start
        len = lv_length - lv_start ).

    APPEND lv_token
      TO rt_tokens.

  ENDMETHOD.


  METHOD read_json_string_end.

    DATA:
      lv_position TYPE i,
      lv_length   TYPE i,
      lv_char     TYPE char1,
      lv_escape   TYPE abap_bool.

    lv_length =
      strlen(
        i_text ).

    IF i_start >= lv_length
       OR i_text+i_start(1) <> '"'.

      RAISE EXCEPTION TYPE zcx_fi_das_error
        EXPORTING
          i_reason =
                     'JSON string does not start with quote'.

    ENDIF.

    lv_position =
      i_start + 1.

    WHILE lv_position < lv_length.

      lv_char =
        i_text+lv_position(1).

      IF lv_escape = abap_true.

        CLEAR lv_escape.

      ELSEIF lv_char = '\'.

        lv_escape =
          abap_true.

      ELSEIF lv_char = '"'.

        rv_end =
          lv_position.

        RETURN.

      ENDIF.

      lv_position =
        lv_position + 1.

    ENDWHILE.

    RAISE EXCEPTION TYPE zcx_fi_das_error
      EXPORTING
        i_reason =
                   'JSON string is not closed'.

  ENDMETHOD.


  METHOD read_json_value_end.

    DATA:
      lv_position TYPE i,
      lv_length   TYPE i,
      lv_char     TYPE char1,
      lv_end      TYPE i.

    lv_length =
      strlen(
        i_text ).

    IF i_start >= lv_length.

      RAISE EXCEPTION TYPE zcx_fi_das_error
        EXPORTING
          i_reason =
                     'JSON value is missing'.

    ENDIF.

    lv_char =
      i_text+i_start(1).

    IF lv_char = '"'.

      lv_end =
        read_json_string_end(
          i_text  = i_text
          i_start = i_start ).

      rv_end =
        lv_end + 1.

      RETURN.

    ENDIF.

    IF lv_char = '{'.

      lv_end =
        find_matching_end(
          i_text  = i_text
          i_start = i_start
          i_open  = '{'
          i_close = '}' ).

      rv_end =
        lv_end + 1.

      RETURN.

    ENDIF.

    IF lv_char = '['.

      lv_end =
        find_matching_end(
          i_text  = i_text
          i_start = i_start
          i_open  = '['
          i_close = ']' ).

      rv_end =
        lv_end + 1.

      RETURN.

    ENDIF.

    lv_position =
      i_start.

    WHILE lv_position < lv_length.

      lv_char =
        i_text+lv_position(1).

      IF lv_char = ','
         OR lv_char = '}'.

        rv_end =
          lv_position.

        RETURN.

      ENDIF.

      lv_position =
        lv_position + 1.

    ENDWHILE.

    rv_end =
      lv_length.

  ENDMETHOD.


  METHOD find_matching_end.

    DATA:
      lv_position TYPE i,
      lv_length   TYPE i,
      lv_depth    TYPE i,
      lv_char     TYPE char1,
      lv_in_text  TYPE abap_bool,
      lv_escape   TYPE abap_bool.

    lv_length =
      strlen(
        i_text ).

    lv_position =
      i_start.

    WHILE lv_position < lv_length.

      lv_char =
        i_text+lv_position(1).

      IF lv_in_text = abap_true.

        IF lv_escape = abap_true.

          CLEAR lv_escape.

        ELSEIF lv_char = '\'.

          lv_escape =
            abap_true.

        ELSEIF lv_char = '"'.

          CLEAR lv_in_text.

        ENDIF.

      ELSE.

        IF lv_char = '"'.

          lv_in_text =
            abap_true.

        ELSEIF lv_char = i_open.

          lv_depth =
            lv_depth + 1.

        ELSEIF lv_char = i_close.

          lv_depth =
            lv_depth - 1.

          IF lv_depth = 0.

            rv_end =
              lv_position.

            RETURN.

          ENDIF.

        ENDIF.

      ENDIF.

      lv_position =
        lv_position + 1.

    ENDWHILE.

    RAISE EXCEPTION TYPE zcx_fi_das_error
      EXPORTING
        i_reason =
                   'JSON object or array is not closed'.

  ENDMETHOD.

  METHOD skip_whitespace.

    DATA:
      lv_length TYPE i,
      lv_char   TYPE char1.

    rv_position =
      i_position.

    lv_length =
      strlen(
        i_text ).

    WHILE rv_position < lv_length.

      lv_char =
        i_text+rv_position(1).

      IF is_whitespace(
           i_char = lv_char ) = abap_false.

        RETURN.

      ENDIF.

      rv_position =
        rv_position + 1.

    ENDWHILE.

  ENDMETHOD.


  METHOD is_whitespace.

    IF i_char = space
       OR i_char = cl_abap_char_utilities=>horizontal_tab
       OR i_char = cl_abap_char_utilities=>newline
       OR i_char = cl_abap_char_utilities=>cr_lf+0(1)
       OR i_char = cl_abap_char_utilities=>cr_lf+1(1).

      rv_result =
        abap_true.

    ENDIF.

  ENDMETHOD.


  METHOD trim_text.

    DATA:
      lv_start         TYPE i,
      lv_end           TYPE i,
      lv_length        TYPE i,
      lv_last_position TYPE i,
      lv_char          TYPE char1.

    lv_length =
      strlen(
        i_text ).

    lv_end =
      lv_length.

    WHILE lv_start < lv_length.

      lv_char =
        i_text+lv_start(1).

      IF is_whitespace(
           i_char = lv_char ) = abap_false.

        EXIT.

      ENDIF.

      lv_start =
        lv_start + 1.

    ENDWHILE.

    WHILE lv_end > lv_start.

      lv_last_position =
        lv_end - 1.

      lv_char =
        i_text+lv_last_position(1).

      IF is_whitespace(
           i_char = lv_char ) = abap_false.

        EXIT.

      ENDIF.

      lv_end =
        lv_end - 1.

    ENDWHILE.

    IF lv_end > lv_start.

      rv_text =
        substring(
          val = i_text
          off = lv_start
          len = lv_end - lv_start ).

    ENDIF.

  ENDMETHOD.


  METHOD unquote_json.

    DATA:
      lv_token        TYPE string,
      lv_length       TYPE i,
      lv_last_offset  TYPE i,
      lv_inner_length TYPE i,
      lv_first_char   TYPE char1,
      lv_last_char    TYPE char1.

    lv_token =
      trim_text(
        i_text = i_token ).

    lv_length =
      strlen(
        lv_token ).

    IF lv_length < 2.

      rv_value =
        lv_token.

      RETURN.

    ENDIF.

    lv_last_offset =
      lv_length - 1.

    lv_first_char =
      lv_token(1).

    lv_last_char =
      lv_token+lv_last_offset(1).

    IF lv_first_char <> '"'
       OR lv_last_char <> '"'.

      rv_value =
        lv_token.

      RETURN.

    ENDIF.

    lv_inner_length =
      lv_length - 2.

    rv_value =
      substring(
        val = lv_token
        off = 1
        len = lv_inner_length ).

*--------------------------------------------------------------------*
* Avoid replacement scans in the normal case where the JSON value has
* no escaped characters.
*--------------------------------------------------------------------*
    IF rv_value NS '\'.
      RETURN.
    ENDIF.

    REPLACE ALL OCCURRENCES OF '\u0026'
      IN rv_value
      WITH '&'.

    REPLACE ALL OCCURRENCES OF '\u003C'
      IN rv_value
      WITH '<'.

    REPLACE ALL OCCURRENCES OF '\u003c'
      IN rv_value
      WITH '<'.

    REPLACE ALL OCCURRENCES OF '\u003E'
      IN rv_value
      WITH '>'.

    REPLACE ALL OCCURRENCES OF '\u003e'
      IN rv_value
      WITH '>'.

    REPLACE ALL OCCURRENCES OF '\u0022'
      IN rv_value
      WITH '"'.

    REPLACE ALL OCCURRENCES OF '\u0027'
      IN rv_value
      WITH ''''.

    REPLACE ALL OCCURRENCES OF '\/'
      IN rv_value
      WITH '/'.

    REPLACE ALL OCCURRENCES OF '\"'
      IN rv_value
      WITH '"'.

    REPLACE ALL OCCURRENCES OF '\n'
      IN rv_value
      WITH cl_abap_char_utilities=>newline.

    REPLACE ALL OCCURRENCES OF '\r'
      IN rv_value
      WITH cl_abap_char_utilities=>cr_lf+0(1).

    REPLACE ALL OCCURRENCES OF '\t'
      IN rv_value
      WITH cl_abap_char_utilities=>horizontal_tab.

    REPLACE ALL OCCURRENCES OF '\\'
      IN rv_value
      WITH '\'.

  ENDMETHOD.


  METHOD add_difference.

    mv_difference_count =
      mv_difference_count + 1.

*--------------------------------------------------------------------*
* Count every mismatch, but retain a bounded diagnostic sample. This
* prevents failures with many missing rows from allocating huge logs.
*--------------------------------------------------------------------*
    IF lines( mt_differences ) >= mc_max_log.
      RETURN.
    ENDIF.

    APPEND i_text
      TO mt_differences.

  ENDMETHOD.


  METHOD assert_no_differences.

    DATA:
      lv_text   TYPE string,
      lv_line   TYPE string,
      lv_shown  TYPE i,
      lv_hidden TYPE i.

    IF mv_difference_count = 0.

      cl_abap_unit_assert=>assert_true(
        act = abap_true ).

      RETURN.

    ENDIF.

    lv_shown =
      lines(
        mt_differences ).

    lv_hidden =
      mv_difference_count
      - lv_shown.

    lv_text =
        |DAS XSA vs SEGW OData parity failed. |
      && |Difference count: { mv_difference_count }.|
      && cl_abap_char_utilities=>newline
      && |--- First { lv_shown } difference(s) ---|.

    LOOP AT mt_differences INTO lv_line.

      lv_text =
          lv_text
        && cl_abap_char_utilities=>newline
        && lv_line.

    ENDLOOP.

    IF lv_hidden > 0.

      lv_text =
          lv_text
        && cl_abap_char_utilities=>newline
        && |--- { lv_hidden } further difference(s) omitted ---|.

    ENDIF.

    cl_abap_unit_assert=>fail(
      msg = lv_text ).

  ENDMETHOD.



ENDCLASS.
