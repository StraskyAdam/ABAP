" ============================================================================
" XSA vs SEGW functional JSON parity test - paged low-memory implementation
"
" Memory strategy:
" - XSA fixture is indexed once with PO/item and object character offsets only.
" - Raw SEGW JSON is fetched one SKIP/TOP page at a time.
" - JSON properties are read directly from both raw payloads for each matched row.
" - Comparison is strict except for documented, field-specific SEGW mappings of
"   XSA SQL NULL values and a temporary unordered-aggregate compatibility rule.
" - No complete XSA row table, no complete property tree, and no copied JSON
"   objects are held in memory.
"
" Paging relies on the stable ORDER BY implemented by all dashboard flows.
" ============================================================================

  METHOD run_paged_case.

    DATA:
      lv_expected_json   TYPE string,
      lv_actual_json     TYPE string,
      lt_expected_index  TYPE tt_expected_index,
      lt_actual_index    TYPE tt_expected_index,
      lv_skip            TYPE i,
      lv_page_rows       TYPE i,
      lv_total_count     TYPE int8,
      lv_finished        TYPE abap_bool.

    lv_expected_json =
      read_mime_fixture(
        i_fixture = is_case-fixture ).

    lt_expected_index =
      build_expected_index(
        i_json = lv_expected_json ).

    IF i_complex_filter = abap_true OR i_cross_or_filter = abap_true.
      DATA lv_cross_or_witnesses TYPE i.
      LOOP AT lt_expected_index ASSIGNING FIELD-SYMBOL(<ls_candidate>).
        DATA(lt_properties) = read_object_properties(
          i_json = lv_expected_json i_object_from = <ls_candidate>-object_from
          i_object_to = <ls_candidate>-object_to ).
        DATA lv_matches TYPE abap_bool.
        IF i_complex_filter = abap_true.
          lv_matches = matches_complex_fixture( lt_properties ).
        ELSE.
          cl_abap_unit_assert=>assert_true(
            act = xsdbool( line_exists( lt_properties[ name = 'SUPPLIERNAME' ] ) ) ).
          DATA(lv_supplier_token) = lt_properties[ name = 'SUPPLIERNAME' ]-token.
          DATA(lv_supplier) = unquote_json( lv_supplier_token ).
          lv_matches = xsdbool(
            <ls_candidate>-ponumber = '8000401022'
            OR find( val = lv_supplier sub = 'Fuji' case = abap_true ) >= 0 ).
          IF lv_matches = abap_true AND <ls_candidate>-ponumber <> '8000401022'.
            lv_cross_or_witnesses = lv_cross_or_witnesses + 1.
          ENDIF.
        ENDIF.
        IF lv_matches = abap_false.
          DELETE TABLE lt_expected_index FROM <ls_candidate>.
        ENDIF.
      ENDLOOP.
      FREE lt_properties.
      cl_abap_unit_assert=>assert_not_initial(
        act = lt_expected_index
        msg = 'The real fixture must contain positive complete-predicate witnesses' ).
      IF i_cross_or_filter = abap_true.
        cl_abap_unit_assert=>assert_true(
          act = xsdbool( lv_cross_or_witnesses > 0 )
          msg = 'Cross-property OR must retain supplier witnesses outside the projected PO key' ).
      ENDIF.
    ENDIF.

    WHILE lv_finished = abap_false.

      CLEAR:
        lv_actual_json,
        lv_total_count,
        lv_page_rows.

      fetch_gateway_page(
        EXPORTING
          is_case       = is_case
          i_skip        = lv_skip
        IMPORTING
          e_json        = lv_actual_json
          e_total_count = lv_total_count ).

      lt_actual_index =
        build_expected_index(
          i_json = lv_actual_json ).

      lv_page_rows = get_result_count( lv_actual_json ).
      cl_abap_unit_assert=>assert_true(
        act = xsdbool( lv_page_rows <= mc_page_size )
        msg = 'Fixture parity must retain at most one bounded Gateway page' ).

      compare_actual_page(
        EXPORTING
          i_fixture       = is_case-fixture
          i_expected_json = lv_expected_json
          i_actual_json   = lv_actual_json
          it_actual_index = lt_actual_index
        CHANGING
          ct_expected_index = lt_expected_index ).

      IF  lv_total_count <> lines( lt_expected_index ).
        add_difference(
          i_text =
              |Inline count differs; Fixture={ is_case-fixture }; |
            && |XSA={ lines( lt_expected_index ) }; |
            && |SEGW={ lv_total_count }| ).
      ENDIF.

      IF lv_page_rows = 0.
        lv_finished = abap_true.
      ENDIF.

      lv_skip = lv_skip + lv_page_rows.

      IF lv_total_count > 0.
        IF lv_skip >= lv_total_count.
          lv_finished = abap_true.
        ENDIF.
      ELSEIF lv_page_rows < mc_page_size.
        lv_finished = abap_true.
      ENDIF.

      IF lv_skip > lines( lt_expected_index ) + mc_page_size.
        add_difference( |Paging exceeded the fixture cardinality; Fixture={ is_case-fixture }| ).
        lv_finished = abap_true.
      ENDIF.
      FREE: lt_actual_index, lv_actual_json.

    ENDWHILE.

    LOOP AT lt_expected_index
      ASSIGNING FIELD-SYMBOL(<ls_expected>)
      WHERE matched = abap_false.

      add_difference(
        i_text =
            |Missing SEGW raw JSON row; Fixture={ is_case-fixture }; |
          && |PO={ <ls_expected>-ponumber }; |
          && |ITEMNO={ <ls_expected>-itemno }| ).

    ENDLOOP.

    FREE:
      lt_expected_index,
      lt_actual_index,
      lv_expected_json,
      lv_actual_json.

  ENDMETHOD.

    CASE is_case-flow.

        rv_path =
            '/sap/opu/odata/sap/ZFI_DAS_SRV/'
          && |thresholdParameters(IP_PSTYPE='{ lv_pstype }',|
          && |IP_EMAIL='{ lv_email }',|
          && |IP_POOWNER='{ lv_poowner }')/Results|.

      WHEN zif_fi_das_dashboard=>mc_exclude.
        rv_path =
            '/sap/opu/odata/sap/ZFI_DAS_SRV/'
          && |exclusionPOParameters(IP_EMAIL='{ lv_email }',|
          && |IP_POOWNER='{ lv_poowner }')/Results|.
    ENDCASE.

    rv_path =
        rv_path
      && |?$format=json|.

    IF is_case-inlinecount = abap_true
       OR is_case-fixture IS NOT INITIAL.
      rv_path = rv_path && |&$inlinecount=allpages|.
    ENDIF.

    IF ( is_case-top_specified = abap_true AND is_case-top > 0 )
       OR ( is_case-top_specified = abap_true
            AND is_case-top = 0
            AND is_case-fixture IS INITIAL )
       OR is_case-fixture IS NOT INITIAL.
      rv_path = rv_path && |&$top={ lv_top }|.
    ENDIF.

    rv_path = rv_path && |&$skip={ i_skip }|.

    IF is_case-orderby IS NOT INITIAL.

      lv_orderby = trim_text( is_case-orderby ).
      lv_orderby_upper = to_upper( trim_text( lv_orderby ) ).

      IF is_case-orderby_desc = abap_true
         AND lv_orderby_upper NP '* DESC'
         AND lv_orderby_upper NP '* DESCENDING'
         AND lv_orderby_upper NP '* ASC'
         AND lv_orderby_upper NP '* ASCENDING'.
        lv_orderby_direction = ` desc`.
      ENDIF.

      lv_orderby = lv_orderby && lv_orderby_direction.

      cl_http_utility=>escape_url(
        EXPORTING
          unescaped = lv_orderby
        RECEIVING
          escaped   = lv_orderby ).

      rv_path = rv_path && |&$orderby={ lv_orderby }|.

    ENDIF.

    IF is_case-select_list IS NOT INITIAL.
      rv_path = rv_path && |&$select={ is_case-select_list }|.
    ENDIF.

    append_filter(
      EXPORTING
        is_case = is_case
      CHANGING
        cv_path = rv_path ).

  ENDMETHOD.

  METHOD append_filter.
    DATA: lt_properties TYPE tt_property_name,
          lt_clauses TYPE stringtab,
          lt_include TYPE stringtab,
          lt_exclude TYPE stringtab,
          lv_part TYPE string,
          lv_group TYPE string.
    LOOP AT is_case-filter_select_options INTO DATA(ls_filter).
      INSERT VALUE #( name = ls_filter-property ) INTO TABLE lt_properties.
    ENDLOOP.
    LOOP AT lt_properties INTO DATA(ls_property).
      CLEAR: lt_include, lt_exclude.
      LOOP AT is_case-filter_select_options INTO ls_filter
        WHERE property = ls_property-name.
        DATA(lv_low) = odata_quote( ls_filter-low ).
        DATA(lv_high) = odata_quote( ls_filter-high ).
        CASE ls_filter-option.
          WHEN 'EQ'.
            lv_part = |{ ls_property-name } eq '{ lv_low }'|.
          WHEN 'NE'.
            lv_part = |{ ls_property-name } ne '{ lv_low }'|.
          WHEN 'BT'.
            lv_part = |({ ls_property-name } ge '{ lv_low }' and |
                   && |{ ls_property-name } le '{ lv_high }')|.
          WHEN 'CP'.
            " Gateway CP contains patterns have only outer asterisks.
            SHIFT lv_low LEFT DELETING LEADING '*'.
            SHIFT lv_low RIGHT DELETING TRAILING '*'.
            cl_abap_unit_assert=>assert_false(
              act = xsdbool( lv_low CS '*' OR lv_low CS '+' )
              msg = 'This fixture URI helper supports only substring CP patterns' ).
            lv_part = |substringof('{ lv_low }',{ ls_property-name })|.
          WHEN OTHERS.
            cl_abap_unit_assert=>fail( msg = |Unsupported fixture range option { ls_filter-option }| ).
        ENDCASE.
        IF ls_filter-sign = 'E'.
          APPEND |not ({ lv_part })| TO lt_exclude.
        ELSE.
          APPEND lv_part TO lt_include.
        ENDIF.
      ENDLOOP.
      IF lt_include IS NOT INITIAL.
        CONCATENATE LINES OF lt_include INTO lv_group SEPARATED BY ' or '.
        APPEND |({ lv_group })| TO lt_clauses.
      ENDIF.
      APPEND LINES OF lt_exclude TO lt_clauses.
    ENDLOOP.
    IF is_case-filter_raw IS NOT INITIAL.
      APPEND |({ is_case-filter_raw })| TO lt_clauses.
    ENDIF.
    IF lt_clauses IS NOT INITIAL.
      CONCATENATE LINES OF lt_clauses INTO lv_group SEPARATED BY ' and '.
      cv_path = cv_path && |&$filter={ cl_http_utility=>escape_url( lv_group ) }|.
    ENDIF.


  ENDMETHOD.

  METHOD odata_quote.

    rv_value = CONV string( i_value ).
    SHIFT rv_value RIGHT DELETING TRAILING space.

    REPLACE ALL OCCURRENCES OF ''''
      IN rv_value
      WITH ''''''.

  ENDMETHOD.

  METHOD read_inline_count.

    DATA:
      lv_offset TYPE i,
      lv_start  TYPE i,
      lv_end    TYPE i,
      lv_length TYPE i,
      lv_token  TYPE string.

    FIND FIRST OCCURRENCE OF '"__count"'
      IN i_json
      MATCH OFFSET lv_offset.

    IF sy-subrc <> 0.
      RETURN.
    ENDIF.

    lv_length = strlen( i_json ).
    lv_start = lv_offset + 9.

    WHILE lv_start < lv_length
      AND i_json+lv_start(1) <> ':'.
      lv_start = lv_start + 1.
    ENDWHILE.

    lv_start = skip_whitespace(
      i_text     = i_json
      i_position = lv_start + 1 ).

    lv_end = read_json_value_end(
      i_text  = i_json
      i_start = lv_start ).

    lv_token = substring(
      val = i_json
      off = lv_start
      len = lv_end - lv_start ).

    lv_token = unquote_json(
      i_token = lv_token ).

    IF lv_token IS NOT INITIAL.
      rv_count = CONV int8( lv_token ).
    ENDIF.

  ENDMETHOD.

  METHOD assert_rows_sorted.

  DATA:
    lv_row_count  TYPE i,
    lv_row_index  TYPE i,
    lv_previous   TYPE string,
    lv_current    TYPE string,
    lv_previous_i TYPE i,
    lv_current_i  TYPE i,
    lv_previous_n TYPE decfloat34,
    lv_current_n  TYPE decfloat34.

  IF i_property <> 'PSTYPE'
     AND i_property <> 'SUPPLIERNAME'
     AND i_property <> 'AMOUNTACCRUED'
     AND i_property <> 'REVIWED_BY'.

    cl_abap_unit_assert=>fail(
      msg = |assert_rows_sorted does not support { i_property }| ).

  ENDIF.

  lv_row_count =
    get_result_count( i_json ).

  WHILE lv_row_index < lv_row_count.

    lv_row_index = lv_row_index + 1.

    DATA(lv_token) =
      read_result_property(
        i_json     = i_json
        i_index    = lv_row_index
        i_property = i_property ).

    lv_current =
      unquote_json(
        i_token = lv_token ).

    IF to_upper( lv_current ) = 'NULL'.
      CLEAR lv_current.
    ENDIF.

    " JSON strings do not contain the fixed-length ABAP trailing
    " spaces, but remove surrounding whitespace consistently.
    lv_current = trim_text( lv_current ).

    IF lv_row_index > 1.

      IF i_property = 'PSTYPE'.

        lv_previous_i =
          COND i(
            WHEN lv_previous IS INITIAL
            THEN 0
            ELSE CONV i( lv_previous ) ).

        lv_current_i =
          COND i(
            WHEN lv_current IS INITIAL
            THEN 0
            ELSE CONV i( lv_current ) ).

        IF i_descending = abap_true.

          cl_abap_unit_assert=>assert_true(
            act = xsdbool( lv_previous_i >= lv_current_i )
            msg =
              |PSTYPE descending violation at row { lv_row_index }: |
              && |previous="{ lv_previous }", current="{ lv_current }"| ).

        ELSE.

          cl_abap_unit_assert=>assert_true(
            act = xsdbool( lv_previous_i <= lv_current_i )
            msg =
              |PSTYPE ascending violation at row { lv_row_index }: |
              && |previous="{ lv_previous }", current="{ lv_current }"| ).

        ENDIF.

      ELSEIF i_property = 'AMOUNTACCRUED'.

        lv_previous_n =
          COND decfloat34(
            WHEN lv_previous IS INITIAL
            THEN 0
            ELSE CONV decfloat34( lv_previous ) ).

        lv_current_n =
          COND decfloat34(
            WHEN lv_current IS INITIAL
            THEN 0
            ELSE CONV decfloat34( lv_current ) ).

        IF i_descending = abap_true.

          cl_abap_unit_assert=>assert_true(
            act = xsdbool( lv_previous_n >= lv_current_n )
            msg =
              |AMOUNTACCRUED descending violation at row { lv_row_index }: |
              && |previous="{ lv_previous }", current="{ lv_current }"| ).

        ELSE.

          cl_abap_unit_assert=>assert_true(
            act = xsdbool( lv_previous_n <= lv_current_n )
            msg =
              |AMOUNTACCRUED ascending violation at row { lv_row_index }: |
              && |previous="{ lv_previous }", current="{ lv_current }"| ).

        ENDIF.

      ELSE.

        " Do not uppercase: production ABAP SORT uses the original
        " character values. The assertion must use the same values.
        IF i_descending = abap_true.

          cl_abap_unit_assert=>assert_true(
            act = xsdbool( lv_previous >= lv_current )
            msg =
              |{ i_property } descending violation at row { lv_row_index }: |
              && |previous="{ lv_previous }", current="{ lv_current }"| ).

        ELSE.

          cl_abap_unit_assert=>assert_true(
            act = xsdbool( lv_previous <= lv_current )
            msg =
              |{ i_property } ascending violation at row { lv_row_index }: |
              && |previous="{ lv_previous }", current="{ lv_current }"| ).

        ENDIF.

      ENDIF.

    ENDIF.

    lv_previous = lv_current.

  ENDWHILE.

ENDMETHOD.

  METHOD build_expected_index.

    DATA:
      lv_array_from  TYPE i,
      lv_array_to    TYPE i,
      lv_position    TYPE i,
      lv_object_from TYPE i,
      lv_object_to   TYPE i,
      lv_ponumber    TYPE string,
      lv_itemno      TYPE string,
      lv_char        TYPE char1,
      ls_expected    TYPE ts_expected_index.

    CLEAR rt_index.

    find_results_array_bounds(
      EXPORTING
        i_json = i_json
      IMPORTING
        e_array_from = lv_array_from
        e_array_to   = lv_array_to ).

    lv_position =
      lv_array_from + 1.

    WHILE lv_position < lv_array_to.

      lv_position =
        skip_whitespace(
          i_text     = i_json
          i_position = lv_position ).

      IF lv_position >= lv_array_to.
        EXIT.
      ENDIF.

      lv_char =
        i_json+lv_position(1).

      IF lv_char = ','.

        lv_position =
          lv_position + 1.

        CONTINUE.

      ENDIF.

      IF lv_char <> '{'.

        RAISE EXCEPTION TYPE zcx_fi_das_error
          EXPORTING
            i_reason =
                       |Unexpected character in results array at offset { lv_position }|.

      ENDIF.

      lv_object_from =
        lv_position.

      lv_object_to =
        find_matching_end(
          i_text  = i_json
          i_start = lv_object_from
          i_open  = '{'
          i_close = '}' ).

      IF lv_object_to > lv_array_to.

        RAISE EXCEPTION TYPE zcx_fi_das_error
          EXPORTING
            i_reason =
                       'JSON result object exceeds the results array'.

      ENDIF.

      CLEAR:
        lv_ponumber,
        lv_itemno.

      read_json_row_key(
        EXPORTING
          i_json        = i_json
          i_object_from = lv_object_from
          i_object_to   = lv_object_to
        IMPORTING
          e_ponumber    = lv_ponumber
          e_itemno      = lv_itemno ).

      IF lv_ponumber IS INITIAL
         OR lv_itemno IS INITIAL.

        RAISE EXCEPTION TYPE zcx_fi_das_error
          EXPORTING
            i_reason =
                       'JSON row does not contain PONUMBER and ITEMNO'.

      ENDIF.

  METHOD read_json_row_key.

    DATA:

      lv_key_start   TYPE i,
      lv_key_end     TYPE i,

      lv_value_end   TYPE i,

         OR i_json+lv_position(1) <> ':'.

                       |JSON property { lv_key } has no colon|.

      ENDIF.

        skip_whitespace(
          i_text     = i_json
          i_position = lv_position + 1 ).

        lv_position.

        read_json_value_end(
          i_text  = i_json
          i_start = lv_value_start ).

        RAISE EXCEPTION TYPE zcx_fi_das_error
          EXPORTING
            i_reason =
                       |JSON property { lv_key } exceeds its object boundary|.

      ENDIF.

*--------------------------------------------------------------------*
* Only create a value substring for the two key properties.
*--------------------------------------------------------------------*
      IF lv_key = 'PONUMBER'
         OR lv_key = 'ITEMNO'.

        lv_token =
          substring(
            val = i_json
            off = lv_value_start
            len = lv_value_end - lv_value_start ).

            unquote_json(
              i_token = lv_token ).

        ELSE.

          e_itemno =
            unquote_json(
              i_token = lv_token ).

        ENDIF.

        FREE lv_token.

        IF e_ponumber IS NOT INITIAL
           AND e_itemno IS NOT INITIAL.

          RETURN.

        ENDIF.

      ENDIF.

      CLEAR lv_key.

      lv_position =
        lv_value_end + 1.

    ENDWHILE.

  METHOD compare_raw_property.

    DATA:
      lv_expected_token TYPE string,
      lv_actual_token   TYPE string,
      lv_expected_text  TYPE string,
      lv_actual_text    TYPE string,
      lv_expected_len   TYPE i,
      lv_actual_len     TYPE i.

    lv_expected_token =
      trim_text(
        i_text = i_expected_token ).

    lv_actual_token =
      trim_text(
        i_text = i_actual_token ).

*--------------------------------------------------------------------*
* Documented functional compatibility for XSA SQL NULL values
*
* Classic SEGW serializes initial ABAP character, decimal, and time
* values differently from XSA SQL NULL. IS_EQUIVALENT_NULL contains a
* field-specific allowlist. There is intentionally no global rule such
* as NULL = empty string or NULL = zero.
*--------------------------------------------------------------------*
    IF is_equivalent_null(
         i_name           = i_name
         i_expected_token = lv_expected_token
         i_actual_token   = lv_actual_token ) = abap_true.
      RETURN.
    ENDIF.

*--------------------------------------------------------------------*
* Normalize non-NULL CHANGEDTIME values
*
* XSA can omit leading zeroes while SEGW emits two-digit components.
* Both representations describe the same Edm.Time value:
*
*   "PT9H40M11S" = "PT09H40M11S"
*   "PT8H24M4S"  = "PT08H24M04S"
*
* Only CHANGEDTIME enters this normalization. The actual hour, minute,
* and second values remain strict.
*--------------------------------------------------------------------*
    IF i_name = 'CHANGEDTIME'.
      lv_expected_token =
        normalize_odata_time(
          i_token = lv_expected_token ).

      lv_actual_token =
        normalize_odata_time(
          i_token = lv_actual_token ).
    ENDIF.

*--------------------------------------------------------------------*
* TODO_PARITY_STRICT_001 - temporary aggregate-order exception
*
* Aggregate segment order can currently differ between XSA and SEGW.
* Both values are decoded first. Content and multiplicity must still be
* identical after ordering is ignored.
*
* Accepted:
*   XSA="A;B;C", SEGW="C;A;B"
*
* Rejected:
*   XSA="A;B;C", SEGW="A;B"
*   XSA="A;A;B", SEGW="A;B"
*
* Remove this branch after aggregate ordering is deterministic and
* aligned between both implementations.
*--------------------------------------------------------------------*
    IF is_unordered_aggregate_field(
         i_name = i_name ) = abap_true.

      lv_expected_text =
        unquote_json(
          i_token = lv_expected_token ).

      lv_actual_text =
        unquote_json(
          i_token = lv_actual_token ).

      IF equal_aggregate_content(
           i_expected = lv_expected_text
           i_actual   = lv_actual_text ) = abap_false.
        add_difference(
          i_text =
              |Aggregate content differs; Fixture={ i_fixture }; |
            && |PO={ i_ponumber }; ITEMNO={ i_itemno }; |
            && |Field={ i_name }; XSA={ lv_expected_token }; |
            && |SEGW={ lv_actual_token }| ).
      ENDIF.

      RETURN.
    ENDIF.

*--------------------------------------------------------------------*
* Compare decoded JSON string values
*
* Different JSON tokens can represent exactly the same string received
* by the UI. UNQUOTE_JSON removes the outer quotes and decodes supported
* JSON escape sequences.
*
* Accepted examples:
*   XSA="S \u0026 K",              SEGW="S & K"
*   XSA="King\u0027s College",    SEGW="King's College"
*   XSA="\/Date(1677240711333)\/",
*   SEGW="/Date(1677240711333)/"
*
* Rejected examples:
*   XSA="King's College", SEGW="Kings College"
*   XSA="A & B",          SEGW="A and B"
*
* NULL, numbers, booleans, arrays, and objects do not enter this branch
* and remain subject to strict token comparison below.
*--------------------------------------------------------------------*
    lv_expected_len =
      strlen( lv_expected_token ).

    lv_actual_len =
      strlen( lv_actual_token ).

    IF lv_expected_len >= 2
       AND lv_actual_len >= 2
       AND lv_expected_token(1) = '"'
       AND lv_actual_token(1) = '"'.

      lv_expected_text =
        unquote_json(
          i_token = lv_expected_token ).

      lv_actual_text =
        unquote_json(
          i_token = lv_actual_token ).

      IF lv_expected_text <> lv_actual_text.
        add_difference(
          i_text =
              |String value differs; Fixture={ i_fixture }; |
            && |PO={ i_ponumber }; ITEMNO={ i_itemno }; |
            && |Field={ i_name }; XSA={ lv_expected_token }; |
            && |SEGW={ lv_actual_token }| ).
      ENDIF.

      RETURN.
    ENDIF.

*--------------------------------------------------------------------*
* Strict comparison for every remaining JSON token
*--------------------------------------------------------------------*
    IF lv_expected_token <> lv_actual_token.
      add_difference(
        i_text =
            |Raw value differs; Fixture={ i_fixture }; |
          && |PO={ i_ponumber }; ITEMNO={ i_itemno }; |
          && |Field={ i_name }; XSA={ lv_expected_token }; |
          && |SEGW={ lv_actual_token }| ).
    ENDIF.

  ENDMETHOD.

*----------------------------------------------------------------------*
  METHOD is_equivalent_null.

*--------------------------------------------------------------------*
* TODO_PARITY_FUNCTIONAL_NULL_001
*
* Functional compatibility between XSA SQL NULL values and standard
* SEGW serialization of initial ABAP values.
*
* Background
* ----------
* XSA receives database NULL values directly from SAP HANA and emits:
*
*   null
*
* Classic SEGW receives typed ABAP values. Elementary ABAP value types
* do not preserve the distinction between an absent source value and an
* explicitly supplied initial value.
*
* Standard SEGW therefore serializes initial ABAP values as follows:
*
*   Character value  -> ""
*   Decimal value    -> "0.00"
*   TIMS value       -> "PT00H00M00S"
*
* Scope
* -----
* The rules below are deliberately field-specific. Do not replace them
* with global rules such as:
*
*   null = ""
*   null = "0.00"
*   null = "PT00H00M00S"
*
* NULL and an initial value can have different business meanings for
* other properties.
*
* The exceptions are also one-directional:
*
*   XSA null -> approved SEGW initial representation
*
* A non-NULL XSA value remains subject to strict comparison.
*
* The following examples are therefore NOT accepted:
*
*   XSA="0.00", SEGW=null
*   XSA=null,   SEGW="1.00"
*   XSA=null,   SEGW="X"
*
* Review this allowlist whenever UI consumption or the OData contract
* changes.
*--------------------------------------------------------------------*
  IF i_expected_token <> 'null'.
    RETURN.
  ENDIF.

*--------------------------------------------------------------------*

*--------------------------------------------------------------------*
* Decimal properties serialized by SEGW as "0.00"
*
* EDGEWTDAMT
*   Business.controller.js and TBS.controller.js explicitly normalize
*   an XSA NULL value to the string "0.00" before passing the amount to
*   the accrual calculation:
*
*     rData.EDGEWTDAMT === null
*       ? "0.00"
*       : rData.EDGEWTDAMT
*
*   XSA NULL and SEGW "0.00" therefore produce the same calculation
*   input and the same functional UI behavior.
*
* GRVALUE_USD
*   The UI already normalizes an XSA NULL amount to 0.00 before currency
*   formatting. SEGW "0.00" therefore produces the same visible result.
*
* INVVALUE_USD
*   This converted/display amount is initial in SEGW when XSA has no
*   source value. The exception is intentionally limited to INVVALUE_USD.
*   INVVALUE remains strict because it participates in accrual arithmetic.
*
* INVVALUE_CAL
*   No active UI consumer was identified for this calculated helper
*   property. XSA NULL and the initial SEGW decimal value "0.00" therefore

*   field-specific and must not be generalized to INVVALUE.
*
* LASTMONTHWRKCOMP
*   XSA NULL displays as an empty value. SEGW "0.00" can display as
*   0.00 in the detail view.
*

*   custom Gateway JSON serialization solely for this field would add
*   disproportionate implementation and maintenance complexity.
*
*   Backend presence information must remain separate from the numeric
*   value. A value of 0.00 must not be used as proof that the source
*   record existed.
*
* Accepted examples:
*
*   EDGEWTDAMT:
*     XSA=null, SEGW="0.00"
*
*   GRVALUE_USD:

*   LASTMONTHWRKCOMP:

*
*   XSA="0.00", SEGW=null

    WHEN 'EDGEWTDAMT'
      OR 'GRVALUE_USD'
      OR 'INVVALUE_CAL'
      OR 'INVVALUE_USD'
      OR 'LASTMONTHWRKCOMP'.

        xsdbool(
          i_actual_token = '"0.00"'
        ).

*--------------------------------------------------------------------*
* Time property serialized by SEGW as "PT00H00M00S"
*
* CHANGEDTIME
*   Initial ABAP TIMS value 000000 is also a valid midnight value.
*   Standard SEGW therefore serializes it as:
*
*     "PT00H00M00S"
*
*   XSA preserves an absent database value as:
*
*     null
*

*

*     XSA=null, SEGW="PT00H00M00S"
*
* Rejected examples:
*
*   CHANGEDTIME:
*     XSA=null, SEGW="PT10H15M00S"
*
*   CHANGEDTIME:
*     XSA="PT00H00M00S", SEGW=null
*--------------------------------------------------------------------*
    WHEN 'CHANGEDTIME'.

      rv_result =
        xsdbool(
          i_actual_token = '"PT00H00M00S"'
        ).

  ENDCASE.

ENDMETHOD.


*----------------------------------------------------------------------*
* Normalize an OData V2 Edm.Time duration token
*
* XSA and SEGW can serialize the same time with different zero padding:
*
*   XSA:  "PT9H40M11S"
*   SEGW: "PT09H40M11S"
*
*   XSA:  "PT8H24M4S"
*   SEGW: "PT08H24M04S"
*
* Valid PT...H...M...S values are returned in canonical two-digit form.
* Invalid, unsupported, NULL, or unrelated tokens are returned unchanged,
* allowing the normal strict comparison to report a real difference.
*----------------------------------------------------------------------*
  METHOD normalize_odata_time.

    DATA:
      lv_value       TYPE string,
      lv_remainder   TYPE string,
      lv_hour_text   TYPE string,
      lv_minute_text TYPE string,
      lv_second_text TYPE string,
      lv_hour        TYPE i,
      lv_minute      TYPE i,
      lv_second      TYPE i,
      lv_offset      TYPE i.

    rv_token =
      trim_text(
        i_text = i_token ).

    lv_value =
      unquote_json(
        i_token = rv_token ).

*--------------------------------------------------------------------*
* A supported value must start with PT and contain H, M, and S.
*--------------------------------------------------------------------*
    IF strlen( lv_value ) < 7
       OR lv_value(2) <> 'PT'
       OR lv_value NP 'PT*H*M*S'.
      RETURN.
    ENDIF.

    lv_remainder =
      substring(
        val = lv_value
        off = 2 ).

*--------------------------------------------------------------------*
* Extract hours.
*--------------------------------------------------------------------*
    FIND FIRST OCCURRENCE OF 'H'
      IN lv_remainder
      MATCH OFFSET lv_offset.

    IF sy-subrc <> 0
       OR lv_offset = 0.
      RETURN.
    ENDIF.

    lv_hour_text =
      substring(
        val = lv_remainder
        off = 0
        len = lv_offset ).

    lv_remainder =
      substring(
        val = lv_remainder
        off = lv_offset + 1 ).

*--------------------------------------------------------------------*
* Extract minutes.
*--------------------------------------------------------------------*
    FIND FIRST OCCURRENCE OF 'M'
      IN lv_remainder
      MATCH OFFSET lv_offset.

    IF sy-subrc <> 0
       OR lv_offset = 0.
      RETURN.
    ENDIF.

    lv_minute_text =
      substring(
        val = lv_remainder
        off = 0
        len = lv_offset ).

    lv_remainder =
      substring(
        val = lv_remainder
        off = lv_offset + 1 ).

*--------------------------------------------------------------------*
* Extract seconds and reject any trailing content.
*--------------------------------------------------------------------*
    FIND FIRST OCCURRENCE OF 'S'
      IN lv_remainder
      MATCH OFFSET lv_offset.

    IF sy-subrc <> 0
       OR lv_offset = 0
       OR lv_offset + 1 <> strlen( lv_remainder ).
      RETURN.
    ENDIF.

    lv_second_text =
      substring(
        val = lv_remainder
        off = 0
        len = lv_offset ).

*--------------------------------------------------------------------*
* Reject non-numeric components before conversion.
*--------------------------------------------------------------------*
    IF lv_hour_text CN '0123456789'
       OR lv_minute_text CN '0123456789'
       OR lv_second_text CN '0123456789'.
      RETURN.
    ENDIF.

    lv_hour =
      CONV i( lv_hour_text ).

    lv_minute =
      CONV i( lv_minute_text ).

    lv_second =
      CONV i( lv_second_text ).

*--------------------------------------------------------------------*
* Validate ordinary time-of-day ranges.
*--------------------------------------------------------------------*
    IF lv_hour > 23
       OR lv_minute > 59
       OR lv_second > 59.
      RETURN.
    ENDIF.

*--------------------------------------------------------------------*
* Canonical JSON token with two-digit components.
*--------------------------------------------------------------------*
    rv_token =
        '"PT'
      && |{ lv_hour WIDTH = 2 PAD = '0' ALIGN = RIGHT }|
      && 'H'
      && |{ lv_minute WIDTH = 2 PAD = '0' ALIGN = RIGHT }|
      && 'M'
      && |{ lv_second WIDTH = 2 PAD = '0' ALIGN = RIGHT }|
      && 'S"'.

  ENDMETHOD.


*----------------------------------------------------------------------*
* TODO_PARITY_CONTRACT_001 - SYNTHETIC XSA PROPERTY
*
* XSA exposes ID as a generated technical property. The migrated SEGW
* result contract identifies rows by PONUMBER and ITEMNO and does not
* expose ID. No active UI consumer uses the XSA ID value.
*
* Only a missing ID is accepted. Any other missing SEGW property remains
* a parity failure. Remove this exception if ID is later added to SEGW or
* becomes part of the functional UI contract.
*----------------------------------------------------------------------*
  METHOD ignore_missing_property.

    rv_result =
      xsdbool( i_name = 'ID' ).

  ENDMETHOD.


*----------------------------------------------------------------------*
* TODO_PARITY_STRICT_001 - TEMPORARY EXCEPTION
* Aggregate values are compared without order. Content and multiplicity
* must still match. Remove this exception after aggregate ordering is
* aligned between XSA and SEGW.
*----------------------------------------------------------------------*
  METHOD is_unordered_aggregate_field.

    CASE i_name.

      WHEN 'GLACCOUNT'
        OR 'COSTCENTER'
        OR 'COSTCENTERDESC'
        OR 'MANAGEUNIT'
        OR 'INTERNALORDER'
        OR 'ASSETID'
        OR 'PROFITCENTER'
        OR 'BBP_ACC_NO'
        OR 'DISTPERCENTAGE'
        OR 'RULETYPE'
        OR 'RULETYPETEXT'.

        rv_result =
          abap_true.

    ENDCASE.

  ENDMETHOD.


  METHOD equal_aggregate_content.

    DATA:
      lt_expected TYPE tt_string,
      lt_actual   TYPE tt_string.

*--------------------------------------------------------------------*
* Semantic unordered parity:
* - order of all segments is ignored;
* - leading, internal and trailing empty segments are preserved;
* - number of empty segments must still match.
*
* Therefore:
*   ;A;B     = A;B;
*   ;;A;B    = A;B;;
* but:
*   A;;B     <> A;B
*--------------------------------------------------------------------*
    lt_expected =
      split_aggregate_empty(
        i_value = i_expected ).

    lt_actual =
      split_aggregate_empty(
        i_value = i_actual ).

    IF lines( lt_expected ) <>
       lines( lt_actual ).

      FREE:
        lt_expected,
        lt_actual.

      RETURN.

    ENDIF.

    SORT lt_expected
      BY table_line.

    SORT lt_actual
      BY table_line.

    rv_equal =
      xsdbool(
        lt_expected = lt_actual ).

    FREE:
      lt_expected,
      lt_actual.

  ENDMETHOD.

  METHOD split_aggregate_empty.

    DATA:
      lv_length TYPE i,
      lv_start  TYPE i,
      lv_pos    TYPE i,
      lv_token  TYPE string.

*--------------------------------------------------------------------*
* Empty aggregate represents zero aggregate values, not one empty
* account-assignment slot. This follows the existing XSA convention.
*--------------------------------------------------------------------*
    IF i_value IS INITIAL.
      RETURN.
    ENDIF.

    lv_length =
      strlen(
        i_value ).

    WHILE lv_pos < lv_length.

      IF i_value+lv_pos(1) = ';'.

        lv_token =
          substring(
            val = i_value
            off = lv_start
            len = lv_pos - lv_start ).

        APPEND lv_token
          TO rt_tokens.

        CLEAR lv_token.

        lv_start =
          lv_pos + 1.

      ENDIF.

      lv_pos =
        lv_pos + 1.

    ENDWHILE.

*--------------------------------------------------------------------*
* Always append the final segment. If the source ends in ';', this is
* deliberately an empty token.
*--------------------------------------------------------------------*
    lv_token =
      substring(
        val = i_value
        off = lv_start
        len = lv_length - lv_start ).

    APPEND lv_token
      TO rt_tokens.

  ENDMETHOD.


  METHOD read_json_string_end.

    DATA:
      lv_position TYPE i,
      lv_length   TYPE i,
      lv_char     TYPE char1,
      lv_escape   TYPE abap_bool.

    lv_length =
      strlen(
        i_text ).

    IF i_start >= lv_length
       OR i_text+i_start(1) <> '"'.

      RAISE EXCEPTION TYPE zcx_fi_das_error
        EXPORTING
          i_reason =
                     'JSON string does not start with quote'.

    ENDIF.

    lv_position =
      i_start + 1.

    WHILE lv_position < lv_length.

      lv_char =
        i_text+lv_position(1).

      IF lv_escape = abap_true.

        CLEAR lv_escape.

      ELSEIF lv_char = '\'.

        lv_escape =
          abap_true.

      ELSEIF lv_char = '"'.

        rv_end =
          lv_position.

        RETURN.

      ENDIF.

      lv_position =
        lv_position + 1.

    ENDWHILE.

    RAISE EXCEPTION TYPE zcx_fi_das_error
      EXPORTING
        i_reason =
                   'JSON string is not closed'.

  ENDMETHOD.


  METHOD read_json_value_end.

    DATA:
      lv_position TYPE i,
      lv_length   TYPE i,
      lv_char     TYPE char1,
      lv_end      TYPE i.

    lv_length =
      strlen(
        i_text ).

    IF i_start >= lv_length.

      RAISE EXCEPTION TYPE zcx_fi_das_error
        EXPORTING
          i_reason =
                     'JSON value is missing'.

    ENDIF.

    lv_char =
      i_text+i_start(1).

    IF lv_char = '"'.

      lv_end =
        read_json_string_end(
          i_text  = i_text
          i_start = i_start ).

      rv_end =
        lv_end + 1.

      RETURN.

    ENDIF.

    IF lv_char = '{'.

      lv_end =
        find_matching_end(
          i_text  = i_text
          i_start = i_start
          i_open  = '{'
          i_close = '}' ).

      rv_end =
        lv_end + 1.

      RETURN.

    ENDIF.

    IF lv_char = '['.

      lv_end =
        find_matching_end(
          i_text  = i_text
          i_start = i_start
          i_open  = '['
          i_close = ']' ).

      rv_end =
        lv_end + 1.

      RETURN.

    ENDIF.

    lv_position =
      i_start.

    WHILE lv_position < lv_length.

      lv_char =
        i_text+lv_position(1).

      IF lv_char = ','
         OR lv_char = '}'.

        rv_end =
          lv_position.

        RETURN.

      ENDIF.

      lv_position =
        lv_position + 1.

    ENDWHILE.

    rv_end =
      lv_length.

  ENDMETHOD.


  METHOD find_matching_end.

    DATA:
      lv_position TYPE i,
      lv_length   TYPE i,
      lv_depth    TYPE i,
      lv_char     TYPE char1,
      lv_in_text  TYPE abap_bool,
      lv_escape   TYPE abap_bool.

    lv_length =
      strlen(
        i_text ).

    lv_position =
      i_start.

    WHILE lv_position < lv_length.

      lv_char =
        i_text+lv_position(1).

      IF lv_in_text = abap_true.

        IF lv_escape = abap_true.

          CLEAR lv_escape.

        ELSEIF lv_char = '\'.

          lv_escape =
            abap_true.

        ELSEIF lv_char = '"'.

          CLEAR lv_in_text.

        ENDIF.

      ELSE.

        IF lv_char = '"'.

          lv_in_text =
            abap_true.

        ELSEIF lv_char = i_open.

          lv_depth =
            lv_depth + 1.

        ELSEIF lv_char = i_close.

          lv_depth =
            lv_depth - 1.

          IF lv_depth = 0.

            rv_end =
              lv_position.

            RETURN.

          ENDIF.

        ENDIF.

      ENDIF.

      lv_position =
        lv_position + 1.

    ENDWHILE.

    RAISE EXCEPTION TYPE zcx_fi_das_error
      EXPORTING
        i_reason =
                   'JSON object or array is not closed'.

  ENDMETHOD.

  METHOD skip_whitespace.

    DATA:
      lv_length TYPE i,
      lv_char   TYPE char1.

    rv_position =
      i_position.

    lv_length =
      strlen(
        i_text ).

    WHILE rv_position < lv_length.

      lv_char =
        i_text+rv_position(1).

      IF is_whitespace(
           i_char = lv_char ) = abap_false.

        RETURN.

      ENDIF.

      rv_position =
        rv_position + 1.

    ENDWHILE.

  ENDMETHOD.


  METHOD is_whitespace.

    IF i_char = space
       OR i_char = cl_abap_char_utilities=>horizontal_tab
       OR i_char = cl_abap_char_utilities=>newline
       OR i_char = cl_abap_char_utilities=>cr_lf+0(1)
       OR i_char = cl_abap_char_utilities=>cr_lf+1(1).

      rv_result =
        abap_true.

    ENDIF.

  ENDMETHOD.


  METHOD trim_text.

    DATA:
      lv_start         TYPE i,
      lv_end           TYPE i,
      lv_length        TYPE i,
      lv_last_position TYPE i,
      lv_char          TYPE char1.

    lv_length =
      strlen(
        i_text ).

    lv_end =
      lv_length.

    WHILE lv_start < lv_length.

      lv_char =
        i_text+lv_start(1).

      IF is_whitespace(
           i_char = lv_char ) = abap_false.

        EXIT.

      ENDIF.

      lv_start =
        lv_start + 1.

    ENDWHILE.

    WHILE lv_end > lv_start.

      lv_last_position =
        lv_end - 1.

      lv_char =
        i_text+lv_last_position(1).

      IF is_whitespace(
           i_char = lv_char ) = abap_false.

        EXIT.

      ENDIF.

      lv_end =
        lv_end - 1.

    ENDWHILE.

    IF lv_end > lv_start.

      rv_text =
        substring(
          val = i_text
          off = lv_start
          len = lv_end - lv_start ).

    ENDIF.

  ENDMETHOD.


  METHOD unquote_json.

    DATA:
      lv_token        TYPE string,
      lv_length       TYPE i,
      lv_last_offset  TYPE i,
      lv_inner_length TYPE i,
      lv_first_char   TYPE char1,
      lv_last_char    TYPE char1.

    lv_token =
      trim_text(
        i_text = i_token ).

    lv_length =
      strlen(
        lv_token ).

    IF lv_length < 2.

      rv_value =
        lv_token.

      RETURN.

    ENDIF.

    lv_last_offset =
      lv_length - 1.

    lv_first_char =
      lv_token(1).

    lv_last_char =
      lv_token+lv_last_offset(1).

    IF lv_first_char <> '"'
       OR lv_last_char <> '"'.

      rv_value =
        lv_token.

      RETURN.

    ENDIF.

    lv_inner_length =
      lv_length - 2.

    rv_value =
      substring(
        val = lv_token
        off = 1
        len = lv_inner_length ).

*--------------------------------------------------------------------*
* Avoid replacement scans in the normal case where the JSON value has
* no escaped characters.
*--------------------------------------------------------------------*
    IF rv_value NS '\'.
      RETURN.
    ENDIF.

    REPLACE ALL OCCURRENCES OF '\u0026'
      IN rv_value
      WITH '&'.

    REPLACE ALL OCCURRENCES OF '\u003C'
      IN rv_value
      WITH '<'.

    REPLACE ALL OCCURRENCES OF '\u003c'
      IN rv_value
      WITH '<'.

    REPLACE ALL OCCURRENCES OF '\u003E'
      IN rv_value
      WITH '>'.

    REPLACE ALL OCCURRENCES OF '\u003e'
      IN rv_value
      WITH '>'.

    REPLACE ALL OCCURRENCES OF '\u0022'
      IN rv_value
      WITH '"'.

    REPLACE ALL OCCURRENCES OF '\u0027'
      IN rv_value
      WITH ''''.

    REPLACE ALL OCCURRENCES OF '\/'
      IN rv_value
      WITH '/'.

    REPLACE ALL OCCURRENCES OF '\"'
      IN rv_value
      WITH '"'.

    REPLACE ALL OCCURRENCES OF '\n'
      IN rv_value
      WITH cl_abap_char_utilities=>newline.

    REPLACE ALL OCCURRENCES OF '\r'
      IN rv_value
      WITH cl_abap_char_utilities=>cr_lf+0(1).

    REPLACE ALL OCCURRENCES OF '\t'
      IN rv_value
      WITH cl_abap_char_utilities=>horizontal_tab.

    REPLACE ALL OCCURRENCES OF '\\'
      IN rv_value
      WITH '\'.

  ENDMETHOD.


  METHOD add_difference.

    mv_difference_count =
      mv_difference_count + 1.

*--------------------------------------------------------------------*
* Count every mismatch, but retain a bounded diagnostic sample. This
* prevents failures with many missing rows from allocating huge logs.
*--------------------------------------------------------------------*
    IF lines( mt_differences ) >= mc_max_log.
      RETURN.
    ENDIF.

    APPEND i_text
      TO mt_differences.

  ENDMETHOD.


  METHOD assert_no_differences.

    DATA:
      lv_text   TYPE string,
      lv_line   TYPE string,
      lv_shown  TYPE i,
      lv_hidden TYPE i.

    IF mv_difference_count = 0.

      cl_abap_unit_assert=>assert_true(
        act = abap_true ).

      RETURN.

    ENDIF.

    lv_shown =
      lines(
        mt_differences ).

    lv_hidden =
      mv_difference_count
      - lv_shown.

    lv_text =
        |DAS XSA vs SEGW OData parity failed. |
      && |Difference count: { mv_difference_count }.|
      && cl_abap_char_utilities=>newline
      && |--- First { lv_shown } difference(s) ---|.

    LOOP AT mt_differences INTO lv_line.

      lv_text =
          lv_text
        && cl_abap_char_utilities=>newline
        && lv_line.

    ENDLOOP.

    IF lv_hidden > 0.

      lv_text =
          lv_text
        && cl_abap_char_utilities=>newline
        && |--- { lv_hidden } further difference(s) omitted ---|.

    ENDIF.

    cl_abap_unit_assert=>fail(
      msg = lv_text ).

  ENDMETHOD.
