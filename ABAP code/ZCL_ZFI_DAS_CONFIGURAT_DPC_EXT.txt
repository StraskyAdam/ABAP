CLASS ZCL_ZFI_DAS_CONFIGURAT_DPC_EXT IMPLEMENTATION.


  METHOD configlang_create_entity.

    DATA ls_entity TYPE zfi_das_lang.

    io_data_provider->read_entry_data( IMPORTING es_data = ls_entity ).

    zcl_fi_das_config_util=>create_language( ls_entity ).

    er_entity = CORRESPONDING #( ls_entity ).

  ENDMETHOD.


  METHOD configlang_delete_entity.

    DATA ls_keys TYPE zcl_zfi_das_configurat_mpc=>ts_configlangtype.

    io_tech_request_context->get_converted_keys( IMPORTING es_key_values = ls_keys ).

    zcl_fi_das_config_util=>delete_language( CORRESPONDING zfi_das_lang( ls_keys ) ).

  ENDMETHOD.


  METHOD configlang_get_entityset.

    et_entityset = CORRESPONDING #( zcl_fi_das_config_util=>get_languages( ) ).

  ENDMETHOD.

          ls_keys   TYPE zcl_zfi_das_configurat_mpc=>ts_thresholdtype.

    io_data_provider->read_entry_data( IMPORTING es_data = ls_entity ).

    ls_entity-currency = ls_keys-currency.

    zcl_fi_das_config_util=>update_threshold( ls_entity ).
