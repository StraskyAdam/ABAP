  METHOD build_request.

    IF io_tech_request_context IS NOT BOUND.
      RAISE EXCEPTION TYPE /iwbep/cx_mgw_tech_exception.
    ENDIF.

    CLEAR rs_request.

    get_key_parameters( EXPORTING it_key_tab = it_key_tab
                        CHANGING  cs_request = rs_request ).

    get_paging( EXPORTING io_tech_request_context = io_tech_request_context
                CHANGING  cs_request = rs_request ).

    get_select( EXPORTING io_tech_request_context = io_tech_request_context
                CHANGING  cs_request = rs_request ).

    get_filter( EXPORTING io_tech_request_context = io_tech_request_context
                CHANGING  cs_request = rs_request ).

    get_order( EXPORTING io_tech_request_context = io_tech_request_context
               CHANGING  cs_request = rs_request ).

  ENDMETHOD.


  METHOD exclusionposet_get_entityset.

    TRY.
        DATA(lo_dashboard) = CAST zif_fi_das_dashboard( NEW zcl_fi_das_dashboard( ) ).

        lo_dashboard->exclude(
          EXPORTING
            is_request    = build_request( it_key_tab              = it_key_tab
                                           io_tech_request_context = io_tech_request_context )
          IMPORTING
            et_result     = DATA(lt_result)
            e_total_count = es_response_context-inlinecount ).

        et_entityset = CORRESPONDING #( lt_result ).

      CATCH cx_sy_conversion_error
            cx_sy_open_sql_db
            zcx_fi_das_error INTO DATA(lx_error).

        RAISE EXCEPTION TYPE /iwbep/cx_mgw_tech_exception
          EXPORTING
            previous = lx_error.

    ENDTRY.

  ENDMETHOD.

  METHOD get_key_parameters.

    cs_request-email = VALUE #( it_key_tab[ name = zif_fi_das_dashboard=>mc_key_email ]-value OPTIONAL ).

    cs_request-pstype = VALUE #( it_key_tab[ name = zif_fi_das_dashboard=>mc_key_pstype ]-value OPTIONAL ).

    cs_request-poowner = VALUE #( it_key_tab[ name = zif_fi_das_dashboard=>mc_key_poowner ]-value OPTIONAL ).

  ENDMETHOD.


  METHOD get_order.

    LOOP AT io_tech_request_context->get_orderby( ) ASSIGNING FIELD-SYMBOL(<fs_orderby>) WHERE property IS NOT INITIAL.

      APPEND VALUE #( property   = to_upper( CONV string( <fs_orderby>-property ) )
                      descending = xsdbool( to_upper( <fs_orderby>-order ) = mc_order_descending ) ) TO cs_request-orderby.

    ENDLOOP.

  ENDMETHOD.


  METHOD get_paging.

    DATA(l_top) = io_tech_request_context->get_top( ).

    cs_request-paging-top_requested = xsdbool( l_top IS NOT INITIAL ).

    cs_request-paging-top = l_top.

    cs_request-paging-skip = io_tech_request_context->get_skip( ).

    cs_request-paging-count_requested = io_tech_request_context->has_inlinecount( ).

  ENDMETHOD.


  METHOD get_select.

    cs_request-select = to_upper( concat_lines_of( table = io_tech_request_context->get_select( )
                                                   sep   = zif_fi_das_dashboard=>mc_comma ) ).

  ENDMETHOD.

  METHOD _parse_odata_eq_ne.
    " Parsuje: PROPERTY eq 'VALUE' a PROPERTY ne 'VALUE'
    " (pre prípady mimo substringof)

    DATA: lv_work         TYPE string,
          lt_parts        TYPE TABLE OF string,
          lv_part         TYPE string,
          lv_pos          TYPE i,
          lv_property     TYPE string,
          lv_value        TYPE string,
          lv_sign         TYPE c LENGTH 1.

    lv_work = iv_filter.

    " Rozdelenie na časti podľa AND (case-INsensitive)
    REPLACE ALL OCCURRENCES OF ' AND ' IN lv_work WITH '|' IGNORING CASE.
    REPLACE ALL OCCURRENCES OF ' OR ' IN lv_work WITH '|' IGNORING CASE.

    SPLIT lv_work AT '|' INTO TABLE lt_parts.

    LOOP AT lt_parts INTO lv_part.
      lv_part = condense( lv_part ).

      IF lv_part CS 'SUBSTRINGOF'.
        " Toto sme už spracovali vyššie
        CONTINUE.
      ENDIF.

      " Skúsiť 'eq'
      lv_pos = find( val = lv_part sub = ' eq ' case = abap_false ).

      IF lv_pos >= 0.
        lv_property = substring( val = lv_part off = 0 len = lv_pos ).
        lv_property = condense( lv_property ).
        lv_property = to_upper( lv_property ).

        lv_value = substring( val = lv_part off = lv_pos + 4 ).
        lv_value = condense( lv_value ).
        REPLACE ALL OCCURRENCES OF `'` IN lv_value WITH ''.

        IF lv_property IS NOT INITIAL AND lv_value IS NOT INITIAL.
          APPEND VALUE #(
            property = lv_property
            sign     = 'I'
            option   = 'EQ'
            low      = lv_value
            high     = ''
          ) TO cs_request-generic_filters.
        ENDIF.

        CONTINUE.
      ENDIF.

      " Skúsiť 'ne'
      lv_pos = find( val = lv_part sub = ' ne ' case = abap_false ).

      IF lv_pos >= 0.
        lv_property = substring( val = lv_part off = 0 len = lv_pos ).
        lv_property = condense( lv_property ).
        lv_property = to_upper( lv_property ).

        lv_value = substring( val = lv_part off = lv_pos + 4 ).
        lv_value = condense( lv_value ).
        REPLACE ALL OCCURRENCES OF `'` IN lv_value WITH ''.

        IF lv_property IS NOT INITIAL AND lv_value IS NOT INITIAL.
          APPEND VALUE #(
            property = lv_property
            sign     = 'E'
            option   = 'EQ'
            low      = lv_value
            high     = ''
          ) TO cs_request-generic_filters.
        ENDIF.
      ENDIF.

    ENDLOOP.

  ENDMETHOD.


   METHOD _parse_odata_filter.
    " Parsuje SQL-like filter syntax z Gateway:
    "   PROPERTY like '%VALUE%'
    "   PROPERTY = 'VALUE'
    "   not (...)
    "   and / or
    "
    " Výstup: generic_filters s SIGN/OPTION/LOW/HIGH

    DATA: lv_work         TYPE string,
          lt_tokens       TYPE TABLE OF string,
          lv_token        TYPE string,
          lv_pos          TYPE i,
          lv_pos_like     TYPE i,
          lv_pos_eq       TYPE i,
          lv_property     TYPE string,
          lv_value        TYPE string,
          lv_is_negated   TYPE abap_bool,
          lv_sign         TYPE c LENGTH 1,
          lv_option       TYPE c LENGTH 2.

    IF iv_filter IS INITIAL.
      RETURN.
    ENDIF.

    lv_work = iv_filter.

    " ========================================================================
    " KROK 1: Nahraď AND/OR za separator PRED odstrániť zátvorky
    " Poznámka: strácame informáciu o AND vs OR, ale dashboard predpokladá
    " že sa všetky filtre rovnakého poľa platia súčasne
    " ========================================================================
    REPLACE ALL OCCURRENCES OF ' AND ' IN lv_work WITH '§§' IGNORING CASE.
    REPLACE ALL OCCURRENCES OF ' OR ' IN lv_work WITH '§§' IGNORING CASE.

    " ========================================================================
    " KROK 2: Odstráň všetky zátvorky
    " ========================================================================
    REPLACE ALL OCCURRENCES OF '(' IN lv_work WITH ''.
    REPLACE ALL OCCURRENCES OF ')' IN lv_work WITH ''.
    lv_work = condense( lv_work ).

    IF lv_work IS INITIAL.
      RETURN.
    ENDIF.

    " ========================================================================
    " KROK 3: Rozdelenie podľa separatora
    " ========================================================================
    SPLIT lv_work AT '§§' INTO TABLE lt_tokens.

    LOOP AT lt_tokens INTO lv_token.
      lv_token = condense( lv_token ).

      IF lv_token IS INITIAL.
        CONTINUE.
      ENDIF.

      " ====================================================================
      " KROK 4: Detekcia NOT prefix
      " ====================================================================
      lv_is_negated = abap_false.

      IF lv_token CP 'NOT *' OR lv_token CP 'not *'.
        lv_is_negated = abap_true.
        " Preskoč 'NOT ' (4 znaky)
        lv_token = substring( val = lv_token off = 4 ).
        lv_token = condense( lv_token ).
      ENDIF.

      IF lv_token IS INITIAL.
        CONTINUE.
      ENDIF.

      " ====================================================================
      " KROK 5: Detekcia operátorov — ' LIKE ' alebo ' = '
      " ====================================================================
      lv_property = ''.
      lv_value = ''.
      lv_option = 'CP'.

      " Kontrola ' LIKE ' (case-INsensitive)
      lv_pos_like = find(
        val = lv_token
        sub = ' LIKE '
        case = abap_false ).

      " Kontrola ' = ' (case-INsensitive)
      lv_pos_eq = find(
        val = lv_token
        sub = ' = '
        case = abap_false ).

      " Priorita: LIKE pred EQ
      IF lv_pos_like >= 0.
        lv_property = substring( val = lv_token off = 0 len = lv_pos_like ).
        lv_value = substring( val = lv_token off = lv_pos_like + 6 ).
        lv_option = 'CP'.

      ELSEIF lv_pos_eq >= 0.
        lv_property = substring( val = lv_token off = 0 len = lv_pos_eq ).
        lv_value = substring( val = lv_token off = lv_pos_eq + 3 ).
        lv_option = 'EQ'.

      ELSE.
        " Žiaden operátor — preskoč
        CONTINUE.
      ENDIF.

      " ====================================================================
      " KROK 6: Očista property a value
      " ====================================================================
      lv_property = condense( lv_property ).
      lv_value = condense( lv_value ).

      IF lv_property IS INITIAL OR lv_value IS INITIAL.
        CONTINUE.
      ENDIF.

      " Property na UPPER
      lv_property = to_upper( lv_property ).

      " ====================================================================
      " KROK 7: Extrakcia hodnoty podľa operátora
      " ====================================================================
      CASE lv_option.

        WHEN 'CP'.
          " '%VALUE%' → '*VALUE*'
          REPLACE ALL OCCURRENCES OF `'` IN lv_value WITH ''.
          REPLACE ALL OCCURRENCES OF '%' IN lv_value WITH '*'.
          lv_value = condense( lv_value ).

        WHEN 'EQ'.
          " Odstráň úvodzovky
          REPLACE ALL OCCURRENCES OF `'` IN lv_value WITH ''.
          lv_value = condense( lv_value ).

      ENDCASE.

      IF lv_value IS INITIAL.
        CONTINUE.
      ENDIF.

      " ====================================================================
      " KROK 8: Nastav SIGN podľa negácie
      " ====================================================================
      lv_sign = COND #( WHEN lv_is_negated = abap_true THEN 'E' ELSE 'I' ).

      " ====================================================================
      " KROK 9: Pridaj filter do generic_filters
      " ====================================================================
      APPEND VALUE #(
        property = lv_property
        sign     = lv_sign
        option   = lv_option
        low      = lv_value
        high     = ''
      ) TO cs_request-generic_filters.

    ENDLOOP.

  ENDMETHOD.
