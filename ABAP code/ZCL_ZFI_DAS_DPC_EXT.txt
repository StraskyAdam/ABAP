CLASS ZCL_ZFI_DAS_DPC_EXT IMPLEMENTATION.


METHOD build_request.

  CLEAR rs_request.

  " Parameters z openPOParameters(...)

  READ TABLE it_key_tab ASSIGNING FIELD-SYMBOL(<ls_key>)
    WITH KEY name = 'IP_EMAIL'.

  IF sy-subrc = 0.
    rs_request-email = <ls_key>-value.
  ENDIF.

  READ TABLE it_key_tab ASSIGNING <ls_key>
    WITH KEY name = 'IP_PSTYPE'.

  IF sy-subrc = 0.
    rs_request-pstype = <ls_key>-value.
  ENDIF.

  READ TABLE it_key_tab ASSIGNING <ls_key>
    WITH KEY name = 'IP_POOWNER'.

  IF sy-subrc = 0.
    rs_request-poowner = <ls_key>-value.
  ENDIF.

  " Paging

  rs_request-paging-top =
    io_tech_request_context->get_top( ).

  rs_request-paging-skip =
    io_tech_request_context->get_skip( ).

  rs_request-paging-count_requested =
    io_tech_request_context->has_inlinecount( ).

  " OData filtre

  DATA(lt_filters) =
    io_tech_request_context->get_filter( )->get_filter_select_options( ).

  LOOP AT lt_filters ASSIGNING FIELD-SYMBOL(<ls_filter>).

    CASE to_upper( <ls_filter>-property ).

      WHEN 'PONUMBER'.

        LOOP AT <ls_filter>-select_options
          ASSIGNING FIELD-SYMBOL(<ls_ponumber>).

          IF <ls_ponumber>-sign = 'I'
             AND <ls_ponumber>-option = 'CP'.

            DATA(l_ponumber_contains) =
              CONV string( <ls_ponumber>-low ).

            SHIFT l_ponumber_contains LEFT DELETING LEADING '*'.
            SHIFT l_ponumber_contains RIGHT DELETING TRAILING '*'.

            IF <ls_ponumber>-low CS '*'
               AND <ls_ponumber>-low NA '+'
               AND l_ponumber_contains IS NOT INITIAL
               AND l_ponumber_contains NA '*'.

              APPEND VALUE #(
                value = CONV zif_fi_das_dashboard=>t_text_500(
                  l_ponumber_contains ) ) TO rs_request-po_contains.
              CONTINUE.

            ENDIF.
          ENDIF.

          APPEND VALUE #(
            sign   = <ls_ponumber>-sign
            option = <ls_ponumber>-option
            low    = CONV ebeln( <ls_ponumber>-low )
            high   = CONV ebeln( <ls_ponumber>-high )
          ) TO rs_request-ponumber.

        ENDLOOP.

      WHEN 'ITEMNO'.

        LOOP AT <ls_filter>-select_options
          ASSIGNING FIELD-SYMBOL(<ls_itemno>).

          IF <ls_itemno>-sign = 'I'
             AND <ls_itemno>-option = 'CP'.

            DATA(l_itemno_contains) =
              CONV string( <ls_itemno>-low ).

            SHIFT l_itemno_contains LEFT DELETING LEADING '*'.
            SHIFT l_itemno_contains RIGHT DELETING TRAILING '*'.

            IF <ls_itemno>-low CS '*'
               AND <ls_itemno>-low NA '+'
               AND l_itemno_contains IS NOT INITIAL
               AND l_itemno_contains NA '*'.

              APPEND VALUE #(
                value = CONV zif_fi_das_dashboard=>t_text_500(
                  l_itemno_contains ) ) TO rs_request-item_contains.
              CONTINUE.

            ENDIF.
          ENDIF.

          APPEND VALUE #(
            sign   = <ls_itemno>-sign
            option = <ls_itemno>-option
            low    = CONV ebelp( <ls_itemno>-low )
            high   = CONV ebelp( <ls_itemno>-high )
          ) TO rs_request-itemno.

        ENDLOOP.

      WHEN 'SUPPLIERNAME'.

        LOOP AT <ls_filter>-select_options
          ASSIGNING FIELD-SYMBOL(<ls_supplier>).

          APPEND VALUE #(
            value = CONV zif_fi_das_dashboard=>t_supplier(
                      <ls_supplier>-low )
          ) TO rs_request-supplier.

        ENDLOOP.

    ENDCASE.

  ENDLOOP.

  DATA(lt_orderby) =
    io_tech_request_context->get_orderby( ).

  READ TABLE lt_orderby
    ASSIGNING FIELD-SYMBOL(<ls_orderby>)
    INDEX 1.

  IF sy-subrc = 0.
    rs_request-orderby-property =
      to_upper( CONV string( <ls_orderby>-property ) ).
    rs_request-orderby-descending =
      xsdbool(
        to_upper( CONV string( <ls_orderby>-order ) ) CS 'DESC' ).
  ENDIF.

  LOOP AT io_tech_request_context->get_select( )
    ASSIGNING FIELD-SYMBOL(<lv_selected_property>).
    APPEND VALUE #(
      name = to_upper( CONV string( <lv_selected_property> ) ) )
      TO rs_request-select_fields.
  ENDLOOP.

  SORT rs_request-select_fields BY name.
  DELETE ADJACENT DUPLICATES FROM rs_request-select_fields COMPARING name.

ENDMETHOD.


METHOD exclusionposet_get_entityset.

  DATA: ls_request   TYPE zif_fi_das_dashboard=>ts_request,
        lo_dashboard TYPE REF TO zif_fi_das_dashboard.

  ls_request = build_request(
    io_tech_request_context = io_tech_request_context
    it_key_tab              = it_key_tab ).

  CREATE OBJECT lo_dashboard
    TYPE zcl_fi_das_dashboard.

  lo_dashboard->exclude(
    EXPORTING
      is_request    = ls_request
    IMPORTING
      et_result     = DATA(lt_result)
      e_total_count = DATA(lv_count) ).

  et_entityset = CORRESPONDING #( lt_result ).

  es_response_context-inlinecount = lv_count.

ENDMETHOD.
