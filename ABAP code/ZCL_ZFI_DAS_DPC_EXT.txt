CLASS ZCL_ZFI_DAS_DPC_EXT IMPLEMENTATION.


METHOD build_request.

  CLEAR rs_request.

  " Parameters z openPOParameters(...)

  READ TABLE it_key_tab ASSIGNING FIELD-SYMBOL(<ls_key>)
    WITH KEY name = mc_key_email.

  IF sy-subrc = 0.
    rs_request-email = <ls_key>-value.
  ENDIF.

  READ TABLE it_key_tab ASSIGNING <ls_key>
    WITH KEY name = mc_key_pstype.

  IF sy-subrc = 0.
    rs_request-pstype = <ls_key>-value.
  ENDIF.

  READ TABLE it_key_tab ASSIGNING <ls_key>
    WITH KEY name = mc_key_poowner.

  IF sy-subrc = 0.
    rs_request-poowner = <ls_key>-value.
  ENDIF.

  " Paging

  rs_request-paging-top =
    io_tech_request_context->get_top( ).

  rs_request-paging-skip =
    io_tech_request_context->get_skip( ).

  rs_request-paging-count_requested =
    io_tech_request_context->has_inlinecount( ).

  " OData filtre

  DATA(lt_filters) =
    io_tech_request_context->get_filter( )->get_filter_select_options( ).

  LOOP AT lt_filters ASSIGNING FIELD-SYMBOL(<ls_filter>).

    CASE to_upper( <ls_filter>-property ).

      WHEN mc_property_ponumber.

        LOOP AT <ls_filter>-select_options
          ASSIGNING FIELD-SYMBOL(<ls_ponumber>).

          IF <ls_ponumber>-option = mc_option_contains
             AND <ls_ponumber>-sign = mc_sign_include.

            DATA(l_po_contains) =
              get_contains_value( <ls_ponumber>-low ).

            IF l_po_contains IS NOT INITIAL.
              APPEND VALUE #( value = l_po_contains )
                TO rs_request-po_contains.
            ENDIF.

          ELSE.

            APPEND VALUE #(
              sign   = <ls_ponumber>-sign
              option = <ls_ponumber>-option
              low    = CONV ebeln( <ls_ponumber>-low )
              high   = CONV ebeln( <ls_ponumber>-high )
            ) TO rs_request-ponumber.

          ENDIF.

        ENDLOOP.

      WHEN mc_property_itemno.

        LOOP AT <ls_filter>-select_options
          ASSIGNING FIELD-SYMBOL(<ls_itemno>).

          IF <ls_itemno>-option = mc_option_contains
             AND <ls_itemno>-sign = mc_sign_include.

            DATA(l_item_contains) =
              get_contains_value( <ls_itemno>-low ).

            IF l_item_contains IS NOT INITIAL.
              APPEND VALUE #( value = l_item_contains )
                TO rs_request-item_contains.
            ENDIF.

          ELSE.

            APPEND VALUE #(
              sign   = <ls_itemno>-sign
              option = <ls_itemno>-option
              low    = CONV ebelp( <ls_itemno>-low )
              high   = CONV ebelp( <ls_itemno>-high )
            ) TO rs_request-itemno.

          ENDIF.

        ENDLOOP.

      WHEN mc_property_suppliername.

        LOOP AT <ls_filter>-select_options
          ASSIGNING FIELD-SYMBOL(<ls_supplier>).

          DATA(l_supplier_contains) =
            get_contains_value( <ls_supplier>-low ).

          IF l_supplier_contains IS NOT INITIAL.
            APPEND VALUE #(
              value = CONV zif_fi_das_dashboard=>t_supplier(
                        l_supplier_contains )
            ) TO rs_request-supplier_contains.
          ENDIF.

        ENDLOOP.

      WHEN OTHERS.

        " Column Settings "Filter..." on any other column: pass it
        " through generically, evaluation happens in the dashboard.
        LOOP AT <ls_filter>-select_options
          ASSIGNING FIELD-SYMBOL(<ls_generic>).

          IF <ls_generic>-sign <> mc_sign_include.
            CONTINUE.
          ENDIF.

          IF <ls_generic>-option = mc_option_contains.

            DATA(l_generic_contains) =
              get_contains_value( <ls_generic>-low ).

            IF l_generic_contains IS NOT INITIAL.
              APPEND VALUE #(
                property = to_upper( <ls_filter>-property )
                option   = mc_option_contains
                low      = CONV zif_fi_das_dashboard=>t_text_500(
                             l_generic_contains )
              ) TO rs_request-generic_filters.
            ENDIF.

          ELSE.

            APPEND VALUE #(
              property = to_upper( <ls_filter>-property )
              option   = <ls_generic>-option
              low      = CONV zif_fi_das_dashboard=>t_text_500(
                           <ls_generic>-low )
              high     = CONV zif_fi_das_dashboard=>t_text_500(
                           <ls_generic>-high )
            ) TO rs_request-generic_filters.

          ENDIF.

        ENDLOOP.


    ENDCASE.

  ENDLOOP.

  " $orderby - first entry only, evaluation happens in the dashboard

  DATA(lt_orderby) = io_tech_request_context->get_orderby( ).

  READ TABLE lt_orderby INTO DATA(ls_orderby) INDEX 1.

  IF sy-subrc = 0.
    rs_request-orderby-property =
      to_upper( ls_orderby-property ).
    rs_request-orderby-descending =
      xsdbool( to_upper( ls_orderby-order ) CS mc_order_descending ).
  ENDIF.

  " $select - raw comma-separated list, evaluation happens
  " in the dashboard

  rs_request-select = to_upper(
    concat_lines_of(
      table = io_tech_request_context->get_select( )
      sep   = zif_fi_das_dashboard=>mc_comma ) ).

ENDMETHOD.


METHOD exclusionposet_get_entityset.

  DATA lo_dashboard TYPE REF TO zif_fi_das_dashboard.

  DATA(ls_request) = build_request(
    io_tech_request_context = io_tech_request_context
    it_key_tab              = it_key_tab ).

  lo_dashboard = NEW zcl_fi_das_dashboard( ).

  lo_dashboard->exclude(
    EXPORTING
      is_request    = ls_request
    IMPORTING
      et_result     = DATA(lt_result)
      e_total_count = DATA(lv_count) ).

  et_entityset = CORRESPONDING #( lt_result ).

  es_response_context-inlinecount = lv_count.

ENDMETHOD.


METHOD get_contains_value.

* Reduce a CP pattern to its literal characters. The UI sends only
* *value* patterns; stripping wildcards makes any pattern safe
* without ever rejecting the request. Inner spaces are preserved
* (supplier names), only leading/trailing whitespace is removed.

  r_value = i_pattern.

  REPLACE ALL OCCURRENCES OF
    zif_fi_das_dashboard=>mc_asterisk_sign IN r_value WITH ``.
  REPLACE ALL OCCURRENCES OF '+' IN r_value WITH ``.

  CONDENSE r_value.

ENDMETHOD.
