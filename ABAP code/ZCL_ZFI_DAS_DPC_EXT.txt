CLASS zcl_zfi_das_dpc_ext DEFINITION
  PUBLIC
  INHERITING FROM zcl_zfi_das_dpc
  CREATE PUBLIC .

  PUBLIC SECTION.
  PROTECTED SECTION.

    METHODS openpotypeset_get_entityset
        REDEFINITION.
    METHODS thresholdset_get_entityset
        REDEFINITION.
    METHODS exclusionposet_get_entityset
        REDEFINITION.

  PRIVATE SECTION.

    CONSTANTS mc_order_descending TYPE string VALUE 'DESC'.

    METHODS get_dashboard
      RETURNING
        VALUE(ro_dashboard) TYPE REF TO zif_fi_das_dashboard .
    METHODS build_request
      IMPORTING
        !io_tech_request_context TYPE REF TO /iwbep/if_mgw_req_entityset
        !it_key_tab              TYPE /iwbep/t_mgw_name_value_pair OPTIONAL
      RETURNING
        VALUE(rs_request)        TYPE zif_fi_das_dashboard=>ts_request
      RAISING
        /iwbep/cx_mgw_tech_exception .
    METHODS get_parameter
      IMPORTING
        !it_key_tab         TYPE /iwbep/t_mgw_name_value_pair
      RETURNING
        VALUE(rs_parameter) TYPE zif_fi_das_dashboard=>ts_request-parameter .
    METHODS get_paging
      IMPORTING
        !io_tech_request_context TYPE REF TO /iwbep/if_mgw_req_entityset
      RETURNING
        VALUE(rs_paging)         TYPE zif_fi_das_dashboard=>ts_request-paging .
    METHODS get_filter
      IMPORTING
        !io_tech_request_context TYPE REF TO /iwbep/if_mgw_req_entityset
      RETURNING
        VALUE(rs_filter)         TYPE zif_fi_das_dashboard=>ts_request-filter
      RAISING
        /iwbep/cx_mgw_tech_exception .
    METHODS get_orderby
      IMPORTING
        !io_tech_request_context TYPE REF TO /iwbep/if_mgw_req_entityset
      RETURNING
        VALUE(rs_orderby)        TYPE zif_fi_das_dashboard=>ts_request-orderby .
ENDCLASS.

CLASS zcl_zfi_das_dpc_ext IMPLEMENTATION.

  METHOD build_request.

    IF io_tech_request_context IS NOT BOUND.
      RAISE EXCEPTION TYPE /iwbep/cx_mgw_tech_exception.
    ENDIF.

    rs_request = VALUE #( parameter = get_parameter( it_key_tab )
                          paging = get_paging( io_tech_request_context )
                          filter = get_filter( io_tech_request_context )
                          orderby = get_orderby( io_tech_request_context ) ).

  ENDMETHOD.

  METHOD exclusionposet_get_entityset.

    TRY.
        get_dashboard( )->exclude(
          EXPORTING
            is_request    = build_request( it_key_tab              = it_key_tab
                                           io_tech_request_context = io_tech_request_context )
          IMPORTING
            et_result     = et_entityset
            e_total_count = es_response_context-inlinecount ).

      CATCH cx_sy_conversion_error
            cx_sy_open_sql_db
            zcx_fi_das_error INTO DATA(lx_error).

        RAISE EXCEPTION TYPE /iwbep/cx_mgw_tech_exception
          EXPORTING
            previous = lx_error.

    ENDTRY.

  ENDMETHOD.

  METHOD get_dashboard.

    ro_dashboard = NEW zcl_fi_das_dashboard( ).

  ENDMETHOD.

  METHOD get_filter.

    TRY.
        DATA(lo_filter) = io_tech_request_context->get_filter( ).

        IF lo_filter IS NOT BOUND.
          RETURN.
        ENDIF.

        rs_filter-filter_select_options = VALUE #( FOR ls_filter IN lo_filter->get_filter_select_options( )
                                                     FOR ls_option IN ls_filter-select_options (
                                                       property = to_upper( ls_filter-property )
                                                       sign = ls_option-sign
                                                       option = ls_option-option
                                                       low = ls_option-low
                                                       high = ls_option-high ) ).

        rs_filter-filter_string = lo_filter->get_filter_string( ).

*       Raise an exception to avoid processing an unsupported filter
        IF rs_filter-filter_string IS INITIAL
           AND rs_filter-filter_select_options IS INITIAL
           AND io_tech_request_context->get_filter_expression_tree( ) IS BOUND.
          RAISE EXCEPTION TYPE /iwbep/cx_mgw_tech_exception.
        ENDIF.

      CATCH cx_root INTO DATA(lx_error).

        CLEAR rs_filter.

        RAISE EXCEPTION TYPE /iwbep/cx_mgw_tech_exception
          EXPORTING
            previous = lx_error.

    ENDTRY.

  ENDMETHOD.

  METHOD get_orderby.

    LOOP AT io_tech_request_context->get_orderby( ) ASSIGNING FIELD-SYMBOL(<fs_orderby>) WHERE property IS NOT INITIAL.

      APPEND VALUE #( property   = to_upper( CONV string( <fs_orderby>-property ) )
                      descending = xsdbool( to_upper( <fs_orderby>-order ) = mc_order_descending ) ) TO rs_orderby.

    ENDLOOP.

  ENDMETHOD.

  METHOD get_paging.

    DATA(l_top) = io_tech_request_context->get_top( ).

    rs_paging-top_requested = xsdbool( l_top IS NOT INITIAL ).

    rs_paging-top = l_top.

    rs_paging-skip = io_tech_request_context->get_skip( ).

    rs_paging-count_requested = io_tech_request_context->has_inlinecount( ).

  ENDMETHOD.

  METHOD get_parameter.

    rs_parameter-email = VALUE #( it_key_tab[ name = zif_fi_das_dashboard=>mc_parameter_email ]-value OPTIONAL ).

    rs_parameter-pstype = VALUE #( it_key_tab[ name = zif_fi_das_dashboard=>mc_parameter_pstype  ]-value OPTIONAL ).

    rs_parameter-poowner = VALUE #( it_key_tab[ name = zif_fi_das_dashboard=>mc_parameter_poowner ]-value OPTIONAL ).

  ENDMETHOD.

  METHOD openpotypeset_get_entityset.

    TRY.
        get_dashboard( )->over(
          EXPORTING
            is_request    = build_request( it_key_tab              = it_key_tab
                                           io_tech_request_context = io_tech_request_context )
          IMPORTING
            et_result     = et_entityset
            e_total_count = es_response_context-inlinecount ).

      CATCH cx_sy_conversion_error
            cx_sy_open_sql_db
            zcx_fi_das_error INTO DATA(lx_error).

        RAISE EXCEPTION TYPE /iwbep/cx_mgw_tech_exception
          EXPORTING
            previous = lx_error.

    ENDTRY.

  ENDMETHOD.

  METHOD thresholdset_get_entityset.

    TRY.
        get_dashboard( )->under(
          EXPORTING
            is_request    = build_request( it_key_tab              = it_key_tab
                                           io_tech_request_context = io_tech_request_context )
          IMPORTING
            et_result     = et_entityset
            e_total_count = es_response_context-inlinecount ).

      CATCH cx_sy_conversion_error
            cx_sy_open_sql_db
            zcx_fi_das_error INTO DATA(lx_error).

        RAISE EXCEPTION TYPE /iwbep/cx_mgw_tech_exception
          EXPORTING
            previous = lx_error.

    ENDTRY.

  ENDMETHOD.

ENDCLASS.
