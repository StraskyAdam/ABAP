CLASS ZCL_ZFI_DAS_DPC_EXT IMPLEMENTATION.


METHOD build_request.

  CLEAR rs_request.

  DATA(lt_supported_orderby) = VALUE stringtab(
      ( `PSTYPE` )
      ( `PONUMBER` )
      ( `ITEMNO` )
      ( `SUPPLIERNAME` )
      ( `SUPPLIERID` )
      ( `GLACCOUNT` )
      ( `COSTCENTER` )
      ( `COSTCENTERDESC` )
      ( `MANAGEUNIT` )
      ( `INTERNALORDER` )
      ( `ASSETID` )
      ( `PROFITCENTER` )
      ( `POCURRENCY` )
      ( `POCREATEDAT` )
      ( `EPNUMBER` )
      ( `POLINEDESC` )
      ( `RULETYPE` )
      ( `RULETYPETEXT` )
      ( `EXCHANGERATE` )
      ( `STATUS` )
      ( `TOTWRKCOMPAMT` )
      ( `TOTWRKCOMPAMT_USD` )
      ( `TOTWRKCOMPAMT_COCDECURR` )
      ( `TOTWRKCOMPPER` )
      ( `AMOUNTACCRUED` )
      ( `AMOUNTACCRUED_USD` )
      ( `AMOUNTACCRUED_COCDECURR` )
      ( `INVVALUE` )
      ( `INVVALUE_USD` )
      ( `INVVALUE_COCDECURR` )
      ( `GRVALUE` )
      ( `GRVALUE_USD` )
      ( `GRVALUE_COCDECURR` )
      ( `ACCRUALMETHOD` )
      ( `MARKASREVIEWED` )
      ( `LASTMONTHWRKCOMP` )
      ( `LAST_NOTIF_ON` )
      ( `NOTIF_24_HR` )
      ( `REVIEWED_BY` )
      ( `BBP_ACC_NO` )
      ( `DISTPERCENTAGE` )
      ( `QUANTITY` ) ).

  " Parameters z openPOParameters(...)

  READ TABLE it_key_tab ASSIGNING FIELD-SYMBOL(<ls_key>)
    WITH KEY name = 'IP_EMAIL'.

  IF sy-subrc = 0.
    rs_request-email = <ls_key>-value.
  ENDIF.

  READ TABLE it_key_tab ASSIGNING <ls_key>
    WITH KEY name = 'IP_PSTYPE'.

  IF sy-subrc = 0.
    rs_request-pstype = <ls_key>-value.
  ENDIF.

  READ TABLE it_key_tab ASSIGNING <ls_key>
    WITH KEY name = 'IP_POOWNER'.

  IF sy-subrc = 0.
    rs_request-poowner = <ls_key>-value.
  ENDIF.

  " Paging

  rs_request-paging-top =
    io_tech_request_context->get_top( ).

  rs_request-paging-skip =
    io_tech_request_context->get_skip( ).

  rs_request-paging-count_requested =
    io_tech_request_context->has_inlinecount( ).

  " OData filtre

  DATA(lt_filters) =
    io_tech_request_context->get_filter( )->get_filter_select_options( ).

  LOOP AT lt_filters ASSIGNING FIELD-SYMBOL(<ls_filter>).

    CASE to_upper( <ls_filter>-property ).

      WHEN 'PONUMBER'.

        LOOP AT <ls_filter>-select_options
          ASSIGNING FIELD-SYMBOL(<ls_ponumber>).

          IF <ls_ponumber>-option = 'CP'.

            DATA(l_ponumber_pattern) =
              CONV string( <ls_ponumber>-low ).

            IF strlen( l_ponumber_pattern ) >= 2.

              DATA(l_ponumber_last_offset) =
                strlen( l_ponumber_pattern ) - 1.

            ENDIF.

            IF <ls_ponumber>-sign = 'I'
               AND strlen( l_ponumber_pattern ) >= 2
               AND l_ponumber_pattern NA '+'
               AND l_ponumber_pattern(1) = '*'
               AND l_ponumber_pattern+l_ponumber_last_offset(1) = '*'.

              APPEND VALUE #(
                value = CONV zif_fi_das_dashboard=>t_text_500(
                  replace(
                    val  = l_ponumber_pattern
                    sub  = '*'
                    with = '' ) ) ) TO rs_request-po_contains.

            ELSE.

              APPEND VALUE #(
                sign   = <ls_ponumber>-sign
                option = <ls_ponumber>-option
                low    = CONV ebeln( <ls_ponumber>-low )
                high   = CONV ebeln( <ls_ponumber>-high )
              ) TO rs_request-ponumber.

            ENDIF.

          ELSE.

            APPEND VALUE #(
              sign   = <ls_ponumber>-sign
              option = <ls_ponumber>-option
              low    = CONV ebeln( <ls_ponumber>-low )
              high   = CONV ebeln( <ls_ponumber>-high )
            ) TO rs_request-ponumber.

          ENDIF.

        ENDLOOP.

      WHEN 'ITEMNO'.

        LOOP AT <ls_filter>-select_options
          ASSIGNING FIELD-SYMBOL(<ls_itemno>).

          IF <ls_itemno>-option = 'CP'.

            DATA(l_itemno_pattern) =
              CONV string( <ls_itemno>-low ).

            IF strlen( l_itemno_pattern ) >= 2.

              DATA(l_itemno_last_offset) =
                strlen( l_itemno_pattern ) - 1.

            ENDIF.

            IF <ls_itemno>-sign = 'I'
               AND strlen( l_itemno_pattern ) >= 2
               AND l_itemno_pattern NA '+'
               AND l_itemno_pattern(1) = '*'
               AND l_itemno_pattern+l_itemno_last_offset(1) = '*'.

              APPEND VALUE #(
                value = CONV zif_fi_das_dashboard=>t_text_500(
                  replace(
                    val  = l_itemno_pattern
                    sub  = '*'
                    with = '' ) ) ) TO rs_request-item_contains.

            ELSE.

              APPEND VALUE #(
                sign   = <ls_itemno>-sign
                option = <ls_itemno>-option
                low    = CONV ebelp( <ls_itemno>-low )
                high   = CONV ebelp( <ls_itemno>-high )
              ) TO rs_request-itemno.

            ENDIF.

          ELSE.

            APPEND VALUE #(
              sign   = <ls_itemno>-sign
              option = <ls_itemno>-option
              low    = CONV ebelp( <ls_itemno>-low )
              high   = CONV ebelp( <ls_itemno>-high )
            ) TO rs_request-itemno.

          ENDIF.

        ENDLOOP.

      WHEN 'SUPPLIERNAME'.

        LOOP AT <ls_filter>-select_options
          ASSIGNING FIELD-SYMBOL(<ls_supplier>).

          APPEND VALUE #(
            value = CONV zif_fi_das_dashboard=>t_supplier(
                      <ls_supplier>-low )
          ) TO rs_request-supplier.

        ENDLOOP.

    ENDCASE.

  ENDLOOP.

  LOOP AT io_tech_request_context->get_orderby( )
    ASSIGNING FIELD-SYMBOL(<ls_orderby>).

    DATA(l_property) =
      to_upper( CONV string( <ls_orderby>-property ) ).

    IF l_property = 'REVIWED_BY'.
      l_property = 'REVIEWED_BY'.
    ENDIF.

    IF line_exists( lt_supported_orderby[ table_line = l_property ] ).
      rs_request-orderby-property = l_property.
      rs_request-orderby-descending =
        xsdbool(
          to_upper( CONV string( <ls_orderby>-order ) ) CS 'DESC' ).
      EXIT.
    ENDIF.

  ENDLOOP.

  DATA(lt_selected_properties) =
    io_tech_request_context->get_select( ).

  LOOP AT lt_selected_properties ASSIGNING FIELD-SYMBOL(<lv_property>).

    DATA(l_selected_property) =
      to_upper( CONV string( <lv_property> ) ).

    IF l_selected_property = 'REVIWED_BY'.
      l_selected_property = 'REVIEWED_BY'.
    ENDIF.

    APPEND VALUE #(
      name = l_selected_property ) TO rs_request-select_fields.

  ENDLOOP.

  SORT rs_request-select_fields BY name.
  DELETE ADJACENT DUPLICATES FROM rs_request-select_fields COMPARING name.

ENDMETHOD.


METHOD exclusionposet_get_entityset.

  DATA: ls_request   TYPE zif_fi_das_dashboard=>ts_request,
        lo_dashboard TYPE REF TO zif_fi_das_dashboard.

  ls_request = build_request(
    io_tech_request_context = io_tech_request_context
    it_key_tab              = it_key_tab ).

  CREATE OBJECT lo_dashboard
    TYPE zcl_fi_das_dashboard.

  lo_dashboard->exclude(
    EXPORTING
      is_request    = ls_request
    IMPORTING
      et_result     = DATA(lt_result)
      e_total_count = DATA(lv_count) ).

  et_entityset = CORRESPONDING #( lt_result ).

  es_response_context-inlinecount = lv_count.

ENDMETHOD.
