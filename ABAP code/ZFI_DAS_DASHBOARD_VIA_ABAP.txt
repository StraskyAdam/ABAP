  LOOP AT SCREEN.

    IF screen-group1 = 'PTY'.

      IF p_info = abap_true.
        screen-active = 0.
        screen-input  = 0.
      ELSE.
        screen-active = 1.
        screen-input  = 1.
      ENDIF.

      MODIFY SCREEN.

    ENDIF.

    IF screen-group1 = 'XSF'.

      IF p_xnone = abap_true.
        screen-active = 0.
        screen-input  = 0.
      ELSE.
        screen-active = 1.
        screen-input  = 1.
      ENDIF.

      MODIFY SCREEN.

    ENDIF.

  ENDLOOP.

AT SELECTION-SCREEN.

  DATA:
    ls_po   LIKE LINE OF s_po,

    ls_supp LIKE LINE OF s_supp.

  IF p_email IS INITIAL.
    MESSAGE 'User e-mail must not be empty' TYPE 'E'.
  ENDIF.

  IF p_skip < 0.
    MESSAGE 'Skip must not be negative' TYPE 'E'.
  ENDIF.

  IF p_top <= 0.
    MESSAGE 'Top must be greater than zero' TYPE 'E'.
  ENDIF.

  IF sy-ucomm = 'ONLI'
     AND p_xnone = abap_false
     AND p_xfile IS INITIAL.

    MESSAGE 'Export file must be entered for an export'
      TYPE 'E'.

  ENDIF.

  LOOP AT s_po INTO ls_po.

    IF ls_po-sign <> 'I'
       OR ls_po-option <> 'EQ'.

      MESSAGE 'PO Number supports only included single values'
        TYPE 'E'.

    ENDIF.

  ENDLOOP.

  LOOP AT s_item INTO ls_item.

    IF ls_item-sign <> 'I'
       OR ls_item-option <> 'EQ'.

      MESSAGE 'Line Item supports only included single values'
        TYPE 'E'.

    ENDIF.

  ENDLOOP.

  LOOP AT s_supp INTO ls_supp.

    IF ls_supp-sign <> 'I'
       OR ls_supp-option <> 'EQ'.

      MESSAGE 'Supplier supports only included single values'
        TYPE 'E'.

    ENDIF.

  ENDLOOP.

*---------------------------------------------------------------------*

CLASS lcl_application IMPLEMENTATION.

  METHOD run.

    DATA:
      lx_das_error TYPE REF TO zcx_fi_das_error,
      lx_error     TYPE REF TO cx_root,
      lv_message   TYPE string.

    TRY.

        validate_selection( ).
        create_provider( ).
        read_data( ).
        export_result( ).
        display_result( ).

      CATCH zcx_fi_das_error INTO lx_das_error.

        lv_message =
          lx_das_error->reason.

        IF lv_message IS INITIAL.
          lv_message =
            lx_das_error->get_text( ).
        ENDIF.

        IF lv_message IS INITIAL.
          lv_message =
            'Unknown DAS dashboard error'.
        ENDIF.

        MESSAGE lv_message TYPE 'S' DISPLAY LIKE 'E'.

      CATCH cx_root INTO lx_error.

        lv_message =
          lx_error->get_text( ).

        IF lv_message IS INITIAL.
          lv_message =
            'Unexpected dashboard error'.
        ENDIF.

        MESSAGE lv_message TYPE 'S' DISPLAY LIKE 'E'.

    ENDTRY.

  ENDMETHOD.


  METHOD create_provider.

    CREATE OBJECT mo_dashboard
      TYPE zcl_fi_das_dashboard.

  ENDMETHOD.


  METHOD validate_selection.

    IF p_email IS INITIAL.

      RAISE EXCEPTION TYPE zcx_fi_das_error
        EXPORTING
          i_reason =
            'User e-mail must not be empty'.

    ENDIF.

    IF p_skip < 0.

      RAISE EXCEPTION TYPE zcx_fi_das_error
        EXPORTING
          i_reason =
            'Skip must not be negative'.

    ENDIF.

    IF p_top <= 0.

      RAISE EXCEPTION TYPE zcx_fi_das_error
        EXPORTING
          i_reason =
            'Top must be greater than zero'.

    ENDIF.

    IF p_info = abap_false
       AND p_serv = abap_false
       AND p_mat = abap_false.

      RAISE EXCEPTION TYPE zcx_fi_das_error
        EXPORTING
          i_reason =
            'Select Service or Material PO type'.

    ENDIF.

  ENDMETHOD.


  METHOD build_request.

    DATA:
      ls_po       LIKE LINE OF s_po,
      ls_item     LIKE LINE OF s_item,
      ls_supplier LIKE LINE OF s_supp.

    CLEAR rs_request.

    rs_request-email =
      CONV zif_fi_das_dashboard=>t_email(
        p_email ).

    rs_request-pstype =
      i_pstype.

    IF p_owner = abap_true.

      rs_request-poowner =
        zif_fi_das_dashboard=>mc_owner_mode.

    ENDIF.

    rs_request-paging-skip =
      p_skip.

    rs_request-paging-top =
      p_top.

    rs_request-paging-count_requested =
      p_count.

    LOOP AT s_po INTO ls_po
      WHERE sign = 'I'
        AND option = 'EQ'.

      IF ls_po-low IS INITIAL.
        CONTINUE.
      ENDIF.

      APPEND VALUE #(
        sign   = 'I'
        option = 'EQ'
        low    = ls_po-low
      ) TO rs_request-ponumber.

    ENDLOOP.

    LOOP AT s_item INTO ls_item
      WHERE sign = 'I'
        AND option = 'EQ'.

      IF ls_item-low IS INITIAL.
        CONTINUE.
      ENDIF.

      APPEND VALUE #(
        sign   = 'I'
        option = 'EQ'
        low    = ls_item-low
      ) TO rs_request-itemno.

    ENDLOOP.

    LOOP AT s_supp INTO ls_supplier
      WHERE sign = 'I'
        AND option = 'EQ'.

      IF ls_supplier-low IS INITIAL.
        CONTINUE.
      ENDIF.

      APPEND VALUE #(
        value =
          CONV zif_fi_das_dashboard=>t_supplier(
            ls_supplier-low )
      ) TO rs_request-supplier_contains.

    ENDLOOP.

  ENDMETHOD.


  METHOD get_section_text.

    IF p_info = abap_true.

      r_section =
        'For Information Only'.

    ELSEIF p_over = abap_true
       AND p_serv = abap_true.

      r_section =
        'Over Threshold - Indirect Services'.

    ELSEIF p_over = abap_true.

      r_section =
        'Over Threshold - Indirect Materials'.

    ELSEIF p_serv = abap_true.

      r_section =
        'Under Threshold - Indirect Services'.

    ELSE.

      r_section =
        'Under Threshold - Indirect Materials'.

    ENDIF.

  ENDMETHOD.

METHOD timestamp_to_text.

  DATA lv_ts TYPE string.

  CLEAR r_text.

  IF i_timestamp IS INITIAL.
    RETURN.
  ENDIF.

  lv_ts = |{ i_timestamp }|.

  IF strlen( lv_ts ) < 14.
    RETURN.
  ENDIF.

  r_text =
      lv_ts+6(2)
    && '.'
    && lv_ts+4(2)
    && '.'
    && lv_ts+0(4)
    && ' '
    && lv_ts+8(2)
    && ':'
    && lv_ts+10(2)
    && ':'
    && lv_ts+12(2).

ENDMETHOD.

*      RETURN.
*    ENDIF.
*
**--------------------------------------------------------------------*

** contract. Do not convert it once more from UTC.

*

  METHOD read_data.

    CLEAR:
      mt_result,
      mt_over,
      mt_under,
      mt_exclude,
      m_total_count.

    IF p_info = abap_true.

      read_exclude( ).

      RETURN.

    ENDIF.

    IF p_serv = abap_true.

      IF p_over = abap_true.

        read_over(
          i_pstype =
            zif_fi_das_dashboard=>mc_pstype_service ).

      ELSE.

        read_under(
          i_pstype =
            zif_fi_das_dashboard=>mc_pstype_service ).

      ENDIF.

    ELSE.

      IF p_over = abap_true.

        read_over(
          i_pstype =
            zif_fi_das_dashboard=>mc_pstype_material ).

      ELSE.

        read_under(
          i_pstype =
            zif_fi_das_dashboard=>mc_pstype_material ).

      ENDIF.

    ENDIF.

  ENDMETHOD.


  METHOD read_over.

    DATA ls_request TYPE zif_fi_das_dashboard=>ts_request.

    CLEAR:
      mt_over,
      m_total_count.

    ls_request =
      build_request(
        i_pstype = i_pstype ).

    mo_dashboard->over(
      EXPORTING
        is_request    = ls_request
      IMPORTING
        et_result     = mt_over
        e_total_count = m_total_count ).

    append_over_page(
      it_page = mt_over ).

  ENDMETHOD.


  METHOD read_under.

    DATA ls_request TYPE zif_fi_das_dashboard=>ts_request.

    CLEAR:
      mt_under,
      m_total_count.

    ls_request =
      build_request(
        i_pstype = i_pstype ).

    mo_dashboard->under(
      EXPORTING
        is_request    = ls_request
      IMPORTING
        et_result     = mt_under
        e_total_count = m_total_count ).

    append_under_page(
      it_page = mt_under ).

  ENDMETHOD.


  METHOD read_exclude.

    DATA ls_request TYPE zif_fi_das_dashboard=>ts_request.

    CLEAR:
      mt_exclude,
      m_total_count.

    ls_request =
      build_request(
        i_pstype =
          zif_fi_das_dashboard=>mc_pstype_all ).

    mo_dashboard->exclude(
      EXPORTING
        is_request    = ls_request
      IMPORTING
        et_result     = mt_exclude
        e_total_count = m_total_count ).

    append_exclude_page(
      it_page = mt_exclude ).

  ENDMETHOD.


  METHOD append_over_page.

    DATA:
      ls_source TYPE zif_fi_das_dashboard=>ts_over,
      ls_target TYPE ts_alv_result.

    LOOP AT it_page INTO ls_source.

      CLEAR ls_target.

      MOVE-CORRESPONDING ls_source
        TO ls_target.

      ls_target-reviewed_by =
        ls_source-reviwed_by.

      ls_target-last_update_cet =
        timestamp_to_text(
          i_timestamp = ls_source-changed_on ).

      ls_target-changed_on_cet =
        timestamp_to_text(
          i_timestamp = ls_source-changed_on ).

      APPEND ls_target
        TO mt_result.

    ENDLOOP.

  ENDMETHOD.


  METHOD append_under_page.

    DATA:
      ls_source TYPE zif_fi_das_dashboard=>ts_under,
      ls_target TYPE ts_alv_result.

    LOOP AT it_page INTO ls_source.

      CLEAR ls_target.

      MOVE-CORRESPONDING ls_source
        TO ls_target.

      ls_target-reviewed_by =
        ls_source-reviwed_by.

      ls_target-last_update_cet =
        timestamp_to_text(
          i_timestamp = ls_source-changed_on ).

      ls_target-changed_on_cet =
        timestamp_to_text(
          i_timestamp = ls_source-changed_on ).

      APPEND ls_target
        TO mt_result.

    ENDLOOP.

  ENDMETHOD.


  METHOD append_exclude_page.

    DATA:
      ls_source TYPE zif_fi_das_dashboard=>ts_exclude,
      ls_target TYPE ts_alv_result.

    LOOP AT it_page INTO ls_source.

      CLEAR ls_target.

      MOVE-CORRESPONDING ls_source
        TO ls_target.

      ls_target-changed_on_cet =
        timestamp_to_text(
          i_timestamp = ls_source-changed_on ).

      CLEAR:
        ls_target-reviewed_by,
        ls_target-last_update_cet,
        ls_target-audit_log_action.

      APPEND ls_target
        TO mt_result.

    ENDLOOP.

  ENDMETHOD.


  METHOD export_result.

    DATA lv_json TYPE string.

    IF p_xnone = abap_true.
      RETURN.
    ENDIF.

*    IF p_xcsv = abap_true.
*
*      RAISE EXCEPTION TYPE zcx_fi_das_error
*        EXPORTING
*          i_reason =
*            'CSV export is unavailable in the new dashboard interface'.
*
*    ENDIF.

    IF p_info = abap_true.

      lv_json =
        /ui2/cl_json=>serialize(
          data = mt_exclude ).

    ELSEIF p_over = abap_true.

      lv_json =
        /ui2/cl_json=>serialize(
          data = mt_over ).

    ELSE.

      lv_json =
        /ui2/cl_json=>serialize(
          data = mt_under ).

    ENDIF.

    download_json(
      i_json = lv_json ).

  ENDMETHOD.


  METHOD download_json.

    DATA lt_lines TYPE stringtab.

    IF p_xfile IS INITIAL.

      RAISE EXCEPTION TYPE zcx_fi_das_error
        EXPORTING
          i_reason =
            'Export file must be entered'.

    ENDIF.

    APPEND i_json
      TO lt_lines.

    cl_gui_frontend_services=>gui_download(
      EXPORTING
        filename  = p_xfile
        filetype  = 'ASC'
        codepage  = '4110'
        write_bom = abap_false
        write_lf  = abap_true
      CHANGING
        data_tab  = lt_lines
      EXCEPTIONS
        OTHERS    = 1 ).

    IF sy-subrc <> 0.

      RAISE EXCEPTION TYPE zcx_fi_das_error
        EXPORTING
          i_reason =
            |Export to { p_xfile } failed|.

    ENDIF.

    MESSAGE |JSON export written: { p_xfile }|
      TYPE 'S'.

  ENDMETHOD.


  METHOD display_result.

    DATA:
      lo_salv       TYPE REF TO cl_salv_table,
      lo_functions  TYPE REF TO cl_salv_functions_list,
      lo_columns    TYPE REF TO cl_salv_columns_table,
      lo_display    TYPE REF TO cl_salv_display_settings,
      lo_selections TYPE REF TO cl_salv_selections,
      lv_title      TYPE lvc_title,
      lv_lines      TYPE i,
      lv_first_row  TYPE int8,
      lv_last_row   TYPE int8,
      lv_section    TYPE string.

    IF mt_result IS INITIAL.

      IF p_count = abap_true.

        MESSAGE
          |No records found. Total filtered records: { m_total_count }|
          TYPE 'S'.

      ELSE.

        MESSAGE 'No records found for the requested page'
          TYPE 'S'.

      ENDIF.

      RETURN.

    ENDIF.

    DESCRIBE TABLE mt_result
      LINES lv_lines.

    lv_first_row =
      p_skip + 1.

    lv_last_row =
      p_skip + lv_lines.

    lv_section =
      get_section_text( ).

    cl_salv_table=>factory(
      IMPORTING
        r_salv_table = lo_salv
      CHANGING
        t_table      = mt_result ).

    lo_functions =
      lo_salv->get_functions( ).

    lo_functions->set_all(
      abap_true ).

    lo_columns =
      lo_salv->get_columns( ).

    configure_columns(
      io_columns = lo_columns ).

    lo_columns->set_optimize(
      abap_true ).

    lo_display =
      lo_salv->get_display_settings( ).

    lo_display->set_striped_pattern(
      abap_true ).

    IF p_count = abap_true.

      lv_title =
          |{ lv_section }: rows { lv_first_row }-{ lv_last_row } |
        && |of { m_total_count }|.

    ELSE.

      lv_title =
        |{ lv_section }: rows { lv_first_row }-{ lv_last_row }|.

    ENDIF.

    lo_display->set_list_header(

  METHOD make_all_columns_visible.

    DATA:
      lt_column_refs TYPE salv_t_column_ref,
      ls_column_ref  TYPE salv_s_column_ref,
      lo_column      TYPE REF TO cl_salv_column_table.

    lt_column_refs =
      io_columns->get( ).

      INTO ls_column_ref.

  METHOD apply_ui_order.

    DATA:
      lv_index  TYPE i,
      lv_column TYPE t_column_name.

    lv_index =
      lines( it_columns ).

    WHILE lv_index > 0.

      READ TABLE it_columns
        INTO lv_column
        INDEX lv_index.

      IF sy-subrc = 0.

        TRY.

            io_columns->set_column_position(
              columnname = lv_column
              position   = 1 ).

          CATCH cx_salv_not_found.
          CATCH cx_salv_error.

        ENDTRY.

      ENDIF.

      lv_index =
        lv_index - 1.

    ENDWHILE.

  ENDMETHOD.
