* Maintain these text elements when installing/updating the report in SAP:
* T01 = Dashboard Selection; T02 = PO Type; T03 = Filters
* T04 = Native JSON Page Export; E04 = Export file
* Selection texts:
* P_OVER = Over Threshold; P_UNDER = Under Threshold
* P_INFO = For Information Only; P_SERV = Services; P_MAT = Materials
* P_EMAIL = User e-mail; P_OWNER = PO owner mode
* P_SKIP = Rows to skip; P_TOP = Page size (0 = no rows)
* P_COUNT = Include total filtered count
* S_PO = PO number; S_ITEM = Item
* S_SUPP = Supplier name (contains, OR)
* P_XNONE = No export; P_XJSON = Export native JSON page
* PO and item support full select-options (inclusions, exclusions and ranges).
* Different filter fields are ANDed; standard range semantics apply within each.
* Supplier accepts only included EQ entries.
* Supplier entries are literal, case-sensitive substrings, not CP patterns.

CLASS lcl_application IMPLEMENTATION.
  METHOD run.
    IF sy-batch = abap_true.
      MESSAGE 'This SALV/frontend report must run in dialog mode' TYPE 'S' DISPLAY LIKE 'E'.
      RETURN.
    ENDIF.
    DATA(l_error) = validate_selection( ).
    IF l_error IS NOT INITIAL.
      MESSAGE l_error TYPE 'S' DISPLAY LIKE 'E'.
      RETURN.
    ENDIF.

    TRY.
        mo_dashboard = NEW zcl_fi_das_dashboard( ).
        read_data( ).
        export_result( ).
        display_result( ).
      CATCH cx_root INTO DATA(lx_error).
        MESSAGE lx_error->get_text( ) TYPE 'S' DISPLAY LIKE 'E'.
    ENDTRY.
  ENDMETHOD.

  METHOD validate_selection.
    IF p_email IS INITIAL.
      r_error = 'User e-mail must not be empty'.
    ELSEIF p_skip < 0 OR p_top < 0.
      r_error = 'Skip and Top must not be negative (Top 0 returns no rows)'.
    ELSEIF p_info = abap_false AND p_serv = abap_false AND p_mat = abap_false.
      r_error = 'Select Service or Material PO type'.
    ELSEIF p_xjson = abap_true AND p_xfile IS INITIAL.
      r_error = 'Export file must be entered for a JSON export'.
    ENDIF.
    IF r_error IS NOT INITIAL.
      RETURN.
    ENDIF.

    LOOP AT s_supp ASSIGNING FIELD-SYMBOL(<ls_supplier>).
      IF <ls_supplier>-sign <> 'I' OR <ls_supplier>-option <> 'EQ'.
        r_error = 'Supplier contains supports only included single values'.
        RETURN.
      ENDIF.
    ENDLOOP.
  ENDMETHOD.

  METHOD build_request.
    DATA:
      lr_po       TYPE tt_string_range,
      lr_item     TYPE tt_string_range,
      lr_supplier TYPE tt_string_range.

    rs_request-parameter = VALUE #(
      email   = p_email
      poowner = p_owner
      pstype  = COND #( WHEN p_info = abap_true THEN zif_fi_das_dashboard=>mc_pstype_all
                       WHEN p_serv = abap_true THEN zif_fi_das_dashboard=>mc_pstype_service
                       ELSE zif_fi_das_dashboard=>mc_pstype_material ) ).
    rs_request-paging = VALUE #(
      skip            = p_skip
      top             = p_top
      top_requested   = abap_true
      count_requested = p_count ).

    lr_po = CORRESPONDING #( s_po[] ).
    lr_item = CORRESPONDING #( s_item[] ).
    LOOP AT s_supp ASSIGNING FIELD-SYMBOL(<ls_supplier>) WHERE low IS NOT INITIAL.
*     Escape CP metacharacters so selection entries remain literal substrings.
      DATA(l_pattern) = CONV string( <ls_supplier>-low ).
      REPLACE ALL OCCURRENCES OF '#' IN l_pattern WITH '##'.
      REPLACE ALL OCCURRENCES OF '*' IN l_pattern WITH '#*'.
      REPLACE ALL OCCURRENCES OF '+' IN l_pattern WITH '#+'.
      APPEND VALUE #( sign = 'I' option = 'CP' low = |*{ l_pattern }*| ) TO lr_supplier.
    ENDLOOP.

    rs_request-filter-filter_select_options = VALUE #(
      FOR ls_po IN lr_po
      ( property = 'PONUMBER' sign = ls_po-sign option = ls_po-option low = ls_po-low high = ls_po-high ) ).
    APPEND LINES OF VALUE zif_fi_das_dashboard=>tt_filter_select_options(
      FOR ls_item IN lr_item
      ( property = 'ITEMNO' sign = ls_item-sign option = ls_item-option low = ls_item-low high = ls_item-high ) )
      TO rs_request-filter-filter_select_options.
    APPEND LINES OF VALUE zif_fi_das_dashboard=>tt_filter_select_options(
      FOR ls_supplier IN lr_supplier
      ( property = 'SUPPLIERNAME' sign = ls_supplier-sign option = ls_supplier-option low = ls_supplier-low ) )
      TO rs_request-filter-filter_select_options.

*   Supply the real SQL fallback required by VALIDATE_REQUEST, using the same
*   range converter as the provider (quoting and full select-option semantics).
    IF rs_request-filter-filter_select_options IS NOT INITIAL.
      rs_request-filter-filter_string = cl_shdb_seltab=>combine_seltabs(
        it_named_seltabs = VALUE #(
          ( name = 'PONUMBER' dref = REF #( lr_po ) )
          ( name = 'ITEMNO' dref = REF #( lr_item ) )
          ( name = 'SUPPLIERNAME' dref = REF #( lr_supplier ) ) ) ).
    ENDIF.
  ENDMETHOD.

  METHOD read_data.
    CLEAR: mt_result, mt_over, mt_under, mt_exclude, m_total_count.
    DATA(ls_request) = build_request( ).

    IF p_info = abap_true.
      mo_dashboard->exclude(
        EXPORTING is_request = ls_request
        IMPORTING et_result = mt_exclude e_total_count = m_total_count ).
      mt_result = CORRESPONDING #( mt_exclude ).
    ELSEIF p_over = abap_true.
      mo_dashboard->over(
        EXPORTING is_request = ls_request
        IMPORTING et_result = mt_over e_total_count = m_total_count ).
      mt_result = CORRESPONDING #( mt_over ).
    ELSE.
      mo_dashboard->under(
        EXPORTING is_request = ls_request
        IMPORTING et_result = mt_under e_total_count = m_total_count ).
      mt_result = CORRESPONDING #( mt_under ).
    ENDIF.
  ENDMETHOD.

  METHOD get_result_json.
*   Serialize the native page, never the broader SALV union.
    IF p_info = abap_true.
      r_json = /ui2/cl_json=>serialize( data = mt_exclude ).
    ELSEIF p_over = abap_true.
      r_json = /ui2/cl_json=>serialize( data = mt_over ).
    ELSE.
      r_json = /ui2/cl_json=>serialize( data = mt_under ).
    ENDIF.
  ENDMETHOD.

  METHOD export_result.
    IF p_xjson = abap_false.
      RETURN.
    ENDIF.

    DATA(lt_lines) = VALUE stringtab( ( get_result_json( ) ) ).
    cl_gui_frontend_services=>gui_download(
      EXPORTING
        filename          = p_xfile
        filetype          = 'ASC'
        codepage          = '4110'
        write_bom         = abap_false
        write_lf          = abap_true
        confirm_overwrite = abap_true
      CHANGING
        data_tab          = lt_lines
      EXCEPTIONS
        OTHERS            = 1 ).

    IF sy-subrc <> 0.
*     A cancelled/failed export must not prevent viewing the fetched page.
      MESSAGE |JSON export cancelled or failed: { p_xfile }| TYPE 'S' DISPLAY LIKE 'E'.
    ELSE.
      MESSAGE |JSON export written: { p_xfile }| TYPE 'S'.
    ENDIF.
  ENDMETHOD.

  METHOD get_section_text.
    IF p_info = abap_true.
      r_section = 'For Information Only'.
    ELSE.
      r_section = COND #( WHEN p_over = abap_true THEN 'Over Threshold' ELSE 'Under Threshold' ).
      r_section = r_section && COND string( WHEN p_serv = abap_true THEN ' - Services' ELSE ' - Materials' ).
    ENDIF.
  ENDMETHOD.

  METHOD display_result.
    IF mt_result IS INITIAL.
      IF p_count = abap_true.
        MESSAGE |No rows on this page. Total filtered records: { m_total_count }| TYPE 'S'.
      ELSE.
        MESSAGE 'No rows on the requested page' TYPE 'S'.
      ENDIF.
      RETURN.
    ENDIF.

    cl_salv_table=>factory(
      IMPORTING r_salv_table = DATA(lo_salv)
      CHANGING t_table = mt_result ).
    lo_salv->get_functions( )->set_all( abap_true ).
    DATA(lo_columns) = lo_salv->get_columns( ).
    configure_columns( lo_columns ).
    lo_columns->set_optimize( abap_true ).

    DATA(lo_display) = lo_salv->get_display_settings( ).
    lo_display->set_striped_pattern( abap_true ).
    DATA(l_first_row) = CONV int8( p_skip ) + 1.
    DATA(l_last_row) = CONV int8( p_skip ) + lines( mt_result ).
    DATA(l_title) = |{ get_section_text( ) }: rows { l_first_row }-{ l_last_row }|.
    IF p_count = abap_true.
      l_title = |{ l_title } of { m_total_count }|.
    ENDIF.
    lo_display->set_list_header( CONV lvc_title( l_title ) ).
    lo_salv->get_selections( )->set_selection_mode( if_salv_c_selection_mode=>row_column ).
    lo_salv->display( ).
  ENDMETHOD.

  METHOD configure_columns.
    DATA(lt_columns) = get_ui_columns( ).
    DATA(lt_refs) = io_columns->get( ).
    LOOP AT lt_refs INTO DATA(ls_ref).
*     Hide fields belonging only to another flow; keep all native flow fields.
      ls_ref-r_column->set_visible( xsdbool( line_exists( lt_columns[ table_line = ls_ref-columnname ] ) ) ).
    ENDLOOP.
    set_column_labels( io_columns ).

    LOOP AT lt_columns INTO DATA(l_column).
      DATA(l_position) = sy-tabix.
      TRY.
          io_columns->set_column_position( columnname = l_column position = l_position ).
        CATCH cx_salv_not_found.
          MESSAGE |Missing SALV column: { l_column }| TYPE 'S' DISPLAY LIKE 'E'.
      ENDTRY.
    ENDLOOP.
  ENDMETHOD.

  METHOD set_column_label.
