CLASS lcl_exclusion IMPLEMENTATION.

  METHOD run.

    DATA lt_result TYPE tt_result.

    TRY.

        zcl_fi_das_utility=>check_table_authority( mc_target ).

        m_used_replication = zcl_fi_das_dashboard=>get_use_replication( ).

        DATA(lt_rules) = read_rules( ).

        APPEND LINES OF get_results( i_pstype = zif_fi_das_dashboard=>mc_pstype_material
                                     it_rules = lt_rules ) TO lt_result.

        APPEND LINES OF get_results( i_pstype = zif_fi_das_dashboard=>mc_pstype_service
                                     it_rules = lt_rules ) TO lt_result.

        IF i_test_mode = abap_true.

          m_deleted  = get_count( ).
          m_inserted = lines( lt_result ).

        ELSE.

          lock( ).

          replace_snapshot( lt_result ).

          COMMIT WORK AND WAIT.

          IF sy-subrc <> 0.
            RAISE EXCEPTION TYPE zcx_fi_das_error MESSAGE e031(zfi_das) WITH mc_target.
          ENDIF.

          unlock( ).

        ENDIF.

      CATCH zcx_fi_das_error INTO DATA(lx_das_error).

        IF m_locked = abap_true.
          ROLLBACK WORK.
          unlock( ).
        ENDIF.

        RAISE EXCEPTION lx_das_error.

      CATCH cx_root INTO DATA(lx_unexpected_error).

        IF m_locked = abap_true.
          ROLLBACK WORK.
          unlock( ).
        ENDIF.

        RAISE EXCEPTION TYPE zcx_fi_das_error
          EXPORTING
            previous = lx_unexpected_error
            i_text   = lx_unexpected_error->get_text( ).

    ENDTRY.

    summary( i_test_mode ).

  ENDMETHOD.

  METHOD read_rules.

    DATA lt_rules TYPE STANDARD TABLE OF ts_rule_internal WITH EMPTY KEY.

    SELECT ruletype, bukrsfr, bukrsto, saktofr, saktoto,
           kostlfr, kostlto, aufnrfr, aufnrto,
           prctrfr, prctrto, ponum, lifnr
      FROM zi_fi_das_accrual_excl( p_use_replication = @m_used_replication )
    INTO TABLE @lt_rules.

    IF sy-subrc <> 0.
      CLEAR lt_rules.
    ENDIF.

    LOOP AT lt_rules ASSIGNING FIELD-SYMBOL(<fs_rule>).
      APPEND normalize_rule( is_rule = <fs_rule> ) TO rt_rules.
    ENDLOOP.

  ENDMETHOD.

  METHOD normalize_rule.

    rs_rule = VALUE #(
      kind = is_rule-ruletype

      company = VALUE #(
        low  = VALUE #( value = is_rule-bukrsfr )
        high = VALUE #( value = is_rule-bukrsto ) )

      account = VALUE #(
        low  = VALUE #( value = is_rule-saktofr )
        high = VALUE #( value = is_rule-saktoto ) )

      cost = VALUE #(
        low  = VALUE #( value = is_rule-kostlfr )
        high = VALUE #( value = is_rule-kostlto ) )

      internal_order = VALUE #(
        low  = VALUE #( value = is_rule-aufnrfr )
        high = VALUE #( value = is_rule-aufnrto ) )

      profit = VALUE #(
        low  = VALUE #( value = is_rule-prctrfr )
        high = VALUE #( value = is_rule-prctrto ) )

      po = strip_zero( is_value = VALUE #( value = is_rule-ponum ) )

      vendor = strip_zero( is_value = VALUE #( value = is_rule-lifnr ) ) ).

    rs_rule-company = normalize( is_range = rs_rule-company
                                 i_low    = mc_company_low
                                 i_high   = mc_company_high ).

    CASE rs_rule-kind.

      WHEN mc_gl.

        rs_rule-account = normalize( is_range = rs_rule-account
                                     i_low    = mc_account_low
                                     i_high   = mc_account_high ).

        rs_rule-account-low = strip_zero( is_value = rs_rule-account-low ).

        rs_rule-account-high = strip_zero( is_value = rs_rule-account-high ).

      WHEN mc_cc.

        rs_rule-cost = normalize( is_range = rs_rule-cost
                                  i_low    = mc_cost_low
                                  i_high   = mc_cost_high ).

        rs_rule-cost-low = strip_zero( rs_rule-cost-low ).

        rs_rule-cost-high = strip_zero( rs_rule-cost-high ).

      WHEN mc_iv.

        rs_rule-internal_order = normalize( is_range = rs_rule-internal_order
                                            i_low    = mc_order_low
                                            i_high   = mc_order_high ).

        rs_rule-internal_order-low = strip_zero( rs_rule-internal_order-low ).

        rs_rule-internal_order-high = strip_zero( rs_rule-internal_order-high ).

      WHEN mc_pc.

        rs_rule-profit = normalize( is_range = rs_rule-profit
                                    i_low    = mc_profit_low
                                    i_high   = mc_profit_high ).

        rs_rule-profit-low = strip_zero( rs_rule-profit-low ).

        rs_rule-profit-high = strip_zero( rs_rule-profit-high ).

    ENDCASE.

  ENDMETHOD.

  METHOD normalize.

    rs_range = is_range.

    IF is_range-low-is_null = abap_false
       AND is_range-high-is_null = abap_false
       AND ( is_range-low-value = mc_empty
          OR is_range-low-value = mc_blank )
       AND ( is_range-high-value = mc_empty
          OR is_range-high-value = mc_blank ).

      rs_range-low-value  = i_low.
      rs_range-high-value = i_high.

    ENDIF.

  ENDMETHOD.

  METHOD strip_zero.

    rs_value = is_value.

    IF rs_value-is_null = abap_false.
      SHIFT rs_value-value LEFT DELETING LEADING mc_zero.
    ENDIF.

  ENDMETHOD.

  METHOD get_results.

    DATA(lt_keys) = get_keys( i_pstype ).
    DATA(lt_items) = enrich_keys( lt_keys ).
    FREE lt_keys.

    DATA(lt_vectors) = evaluate( it_items = lt_items
                                 it_rules = it_rules ).
    FREE lt_items.

    rt_result = get_rules( it_vectors = lt_vectors
                           i_pstype   = i_pstype ).

  ENDMETHOD.

  METHOD get_keys.

    DATA lr_dashboard TYPE REF TO zcl_fi_das_dashboard.
    CREATE OBJECT lr_dashboard.

    DATA(lt_candidates) = lr_dashboard->get_exclusion_candidates( i_mode   = zif_fi_das_dashboard=>mc_over
                                                                  i_pstype = i_pstype ).

    APPEND LINES OF lr_dashboard->get_exclusion_candidates( i_mode   = zif_fi_das_dashboard=>mc_under
                                                            i_pstype = i_pstype ) TO lt_candidates.

    LOOP AT lt_candidates ASSIGNING FIELD-SYMBOL(<fs_candidate>).

      INSERT VALUE ts_key( ponumber = <fs_candidate>-ponumber
                           itemno   = <fs_candidate>-itemno ) INTO TABLE rt_keys.

    ENDLOOP.

  ENDMETHOD.

  METHOD enrich_keys.

    IF it_keys IS INITIAL.
      RETURN.
    ENDIF.

    SELECT ekpo~ebeln AS ponumber,
           ekpo~ebelp AS itemno,
           t001~bukrs AS company,
           ekko~lifnr AS vendor,
           ekkn~sakto AS account,
           ekkn~kostl AS cost,
           ekkn~aufnr AS internal_order,
           ekkn~prctr AS profit,
           ekko~ebeln AS ekko_exists,
           t001~bukrs AS bukrs_exists,
           ekkn~ebeln AS ekkn_exists
      FROM zi_fi_das_ekpo( p_use_replication = @m_used_replication ) AS ekpo
        LEFT OUTER JOIN zi_fi_das_ekko( p_use_replication = @m_used_replication ) AS ekko ON ekko~ebeln = ekpo~ebeln
        LEFT OUTER JOIN zi_fi_das_t001( p_use_replication = @m_used_replication ) AS t001 ON t001~bukrs = ekko~bukrs
        LEFT OUTER JOIN zi_fi_das_ekkn( p_use_replication = @m_used_replication ) AS ekkn ON ekkn~ebeln = ekpo~ebeln AND
                                                                                             ekkn~ebelp = ekpo~ebelp
      FOR ALL ENTRIES IN @it_keys
        WHERE ekpo~ebeln = @it_keys-ponumber AND
              ekpo~ebelp = @it_keys-itemno
      INTO TABLE @DATA(lt_content).

    IF sy-subrc <> 0.
      CLEAR lt_content.
    ENDIF.

    rt_items = VALUE #(
      FOR ls_content IN lt_content
      ( ponumber = ls_content-ponumber
        itemno   = ls_content-itemno

        company = VALUE #(
          value   = ls_content-company
          is_null = xsdbool( ls_content-bukrs_exists IS INITIAL ) )

        account = strip_zero(
          is_value = VALUE #(
            value   = ls_content-account
            is_null = xsdbool( ls_content-ekkn_exists IS INITIAL ) ) )

        cost = strip_zero(
          is_value = VALUE #(
            value   = ls_content-cost
            is_null = xsdbool( ls_content-ekkn_exists IS INITIAL ) ) )

        internal_order = strip_zero(
          is_value = VALUE #(
            value   = ls_content-internal_order
            is_null = xsdbool( ls_content-ekkn_exists IS INITIAL ) ) )

        profit = VALUE #(
          value   = ls_content-profit
          is_null = xsdbool( ls_content-ekkn_exists IS INITIAL ) )

        vendor = strip_zero(
          is_value = VALUE #(
            value   = ls_content-vendor
            is_null = xsdbool( ls_content-ekko_exists IS INITIAL ) ) ) ) ).

  ENDMETHOD.

  METHOD evaluate.

    DATA(ls_index) = build_index( it_rules ).

    LOOP AT it_items ASSIGNING FIELD-SYMBOL(<fs_item>).

*     Only rules that can produce a match are evaluated
      DATA(lt_candidates) = find_candidates( is_item  = <fs_item>
                                             is_index = ls_index ).

      LOOP AT lt_candidates ASSIGNING FIELD-SYMBOL(<f_rule_no>).

        DATA(ls_vector) = match( is_item = <fs_item>
                                 is_rule = it_rules[ <f_rule_no> ] ).

        IF has_match( ls_vector ).
          INSERT ls_vector INTO TABLE rt_vectors.
        ENDIF.

      ENDLOOP.

    ENDLOOP.

  ENDMETHOD.

  METHOD build_index.

    LOOP AT it_rules ASSIGNING FIELD-SYMBOL(<fs_rule>).

      DATA(l_rule_no) = sy-tabix.
      DATA(ls_range)  = get_rule_range( <fs_rule> ).

*     Value of the field the rule type is about (account, cost centre, order, profit centre)
      IF ls_range-low-is_null = abap_false.
        add_to_bucket( EXPORTING i_slot     = <fs_rule>-kind
                                 i_value    = ls_range-low-value
                                 i_rule_no  = l_rule_no
                       CHANGING  ct_buckets = rs_index-buckets ).
      ENDIF.

*     Real intervals can match values other than LOW, so they are always checked
      IF ls_range-low-is_null  = abap_false AND ls_range-high-is_null = abap_false AND ls_range-low-value < ls_range-high-value.
        INSERT l_rule_no INTO TABLE rs_index-spans.
      ENDIF.

*     Purchase order and vendor are evaluated for every rule, regardless of its type
      IF <fs_rule>-po-is_null = abap_false.
        add_to_bucket( EXPORTING i_slot     = mc_po
                                 i_value    = <fs_rule>-po-value
                                 i_rule_no  = l_rule_no
                       CHANGING  ct_buckets = rs_index-buckets ).
      ENDIF.

      IF <fs_rule>-vendor-is_null = abap_false.
        add_to_bucket( EXPORTING i_slot     = mc_vd
                                 i_value    = <fs_rule>-vendor-value
                                 i_rule_no  = l_rule_no
                       CHANGING  ct_buckets = rs_index-buckets ).
      ENDIF.

    ENDLOOP.

  ENDMETHOD.

  METHOD add_to_bucket.

    READ TABLE ct_buckets ASSIGNING FIELD-SYMBOL(<fs_bucket>) WITH TABLE KEY slot  = i_slot
                                                                             value = i_value.

    IF sy-subrc <> 0.
      INSERT VALUE #( slot  = i_slot
                      value = i_value ) INTO TABLE ct_buckets ASSIGNING <fs_bucket>.
    ENDIF.

    INSERT i_rule_no INTO TABLE <fs_bucket>-rules.

  ENDMETHOD.

  METHOD get_rule_range.

    CASE is_rule-kind.

      WHEN mc_gl.
        rs_range = is_rule-account.

      WHEN mc_cc.
        rs_range = is_rule-cost.

      WHEN mc_iv.
        rs_range = is_rule-internal_order.

      WHEN mc_pc.
        rs_range = is_rule-profit.

      WHEN OTHERS.
        rs_range = VALUE #( low  = VALUE #( is_null = abap_true )
                            high = VALUE #( is_null = abap_true ) ).

    ENDCASE.

  ENDMETHOD.

  METHOD get_item_value.

    CASE i_slot.

      WHEN mc_gl.
        rs_value = is_item-account.

      WHEN mc_cc.
        rs_value = is_item-cost.

      WHEN mc_iv.
        rs_value = is_item-internal_order.

      WHEN mc_pc.
        rs_value = is_item-profit.

      WHEN mc_po.
        rs_value = VALUE #( value = is_item-ponumber ).

      WHEN mc_vd.
        rs_value = is_item-vendor.

      WHEN OTHERS.
        rs_value = VALUE #( is_null = abap_true ).

    ENDCASE.

  ENDMETHOD.

  METHOD find_candidates.

    DATA(lt_slots) = VALUE tt_slot( ( mc_gl ) ( mc_cc ) ( mc_iv )
                                    ( mc_pc ) ( mc_po ) ( mc_vd ) ).

*   Start with interval rules
    rt_rules = is_index-spans.

    LOOP AT lt_slots ASSIGNING FIELD-SYMBOL(<fs_slot>).

      DATA(ls_value) = get_item_value( is_item = is_item
                                       i_slot  = <fs_slot> ).

      IF ls_value-is_null = abap_true.
        CONTINUE.
      ENDIF.

      READ TABLE is_index-buckets ASSIGNING FIELD-SYMBOL(<fs_bucket>) WITH TABLE KEY slot  = <fs_slot>
                                                                                     value = ls_value-value.

      IF sy-subrc <> 0.
        CONTINUE.
      ENDIF.

*     A rule found through several slots is evaluated only once
      LOOP AT <fs_bucket>-rules ASSIGNING FIELD-SYMBOL(<f_rule_no>).
        INSERT <f_rule_no> INTO TABLE rt_rules.
      ENDLOOP.

    ENDLOOP.

  ENDMETHOD.

  METHOD get_rules.

    DATA l_timestamp TYPE zfi_das_po_excl-last_replicated_at.

    GET TIME STAMP FIELD l_timestamp.

    LOOP AT it_vectors ASSIGNING FIELD-SYMBOL(<fs_vector>).

      DATA(l_rule) = COND zfi_das_po_excl-ruletype(
        WHEN <fs_vector>-gl = abap_true THEN mc_gl
        WHEN <fs_vector>-cc = abap_true THEN mc_cc
        WHEN <fs_vector>-iv = abap_true THEN mc_iv
        WHEN <fs_vector>-pc = abap_true THEN mc_pc
        WHEN <fs_vector>-po = abap_true THEN mc_po
        WHEN <fs_vector>-vd = abap_true THEN mc_vd ).

      APPEND VALUE #( mandt    = sy-mandt
                      ponumber = <fs_vector>-ponumber
                      itemno   = <fs_vector>-itemno
                      ruletype = l_rule
                      pstype   = i_pstype
                      last_replicated_at = l_timestamp ) TO rt_result.
    ENDLOOP.

    SORT rt_result BY mandt ponumber itemno ruletype pstype.

    DELETE ADJACENT DUPLICATES FROM rt_result COMPARING mandt ponumber itemno ruletype pstype.

  ENDMETHOD.

  METHOD equal.

    rv_match = xsdbool(
      is_left-is_null = abap_false
      AND is_right-is_null = abap_false
      AND is_left-value = is_right-value ).

  ENDMETHOD.

  METHOD between.

    rv_match = xsdbool(
      is_value-is_null = abap_false
      AND is_range-low-is_null = abap_false
      AND is_range-high-is_null = abap_false
      AND is_value-value >= is_range-low-value
      AND is_value-value <= is_range-high-value ).

  ENDMETHOD.

  METHOD in_range.

    rv_match = xsdbool(
      equal(
        is_left  = is_value
        is_right = is_range-low ) = abap_true
      OR between(
        is_value = is_value
        is_range = is_range ) = abap_true ).

  ENDMETHOD.

  METHOD match.

    rs_vector-ponumber = is_item-ponumber.
    rs_vector-itemno   = is_item-itemno.

    DATA(lv_company_equal) = equal( is_left  = is_item-company
                                    is_right = is_rule-company-low ).

    DATA(lv_company_between) = between( is_value = is_item-company
                                        is_range = is_rule-company ).

    DATA(lv_company) = xsdbool( lv_company_equal = abap_true OR lv_company_between = abap_true ).

    CASE is_rule-kind.

      WHEN mc_gl.
        rs_vector-gl = xsdbool(
          lv_company = abap_true
          AND in_range(
            is_value = is_item-account
            is_range = is_rule-account ) = abap_true ).

      WHEN mc_cc.
        rs_vector-cc = xsdbool(
          lv_company = abap_true
          AND in_range(
            is_value = is_item-cost
            is_range = is_rule-cost ) = abap_true ).

      WHEN mc_iv.
        rs_vector-iv = xsdbool(
          lv_company = abap_true
          AND in_range(
            is_value = is_item-internal_order
            is_range = is_rule-internal_order ) = abap_true ).

      WHEN mc_pc.
        IF lv_company_equal = abap_true.

          rs_vector-pc = in_range(
            is_value = is_item-profit
            is_range = is_rule-profit ).

        ELSEIF lv_company_between = abap_true.

          rs_vector-pc = between(
            is_value = is_item-profit
            is_range = is_rule-profit ).

        ENDIF.

    ENDCASE.

    rs_vector-po = xsdbool(
      lv_company = abap_true
      AND equal(
        is_left  = VALUE ts_nullable(
          value = is_item-ponumber )
        is_right = is_rule-po ) = abap_true ).

    rs_vector-vd = xsdbool(
      lv_company = abap_true
      AND equal(
        is_left  = is_item-vendor
        is_right = is_rule-vendor ) = abap_true ).

  ENDMETHOD.

  METHOD has_match.

    rv_match = xsdbool(
      is_vector-gl = abap_true
      OR is_vector-cc = abap_true
      OR is_vector-iv = abap_true
      OR is_vector-pc = abap_true
      OR is_vector-po = abap_true
      OR is_vector-vd = abap_true ).

  ENDMETHOD.

  METHOD lock.

    zcl_fi_das_utility=>lock_table( mc_target ).

    m_locked = abap_true.

  ENDMETHOD.

  METHOD unlock.

    IF m_locked = abap_false.
      RETURN.
    ENDIF.

    zcl_fi_das_utility=>unlock_table( mc_target ).

    m_locked = abap_false.

  ENDMETHOD.

  METHOD get_count.

    SELECT COUNT( * )
      FROM zfi_das_po_excl
    INTO @r_count.

    IF sy-subrc <> 0.
      CLEAR r_count.
    ENDIF.

  ENDMETHOD.

  METHOD replace_snapshot.

    DATA lt_package  TYPE tt_result.

    DELETE FROM zfi_das_po_excl.
    IF sy-subrc = 0.
      m_deleted = sy-dbcnt.
    ELSE.
      RAISE EXCEPTION TYPE zcx_fi_das_error MESSAGE e032(zfi_das) WITH mc_target.
    ENDIF.

    CLEAR m_inserted.

    LOOP AT it_result ASSIGNING FIELD-SYMBOL(<fs_insert>).

      APPEND <fs_insert> TO lt_package.

      IF lines( lt_package ) >= mc_package_size.

        insert_package( lt_package ).
        CLEAR lt_package.

      ENDIF.

    ENDLOOP.

    IF lt_package IS NOT INITIAL.
      insert_package( it_package = lt_package ).
    ENDIF.

  ENDMETHOD.

  METHOD insert_package.

    IF it_package IS INITIAL.
      RETURN.
    ENDIF.

    INSERT zfi_das_po_excl FROM TABLE @it_package ACCEPTING DUPLICATE KEYS.

    IF sy-subrc <> 0.
      RAISE EXCEPTION TYPE zcx_fi_das_error MESSAGE e030(zfi_das) WITH mc_target.
    ENDIF.

    m_inserted = m_inserted + sy-dbcnt.

  ENDMETHOD.

  METHOD summary.

    IF i_test_mode = abap_true.

      WRITE:
        / TEXT-tst,
        / |{ TEXT-dep } { m_deleted }|,
        / |{ TEXT-inp } { m_inserted }|.

    ELSE.

      WRITE:
        / TEXT-rel,
        / |{ TEXT-del } { m_deleted }|,
        / |{ TEXT-ins } { m_inserted }|.

    ENDIF.

  ENDMETHOD.

ENDCLASS.
