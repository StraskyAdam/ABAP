@AbapCatalog.sqlViewName: 'ZIFIDASPOBASE'
@AbapCatalog.compiler.compareFilter: true
@AbapCatalog.preserveKey: true
@AccessControl.authorizationCheck: #NOT_REQUIRED
@EndUserText.label: 'DAS open purchase-order item base'
@Metadata.ignorePropagatedAnnotations: true
define view ZI_FI_DAS_PO_BASE
  with parameters
    p_use_replication : abap.char(1)
  as select from ZI_FI_DAS_EKPO(
    p_use_replication: $parameters.p_use_replication ) as item
    inner join ZI_FI_DAS_EKKO(
      p_use_replication: $parameters.p_use_replication ) as header
      on header.ebeln = item.ebeln

    inner join t001
      on t001.bukrs = header.bukrs

    left outer join lfa1
      on lfa1.lifnr = header.lifnr
{
  key item.ebeln        as PoNumber,
  key item.ebelp        as ItemNo,

      item.pstyp        as PsType,
      item.zstart_date  as WrkStrtDate,
      item.zend_date    as WrkEndDate,
      item.bukrs        as ItemCompanyCode,
      @Semantics.amount.currencyCode: 'PoCurrency'
      item.netwr        as NetValue,
      @Semantics.amount.currencyCode: 'PoCurrency'
      item.brtwr        as GrossValue,
      @Semantics.amount.currencyCode: 'PoCurrency'
      item.netpr        as NetPrice,
      item.peinh        as PriceUnit,
      @Semantics.quantity.unitOfMeasure: 'OrderUnit'
      item.menge        as OrderQuantity,
      @Semantics.unitOfMeasure: true
      item.meins        as OrderUnit,
      item.po_requestor as PoRequestor,
      item.po_preparer  as PoPreparer,
      item.zserv_amt    as ServiceAmount,
      item.zserv_perc   as ServicePercentage,
      item.txz01        as PoLineDesc,
      item.zactivitynbr,
      header.bukrs        as CompanyCode,
      @Semantics.currencyCode: true
      header.waers        as PoCurrency,
      header.lifnr        as SupplierId,
      header.aedat        as PoCreatedAt,
      header.bedat        as DocumentDate,
      header.ekorg        as PurchasingOrganization,
      header.unsez        as EpNumber,

      lfa1.name1        as SupplierName,
      
      t001.butxt        as PurchasingUnit,
      t001.land1        as Region,
      @Semantics.currencyCode: true
      t001.waers        as CocDeCurr }
