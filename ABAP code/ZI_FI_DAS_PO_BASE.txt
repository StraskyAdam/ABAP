@AbapCatalog.sqlViewName: 'ZIFIDASPOBASE'
@AbapCatalog.compiler.compareFilter: true
@AbapCatalog.preserveKey: true
@AccessControl.authorizationCheck: #NOT_REQUIRED
@EndUserText.label: 'DAS open purchase-order item base'
@Metadata.ignorePropagatedAnnotations: true
define view ZI_FI_DAS_PO_BASE
  as select from ekpo
    inner join ekko
      on ekko.ebeln = ekpo.ebeln

    inner join zfi_das_bukrs as header_scope
      on  header_scope.bukrs = ekko.bukrs
      and header_scope.scope = 'H'

    inner join zfi_das_bukrs as item_scope
      on  item_scope.bukrs = ekpo.bukrs
      and item_scope.scope = 'I'

    inner join t001
      on t001.bukrs = ekko.bukrs

    left outer join lfa1
      on lfa1.lifnr = ekko.lifnr
{
  key ekpo.ebeln        as PoNumber,
  key ekpo.ebelp        as ItemNo,

      ekpo.pstyp        as PsType,
      ekpo.zstart_date  as WorkStartDateRaw,
      ekpo.zend_date    as WorkEndDateRaw,
      ekpo.bukrs        as ItemCompanyCode,

      @Semantics.amount.currencyCode: 'PoCurrency'
      ekpo.netwr        as NetValue,

      @Semantics.amount.currencyCode: 'PoCurrency'
      ekpo.brtwr        as GrossValue,

      @Semantics.amount.currencyCode: 'PoCurrency'
      ekpo.netpr        as NetPrice,

      ekpo.peinh        as PriceUnit,

      @Semantics.quantity.unitOfMeasure: 'OrderUnit'
      ekpo.menge        as OrderQuantity,

      @Semantics.unitOfMeasure: true
      ekpo.meins        as OrderUnit,

      ekpo.po_requestor as PoRequestor,
      ekpo.po_preparer  as PoPreparer,
      ekpo.zserv_amt    as ServiceAmount,
      ekpo.zserv_perc   as ServicePercentage,
      ekpo.txz01        as PoLineDescription,
      ekpo.zactivitynbr as ActivityNumber,

      ekko.bukrs        as CompanyCode,

      @Semantics.currencyCode: true
      ekko.waers        as PoCurrency,

      ekko.lifnr        as SupplierId,
      ekko.aedat        as PoCreatedAt,
      ekko.bedat        as DocumentDate,
      ekko.ekorg        as PurchasingOrganization,
      ekko.unsez        as EpNumber,

      lfa1.name1        as SupplierName,
      t001.butxt        as PurchasingUnit,
      t001.land1        as Region,

      @Semantics.currencyCode: true
      t001.waers        as CompanyCodeCurrency
}
where ekko.bsart       = 'ZARP'
  and ekko.reason_code = ''
  and ekko.ebeln       between '8000000000' and '8999999999'
  and ( ekpo.pstyp = '0' or ekpo.pstyp = '9' )
  and ekpo.erekz       = ''
  and ekpo.loekz       = ''
  and (
       ekpo.bukrs <> '2780'
       or ekpo.xersy = ''
       or ekpo.xersy = ' '
       or (
            ekpo.xersy = 'X'
            and ekpo.zstart_date <> '00000000'
          )
      )
