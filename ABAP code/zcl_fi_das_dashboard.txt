CLASS zcl_fi_das_dashboard IMPLEMENTATION.

  METHOD get_audit
   BY DATABASE PROCEDURE
   FOR HDB
   LANGUAGE SQLSCRIPT
   OPTIONS READ-ONLY
   USING zfi_das_auditlog.

    DECLARE lc_services_completed CONSTANT NVARCHAR(50) = 'Services Completed Amount';

    DECLARE lc_client_context CONSTANT NVARCHAR(10) = 'CLIENT';

    lt_ranked =
      SELECT
        audit.ponumber,
        audit.lineitem AS itemno,
        audit.newvalue,
        ROW_NUMBER() OVER (
          PARTITION BY audit.ponumber,
                       audit.lineitem
          ORDER BY audit.changedon DESC
        ) AS row_number
      FROM zfi_das_auditlog AS audit
      WHERE audit.mandt = SESSION_CONTEXT( lc_client_context )
        AND audit.fname = lc_services_completed
        AND ( audit.ponumber, audit.lineitem ) IN (
              SELECT ponumber,
                     itemno
                FROM :it_keys )
        AND audit.changedon IN (
              SELECT MAX( _max.changedon )
                FROM zfi_das_auditlog AS _max
               WHERE _max.mandt = SESSION_CONTEXT( lc_client_context )
                 AND TO_INTEGER( SUBSTR( TO_NVARCHAR( TO_BIGINT( _max.changedon ) ), 5, 2 ) )
                     = MONTH( ADD_MONTHS( CURRENT_DATE, -1 ) )
               GROUP BY _max.ponumber,
                        _max.lineitem );

    et_audit =
      SELECT
        ponumber,
        itemno,
        TO_DECIMAL( newvalue, 13, 2 ) AS amount
      FROM :lt_ranked
      WHERE row_number = 1;

  ENDMETHOD.

    CLEAR ct_result.

      EXPORTING

      IMPORTING

        EXPORTING

                                    is_request = ls_request )

                      CHANGING  ct_result  = ct_result ).

  METHOD apply_sort.

    DATA lt_sort_order TYPE abap_sortorder_tab.

    FIELD-SYMBOLS: <fs_sample>    TYPE any,
                   <f_sort_value> TYPE any.

    IF ct_result IS INITIAL.
      RETURN.
    ENDIF.

    ASSIGN ct_result[ 1 ] TO <fs_sample>.

*   Keep only valid and unique requested sort fields
    LOOP AT is_request-orderby ASSIGNING FIELD-SYMBOL(<fs_orderby>).

      DATA(l_property) = to_upper( <fs_orderby>-property ).

      UNASSIGN <f_sort_value>.

      ASSIGN COMPONENT l_property OF STRUCTURE <fs_sample> TO <f_sort_value>.

      IF <f_sort_value> IS NOT ASSIGNED OR line_exists( lt_sort_order[ name = l_property ] ).
        CONTINUE.
      ENDIF.

      APPEND VALUE #( name       = l_property
                      descending = <fs_orderby>-descending ) TO lt_sort_order.

    ENDLOOP.

*   Append missing stable tie-breakers after the requested sort fields
    IF lt_sort_order IS NOT INITIAL.

      LOOP AT get_default_orderby( ) ASSIGNING FIELD-SYMBOL(<fs_default_orderby>).

        DATA(l_default_property) = to_upper( <fs_default_orderby>-property ).

        UNASSIGN <f_sort_value>.

        ASSIGN COMPONENT l_default_property OF STRUCTURE <fs_sample> TO <f_sort_value>.

        IF <f_sort_value> IS NOT ASSIGNED OR line_exists( lt_sort_order[ name = l_default_property ] ).
          CONTINUE.
        ENDIF.

        APPEND VALUE #( name       = l_default_property
                        descending = <fs_default_orderby>-descending ) TO lt_sort_order.

      ENDLOOP.

      SORT ct_result BY (lt_sort_order).

    ENDIF.

  ENDMETHOD.

  METHOD apply_paging.

    IF ct_result IS INITIAL.
      RETURN.
    ENDIF.

    DATA(l_skip) = is_request-paging-skip.
    DATA(l_top)  = is_request-paging-top.

    IF is_request-paging-top_requested = abap_true AND l_top = 0.
      CLEAR ct_result.
      RETURN.
    ENDIF.

    IF l_skip >= lines( ct_result ).
      CLEAR ct_result.
      RETURN.
    ENDIF.

    IF l_skip > 0.
      DELETE ct_result TO l_skip.
    ENDIF.

    IF l_top > 0 AND l_top < lines( ct_result ).
      DELETE ct_result FROM l_top + 1.
    ENDIF.

  ENDMETHOD.

  METHOD calculate_accrual.

*   Reset calculated fields before applying XSA calculation logic
    CLEAR: cs_result-totwrkcompamt,
           cs_result-totwrkcompamt_usd,
           cs_result-totwrkcompper,
           cs_result-straightlinemthd.

*   Material items do not use Service WTD/STL calculation
    IF is_detail-effective_pstype <> zif_fi_das_dashboard=>mc_pstype_service.
      cs_result-accrualmethod = mc_accrual_stl.
      RETURN.
    ENDIF.

*   Missing SERVICE_AMOUNT -> Straight Line
*   Provided SERVICE_AMOUNT -> Work To Date
    DATA(l_service_amount_supplied) = xsdbool( is_detail-serviceamount IS NOT INITIAL ).

    IF l_service_amount_supplied = abap_true.

      TRY.
          DATA(l_service_amount) = CONV t_amount_precision( is_detail-serviceamount ).

        CATCH cx_sy_conversion_error INTO DATA(lx_conversion).

          RAISE EXCEPTION TYPE zcx_fi_das_error
            MESSAGE e011(zfi_das) WITH is_detail-serviceamount
          EXPORTING previous = lx_conversion.
      ENDTRY.
    ENDIF.

*   Unsupported calculation methods produce no result
    IF cs_result-accrualmethod IS NOT INITIAL AND cs_result-accrualmethod <> mc_accrual_wtd AND cs_result-accrualmethod <> mc_accrual_stl.
      CLEAR cs_result-accrualmethod.
      RETURN.
    ENDIF.

*   calculates the STL candidate before method selection
    DATA(ls_straight_line) = calculate_straight_line( i_po_value = CONV t_amount_precision( is_detail-grossvalue * get_currency_factor( is_detail-pocurrency ) )
                                                      i_po_currency     = is_detail-pocurrency
                                                      i_work_start_date = is_detail-wrkstrtdate
                                                      i_work_end_date   = is_detail-wrkenddate ).

    IF cs_result-accrualmethod <> mc_accrual_stl AND l_service_amount_supplied = abap_true.

      cs_result-totwrkcompamt = CONV t_amount( zcl_fi_das_utility=>truncate_decimal( i_value    = CONV decfloat34( l_service_amount )
                                                                                     i_decimals = mc_amount_decimals ) ).

      DATA(l_usd_source_amount) = CONV t_amount_precision( l_service_amount / get_currency_factor( is_detail-pocurrency ) ).

      cs_result-totwrkcompamt_usd = convert_to_usd( i_amount          = l_usd_source_amount
                                                    i_source_currency = is_detail-pocurrency ).

      cs_result-totwrkcompper = zcl_fi_das_utility=>parse_percentage( CONV string( is_detail-servicepercentage ) ).

      cs_result-accrualmethod = mc_accrual_wtd.

      RETURN.

    ENDIF.

*   Straight Line candidate
    MOVE-CORRESPONDING ls_straight_line TO cs_result.
    cs_result-accrualmethod    = mc_accrual_stl.
    cs_result-straightlinemthd = abap_true.

  ENDMETHOD.

  METHOD calculate_accrued_amount.

    DATA(l_base) = COND t_amount( WHEN i_pstype = zif_fi_das_dashboard=>mc_pstype_service
                                    THEN i_total_work_amount
                                    ELSE i_gr_value ).

    r_amount = l_base - i_invoice_value.

    r_amount = COND #( WHEN r_amount < 0
                        THEN 0
                        ELSE r_amount ).

  ENDMETHOD.

  METHOD calculate_invoice_value.

    r_invoice_value = COND #( WHEN i_invoice_value > 0
                                THEN i_invoice_value
                                ELSE 0 ).

  ENDMETHOD.

  METHOD calculate_straight_line.

    rs_result-accrualmethod    = mc_accrual_stl.
    rs_result-straightlinemthd = abap_true.

*   Missing service dates -> initial amounts, method STL
    IF i_work_start_date IS INITIAL OR i_work_end_date IS INITIAL.
      RETURN.
    ENDIF.

    get_month_difference(
      EXPORTING
        i_work_start_date = i_work_start_date
        i_work_end_date   = i_work_end_date
      IMPORTING
        e_total_months    = DATA(l_total_months)
        e_elapsed_months  = DATA(l_elapsed_months) ).

*   If all calculated service periods have elapsed, calculates the completed value as a 1 / 1 interval
    IF l_total_months = l_elapsed_months.
      l_total_months   = 1.
      l_elapsed_months = 1.
    ENDIF.

    IF l_total_months > 0.

      DATA(l_division) = CONV t_amount_high( i_po_value / l_total_months ).

      DATA(l_monthly_amount) = CONV t_amount_precision( zcl_fi_das_utility=>truncate_decimal( i_value    = CONV decfloat34( l_division )
                                                                                              i_decimals = mc_calculation_decimals ) ).
      DATA(l_unrounded_amount) = CONV t_amount_high( l_monthly_amount * l_elapsed_months ).

      DATA(l_amount) = CONV t_amount( round( val  = CONV decfloat34( l_unrounded_amount )
                                             dec  = mc_amount_decimals
                                             mode = cl_abap_math=>round_half_up ) ).

*     Completed amount cannot exceed the original PO value
      IF l_amount > CONV t_amount( i_po_value ).
        l_amount = CONV t_amount( i_po_value ).
      ENDIF.
    ENDIF.

    rs_result-totwrkcompamt = l_amount.

    rs_result-totwrkcompamt_usd = convert_to_usd( i_amount          = l_amount / get_currency_factor( i_po_currency )
                                                  i_source_currency = i_po_currency ).
    IF i_po_value > 0.

      DATA(l_percentage) = CONV decfloat34( l_amount ) / CONV decfloat34( i_po_value ) * 100.

      rs_result-totwrkcompper =  CONV zcl_fi_das_dashboard=>t_percentage( round( val  = l_percentage
                                                                                 dec  = mc_percentage_decimals
                                                                                 mode = cl_abap_math=>round_half_up ) ).
    ENDIF.

  ENDMETHOD.

  METHOD determine_status.

*   Closure always yields Reviewed
    IF is_closed( is_status-po_closure ).
      cs_result-status = mc_status_reviewed.
      RETURN.
    ENDIF.

    IF is_status-status IS INITIAL OR is_status-status = mc_status_pending.
      cs_result-status = mc_status_pending.
      RETURN.
    ENDIF.

*   Any other status, or Reviewed without a date -> INITIAL
    IF is_status-status <> mc_status_reviewed OR is_status-changed_date IS INITIAL.
      CLEAR cs_result-status.
      RETURN.
    ENDIF.

    CASE get_review_action( i_changed_date = is_status-changed_date
                            i_review_day   = i_review_day ).
      WHEN mc_review_action_keep.
        cs_result-status = mc_status_reviewed.
      WHEN mc_review_action_reset.
        cs_result-status = mc_status_pending.
      WHEN OTHERS.
        CLEAR cs_result-status.
    ENDCASE.

  ENDMETHOD.

  METHOD determine_review.

    DATA(l_status) = is_status-status.

    IF ( i_mode = zif_fi_das_dashboard=>mc_under OR i_mode = zif_fi_das_dashboard=>mc_exclude ) AND
         is_closed( is_status-po_closure ) = abap_true.
      l_status = mc_status_reviewed.
    ENDIF.

    IF l_status IS INITIAL OR l_status = mc_status_pending.
      cs_result-changeddate = is_status-changed_date.
      cs_result-changedtime = is_status-changed_time.
      RETURN.
    ENDIF.

    IF l_status <> mc_status_reviewed OR is_status-changed_date IS INITIAL.
      CLEAR: cs_result-changeddate,
             cs_result-changedtime.
      RETURN.
    ENDIF.

    IF get_review_action( i_changed_date = is_status-changed_date
                          i_review_day   = i_review_day ) = mc_review_action_keep.
      cs_result-changeddate = is_status-changed_date.
      cs_result-changedtime = is_status-changed_time.
    ELSE.
      CLEAR: cs_result-changeddate,
             cs_result-changedtime.
    ENDIF.

  ENDMETHOD.

  METHOD is_closed.

    DATA(l_closure) = i_po_closure.
    TRANSLATE l_closure TO LOWER CASE.

    r_closed = xsdbool( l_closure = mc_boolean_true ).

  ENDMETHOD.

  METHOD convert_to_usd.

    DATA: l_source_amount TYPE t_amount_precision,
          l_usd_amount    TYPE decfloat34.

    CLEAR r_amount.

    IF i_keep_precision = abap_true.
      l_source_amount = i_amount.
    ELSE.
      l_source_amount = zcl_fi_das_utility=>truncate_decimal( i_value    = CONV decfloat34( i_amount )
                                                              i_decimals = mc_amount_decimals  ).
    ENDIF.

    IF l_source_amount IS INITIAL.
      RETURN.
    ENDIF.

*   Apply the historical currency factor before conversion.
    l_source_amount = l_source_amount * get_currency_factor( i_source_currency ).

    DATA(l_rate) = get_exchange_rate( i_source_currency ).

    l_usd_amount =  CONV decfloat34( l_source_amount ) * CONV decfloat34( l_rate ).

    r_amount = round( val  = l_usd_amount
                      dec  = mc_amount_decimals
                      mode = cl_abap_math=>round_half_up  ).

  ENDMETHOD.

  METHOD extract_filters.

    DATA: lr_item                     TYPE tt_string_range,
          lr_ponumber                 TYPE tt_string_range,
          lr_suppliername             TYPE tt_string_range,

          l_has_non_standard_for_amdp TYPE abap_bool VALUE abap_false.

    IF cs_request-filter-filter_select_options IS INITIAL.
      RETURN.
    ENDIF.

*   Separate filters that can be pushed down to the AMDP layer
*   All remaining filters will be handled by the generic ABAP filter logic
    LOOP AT cs_request-filter-filter_select_options ASSIGNING FIELD-SYMBOL(<fs_filter_select_options>).

      DATA(ls_range) = VALUE ts_string_range( sign = <fs_filter_select_options>-sign
                                              option = <fs_filter_select_options>-option
                                              low = <fs_filter_select_options>-low
                                              high = <fs_filter_select_options>-high ).

      CASE to_upper( <fs_filter_select_options>-property ).
        WHEN mc_ponumber.
          APPEND ls_range TO lr_ponumber.
        WHEN mc_itemno.
          APPEND ls_range TO lr_item.
        WHEN mc_suppliername.
          APPEND ls_range TO lr_suppliername.
        WHEN OTHERS.
          l_has_non_standard_for_amdp = abap_true.
      ENDCASE.

    ENDLOOP.

    TRY.
        cs_request-filter_for_amdp = VALUE #(
           ponumber     = convert_range_to_where( i_column = mc_ponumber     ir_range = lr_ponumber )
           item         = convert_range_to_where( i_column = mc_itemno       ir_range = lr_item )
           suppliername = convert_range_to_where( i_column = mc_suppliername ir_range = lr_suppliername ) ).

      CATCH cx_shdb_exception.
        CLEAR cs_request-filter_for_amdp.
        RETURN.
    ENDTRY.

*   All filter conditions were converted to AMDP filters, generic filter processing is no longer required
    IF l_has_non_standard_for_amdp = abap_false.
      CLEAR cs_request-filter-filter_string.
    ENDIF.

  ENDMETHOD.

  METHOD get_accounts.

    IF it_keys IS INITIAL.
      RETURN.
    ENDIF.

    SELECT
      ekkn~ebeln   AS ponumber,
      ekkn~ebelp   AS itemno,
      ekkn~sakto   AS glaccount,
      ekkn~kostl   AS costcenter,
      csks~zzrptun AS manageunit,
      csks~kostl   AS management_unit_found,
      cskt~datbi   AS valid_to,
      csks~datbi   AS csks_valid_to,
      ekkn~aufnr   AS internalorder,
      ekkn~anln1   AS assetid,
      ekkn~prctr   AS profitcenter,
      ekkn~zekkn   AS bbp_acc_no,
      ekkn~vproz   AS distpercentage,
      ekkn~menge   AS quantity
    FROM ekkn

     LEFT OUTER JOIN cskt ON cskt~kokrs = ekkn~kokrs AND
                             cskt~kostl = ekkn~kostl AND
                             cskt~spras = 'E' AND
                             cskt~datbi >= @sy-datum

     LEFT OUTER JOIN csks ON csks~kokrs = ekkn~kokrs AND
                             csks~kostl = ekkn~kostl AND
                             csks~datbi >= @sy-datum
     FOR ALL ENTRIES IN @it_keys
         WHERE ekkn~ebeln = @it_keys-ponumber AND
               ekkn~ebelp = @it_keys-itemno
     INTO TABLE @DATA(lt_accounts).

    SORT lt_accounts BY ponumber itemno bbp_acc_no.

    LOOP AT lt_accounts ASSIGNING FIELD-SYMBOL(<fs_account>).

      ASSIGN rt_accounts[ ponumber = <fs_account>-ponumber
                          itemno   = <fs_account>-itemno ] TO FIELD-SYMBOL(<fs_aggregate>).

      IF <fs_aggregate> IS NOT ASSIGNED OR sy-subrc <> 0.

        INSERT VALUE #( ponumber = <fs_account>-ponumber
                        itemno   = <fs_account>-itemno ) INTO TABLE rt_accounts ASSIGNING <fs_aggregate>.

      ENDIF.

      DATA(l_is_first_entry) = xsdbool( <fs_aggregate>-entry_count = 0 ).

      DATA(l_value) = zcl_fi_das_utility=>alpha_output( CONV string( <fs_account>-glaccount ) ).
      <fs_aggregate>-glaccount = zcl_fi_das_utility=>append_delimited(
                                    i_current = <fs_aggregate>-glaccount
                                    i_value   = l_value
                                    i_first   = l_is_first_entry ).

      l_value = zcl_fi_das_utility=>alpha_output( CONV string( <fs_account>-costcenter ) ).
      <fs_aggregate>-costcenter = zcl_fi_das_utility=>append_delimited(
                                     i_current = <fs_aggregate>-costcenter
                                     i_value   = l_value
                                     i_first   = l_is_first_entry ).

*     Management unit is aggregated only if the corresponding CSKS record exists.
      IF <fs_account>-management_unit_found IS NOT INITIAL.

        <fs_aggregate>-manageunit = zcl_fi_das_utility=>append_delimited(
                                            i_current = <fs_aggregate>-manageunit
                                            i_value   = CONV string( <fs_account>-manageunit )
                                            i_first   = xsdbool( <fs_aggregate>-management_unit_count = 0 ) ).

        <fs_aggregate>-management_unit_count = <fs_aggregate>-management_unit_count + 1.

      ENDIF.

      l_value = zcl_fi_das_utility=>alpha_output( CONV string( <fs_account>-internalorder ) ).
      <fs_aggregate>-internalorder = zcl_fi_das_utility=>append_delimited(
                                         i_current = <fs_aggregate>-internalorder
                                         i_value   = l_value
                                         i_first   = l_is_first_entry ).

      l_value = zcl_fi_das_utility=>alpha_output( CONV string( <fs_account>-assetid ) ).
      <fs_aggregate>-assetid = zcl_fi_das_utility=>append_delimited(
                                  i_current = <fs_aggregate>-assetid
                                  i_value   = l_value
                                  i_first   = l_is_first_entry ).

      <fs_aggregate>-profitcenter = zcl_fi_das_utility=>append_delimited(
                                         i_current = <fs_aggregate>-profitcenter
                                         i_value   = CONV string( <fs_account>-profitcenter )
                                         i_first   = l_is_first_entry ).

      <fs_aggregate>-bbp_acc_no = zcl_fi_das_utility=>append_delimited(
                                     i_current = <fs_aggregate>-bbp_acc_no
                                     i_value   = CONV string( <fs_account>-bbp_acc_no )
                                     i_first   = l_is_first_entry ).

      l_value = |{ <fs_account>-distpercentage NUMBER = RAW }|.
      CONDENSE l_value.
      <fs_aggregate>-distpercentage = zcl_fi_das_utility=>append_delimited(
                                          i_current = <fs_aggregate>-distpercentage
                                          i_value   = l_value
                                          i_first   = l_is_first_entry ).

      <fs_aggregate>-quantity = <fs_aggregate>-quantity + <fs_account>-quantity.

      <fs_aggregate>-entry_count = <fs_aggregate>-entry_count + 1.

    ENDLOOP.

    LOOP AT rt_accounts ASSIGNING <fs_aggregate>.

      <fs_aggregate>-assetid = zcl_fi_das_utility=>trim_delimiters( <fs_aggregate>-assetid ).

    ENDLOOP.

  ENDMETHOD.

  METHOD get_decimal_format.

           user_data~bname,

    INNER JOIN adr6 AS email
        ON email~persnumber = user_data~persnumber AND
           email~addrnumber = user_data~addrnumber
    INNER JOIN usr01 AS user_defaults ON user_defaults~bname = user_data~bname
    INNER JOIN usr02 AS logon_data ON logon_data~bname = user_data~bname
      WHERE email~smtp_addr = @l_email AND
            logon_data~ustyp = @mc_user_type_dialog AND
            logon_data~class NOT LIKE @mc_rpa_pattern AND
            logon_data~gltgv < @sy-datum AND
            logon_data~gltgb >= @sy-datum
    ORDER BY user_data~bname
      INTO TABLE @DATA(lt_candidates).

      IF <fs_candidate>-bname+0(1) CO mc_digit_chars.

      r_decimal_format = <fs_candidate>-dcpfm.

  METHOD get_history.

    IF it_keys IS INITIAL.
      RETURN.
    ENDIF.

    DATA(l_cutoff_date) = get_history_cutoff_date( ).

    SELECT
      ebeln AS ponumber,
      ebelp AS itemno,
      zekkn AS acc_assign,
      gjahr AS fiscal_year,
      belnr AS document_no,
      buzei AS document_item,
      bewtp AS category,
      vgabe AS transaction,
      menge AS quantity,
      wrbtr AS amount,
      dmbtr AS local_amount,
      shkzg AS debit_credit,
      budat AS posting_date
    FROM ekbe
      FOR ALL ENTRIES IN @it_keys
    WHERE ebeln = @it_keys-ponumber AND
          ebelp = @it_keys-itemno AND
          budat <= @l_cutoff_date AND
          ( bewtp = @mc_bewtp_e OR
          ( bewtp IN ( @mc_bewtp_q, @mc_bewtp_n )  AND
            vgabe IN ( @mc_vgabe_2, @mc_vgabe_3 ) ) )
      INTO TABLE @DATA(lt_history).

    LOOP AT lt_history ASSIGNING FIELD-SYMBOL(<fs_history>).

      ASSIGN it_details[ ponumber = <fs_history>-ponumber
                         itemno   = <fs_history>-itemno ] TO FIELD-SYMBOL(<fs_detail>).

      IF <fs_detail> IS NOT ASSIGNED OR sy-subrc <> 0.
        CONTINUE.
      ENDIF.

      ASSIGN rt_history[ ponumber = <fs_history>-ponumber
                         itemno   = <fs_history>-itemno ] TO FIELD-SYMBOL(<fs_aggregate>).

      IF <fs_aggregate> IS NOT ASSIGNED OR sy-subrc <> 0.

        INSERT VALUE #( ponumber = <fs_history>-ponumber
                        itemno   = <fs_history>-itemno ) INTO TABLE rt_history ASSIGNING <fs_aggregate>.
      ENDIF.

      DATA(l_currency_factor) = get_currency_factor( <fs_detail>-pocurrency ).

      DATA(l_sign) = COND i( WHEN <fs_history>-debit_credit = mc_credit_indicator
                              THEN -1
                              ELSE 1 ).

      IF <fs_history>-category = mc_bewtp_e.

        IF <fs_detail>-effective_pstype <> zif_fi_das_dashboard=>mc_pstype_material OR <fs_detail>-priceunit IS INITIAL.
          CONTINUE.
        ENDIF.

        DATA(l_gr_amount) = CONV decfloat34( <fs_history>-quantity ) * CONV decfloat34( <fs_detail>-netprice ) /
                            CONV decfloat34( <fs_detail>-priceunit ) * l_sign.

        <fs_aggregate>-gr_value = <fs_aggregate>-gr_value + l_gr_amount * l_currency_factor.

        <fs_aggregate>-grvalue_usd = <fs_aggregate>-grvalue_usd + convert_to_usd(
                                                                      i_amount          = CONV #( l_gr_amount )
                                                                      i_source_currency = <fs_detail>-pocurrency
                                                                      i_keep_precision  = abap_true ).

      ELSE.

        DATA(l_invoice_amount) = CONV t_amount_precision( <fs_history>-amount * l_sign  ).

        <fs_aggregate>-invvalue = <fs_aggregate>-invvalue + l_invoice_amount * l_currency_factor.

        <fs_aggregate>-invvalue_cocdecurr = <fs_aggregate>-invvalue_cocdecurr + <fs_history>-local_amount * l_currency_factor * l_sign.

        <fs_aggregate>-invvalue_usd = <fs_aggregate>-invvalue_usd + convert_to_usd(
                                                                      i_amount          = l_invoice_amount
                                                                      i_source_currency = <fs_detail>-pocurrency ).
      ENDIF.

    ENDLOOP.

  ENDMETHOD.

  METHOD get_history_cutoff_date.

    DATA(l_first_of_month) = CONV d( sy-datum(6) && mc_first_day_of_month ).

    IF sy-datum+6(2) <= mc_first_half_month_day.

      r_cutoff_date = l_first_of_month - 1.

    ELSE.

      DATA(l_next_month) = CONV d( l_first_of_month + mc_next_month_offset ).
      r_cutoff_date = CONV d( l_next_month(6) && mc_first_day_of_month ) - 1.

    ENDIF.

  ENDMETHOD.

  METHOD get_keys
    BY DATABASE PROCEDURE
    FOR HDB
    LANGUAGE SQLSCRIPT
    OPTIONS READ-ONLY
    USING
      zifidaspobase
      ekkn
      csks
      zfi_das_po_excl
      zfi_das_exchrt.

    DECLARE lc_empty_string     CONSTANT NVARCHAR(1) = '';
    DECLARE lc_space            CONSTANT NVARCHAR(1) = ' ';
    DECLARE lc_true             CONSTANT NVARCHAR(1) = 'X';
    DECLARE lc_false            CONSTANT NVARCHAR(1) = '';
    DECLARE lc_zero             CONSTANT NVARCHAR(1) = '0';
    DECLARE lc_all              CONSTANT NVARCHAR(1) = '*';

    DECLARE lc_pstype_material  CONSTANT NVARCHAR(1) = '0';
    DECLARE lc_pstype_service   CONSTANT NVARCHAR(1) = '9';

    DECLARE lc_currency_jpy     CONSTANT NVARCHAR(3) = 'JPY';
    DECLARE lc_currency_vnd     CONSTANT NVARCHAR(3) = 'VND';

    DECLARE lc_mode_over        CONSTANT NVARCHAR(1) = '1';
    DECLARE lc_mode_under       CONSTANT NVARCHAR(1) = '2';
    DECLARE lc_mode_all         CONSTANT NVARCHAR(1) = '3';

    DECLARE lc_initial_date     CONSTANT NVARCHAR(8) = '00000000';

    DECLARE lv_excluded_only NVARCHAR(1) =
      CASE
        WHEN :i_mode = :lc_mode_all THEN :lc_true
        ELSE :lc_false
      END;

*   Base dataset enriched
    lt_base_flags =
      SELECT
        base.ponumber,
        base.itemno,
        base.pstype,
        base.wrkstrtdate,
        base.wrkenddate,
        base.porequestor,
        base.popreparer,
        base.companycode,
        base.itemcompanycode,
        base.pocurrency,
        base.netvalue,
        base.suppliername,
        CASE
          WHEN EXISTS
          (
            SELECT 1
              FROM :it_japan_brazil AS japan_brazil
             WHERE japan_brazil.value = base.itemcompanycode
          )
          THEN :lc_true
          ELSE :lc_false
        END AS is_japan_brazil

      FROM zifidaspobase AS base
      WHERE base.mandt = :i_client;

*   Apply owner and PSTYPE/business selection logic
    lt_base_candidates =
      SELECT
        base.ponumber,
        base.itemno,
        CASE
          WHEN base.pstype = :lc_pstype_material
           AND base.wrkstrtdate <> :lc_initial_date
           AND base.is_japan_brazil = :lc_true
          THEN :lc_pstype_service
          ELSE base.pstype
        END AS pstype,
        LOWER( base.porequestor ) AS poowner_email,
        base.companycode,
        base.pocurrency,
        base.netvalue,
        base.suppliername

      FROM :lt_base_flags AS base

      WHERE
        (
             :i_poowner <> :lc_true
          OR LOWER( base.porequestor ) LIKE :i_email
          OR LOWER( base.popreparer )  LIKE :i_email
        )
        AND
        (
          (
            :i_pstype IN ( :lc_pstype_service, :lc_all )
            AND
            (
              (
                base.pstype = :lc_pstype_service
                AND base.wrkstrtdate <> :lc_initial_date
                AND base.wrkenddate  <> :lc_initial_date
              )
              OR
              (
                base.pstype = :lc_pstype_material
                AND base.wrkstrtdate <> :lc_initial_date
                AND base.is_japan_brazil = :lc_true
              )
            )
          )
          OR
          (
            :i_pstype IN ( :lc_pstype_material, :lc_all )
            AND base.pstype = :lc_pstype_material
            AND
            (
                 base.is_japan_brazil = :lc_false
              OR
              (
                base.wrkstrtdate = :lc_initial_date
                AND base.is_japan_brazil = :lc_true
              )
            )
          )
        );

    IF :i_ponumber_filter <> :lc_empty_string THEN
      lt_base_candidates = APPLY_FILTER( :lt_base_candidates, :i_ponumber_filter );
    END IF;

    IF :i_supplier_filter <> :lc_empty_string THEN
      lt_base_candidates = APPLY_FILTER( :lt_base_candidates, :i_supplier_filter );
    END IF;

*   Owner mode skips access scoping
    IF :i_poowner = :lc_true THEN

      lt_scoped_candidates =
        SELECT
          candidate.ponumber,
          candidate.itemno,
          candidate.pstype,
          candidate.poowner_email,
          candidate.companycode,
          candidate.pocurrency,
          candidate.netvalue
        FROM :lt_base_candidates AS candidate;

    ELSE

*     Non-owner mode applies company code, cost center and management unit scope
      lt_scoped_candidates =
        SELECT DISTINCT
          candidate.ponumber,
          candidate.itemno,
          candidate.pstype,
          candidate.poowner_email,
          candidate.companycode,
          candidate.pocurrency,
          candidate.netvalue
        FROM :lt_base_candidates AS candidate

        LEFT OUTER JOIN ekkn AS account
          ON  account.ebeln = candidate.ponumber
          AND account.ebelp = candidate.itemno
          AND account.mandt = :i_client

        LEFT OUTER JOIN csks AS cost_center
          ON  cost_center.kostl = account.kostl
          AND cost_center.kokrs = account.kokrs
          AND cost_center.datbi >= CURRENT_DATE
          AND cost_center.mandt = :i_client

        WHERE
          EXISTS
          (
            SELECT 1
              FROM :it_company_codes AS scope_company
             WHERE scope_company.wildcard = :lc_true
                OR scope_company.value = candidate.companycode
          )

          AND
          (
               (
                 EXISTS
                 (
                   SELECT 1
                     FROM :it_cost_centers AS scope_cost
                    WHERE scope_cost.wildcard = :lc_true
                      AND scope_cost.is_null = :lc_false
                 )
                 AND account.kostl IS NOT NULL
               )
            OR
               EXISTS
               (
                 SELECT 1
                   FROM :it_cost_centers AS scope_cost
                  WHERE scope_cost.wildcard = :lc_false
                    AND scope_cost.is_null = :lc_false
                    AND LTRIM( account.kostl, :lc_zero )
                        = LTRIM( scope_cost.value, :lc_zero )
               )
            OR
               (
                 account.kostl IS NULL
                 AND EXISTS
                 (
                   SELECT 1
                     FROM :it_cost_centers AS scope_cost
                    WHERE scope_cost.is_null = :lc_true
                 )
               )
          )

          AND
          (
               cost_center.zzrptun = :lc_empty_string
            OR cost_center.zzrptun = :lc_space
            OR
               (
                 EXISTS
                 (
                   SELECT 1
                     FROM :it_management_units AS scope_management
                    WHERE scope_management.wildcard = :lc_true
                 )
                 AND NOT EXISTS
                 (
                   SELECT 1
                     FROM :it_management_units AS scope_management_null
                    WHERE scope_management_null.is_null = :lc_true
                 )
               )
            OR
               (
                 cost_center.zzrptun IS NULL
                 AND EXISTS
                 (
                   SELECT 1
                     FROM :it_management_units AS scope_management_null
                    WHERE scope_management_null.is_null = :lc_true
                 )
               )
            OR
               EXISTS
               (
                 SELECT 1
                   FROM :it_management_units AS scope_management
                  WHERE scope_management.wildcard = :lc_false
                    AND scope_management.is_null  = :lc_false
                    AND LTRIM( scope_management.value, :lc_zero )
                        = cost_center.zzrptun
               )
          );

    END IF;

  lt_po_totals =
      SELECT
        candidate.ponumber,
        candidate.pstype,
        candidate.pocurrency,
        SUM(
          CASE
            WHEN candidate.pocurrency IN ( :lc_currency_jpy, :lc_currency_vnd )
            THEN candidate.netvalue * 100
            ELSE candidate.netvalue
          END
        ) AS po_total
      FROM :lt_scoped_candidates AS candidate
      GROUP BY
        candidate.ponumber,
        candidate.pstype,
        candidate.pocurrency;

    lt_po_threshold =
      SELECT
        total.ponumber,
        total.pstype,
        total.po_total,
        ROUND(
          :i_threshold_usd * CAST( 1 / exchange_rate.exchange_rate AS DECIMAL(15,6) ),2 ) AS threshold_converted
      FROM :lt_po_totals AS total
      INNER JOIN zfi_das_exchrt AS exchange_rate
        ON exchange_rate.source_curr = total.pocurrency
       AND exchange_rate.mandt = :i_client;

    lt_threshold_classified =
      SELECT DISTINCT
        candidate.ponumber,
        candidate.itemno,
        candidate.pstype,
        candidate.poowner_email
      FROM :lt_scoped_candidates AS candidate
      INNER JOIN :lt_po_threshold AS threshold
        ON  threshold.ponumber = candidate.ponumber
        AND threshold.pstype   = candidate.pstype
      WHERE
           (
             :i_mode = :lc_mode_over
             AND threshold.po_total > threshold.threshold_converted
           )
        OR (
             :i_mode = :lc_mode_under
             AND threshold.po_total <= threshold.threshold_converted
           )
        OR (
             :lv_excluded_only = :lc_true
             AND
             (
                  threshold.po_total > threshold.threshold_converted
               OR threshold.po_total <= threshold.threshold_converted
             )
           );
*   Candidate lookup bypasses only exclusion membership, not business selection
    IF :i_apply_exclusions = :lc_false THEN

      lt_exclusion_applied =
        SELECT
          candidate.ponumber,
          candidate.itemno,
          candidate.pstype,
          candidate.poowner_email
        FROM :lt_threshold_classified AS candidate;

*   Exclude mode returns only excluded items; other modes remove excluded items
    ELSEIF :lv_excluded_only = :lc_true THEN

      lt_exclusion_applied =
        SELECT
          candidate.ponumber,
          candidate.itemno,
          candidate.pstype,
          candidate.poowner_email
        FROM :lt_threshold_classified AS candidate
        WHERE EXISTS
        (
          SELECT 1
            FROM zfi_das_po_excl AS exclusion
           WHERE exclusion.ponumber = candidate.ponumber
             AND exclusion.itemno   = candidate.itemno
             AND exclusion.mandt    = :i_client
        );

    ELSE

      lt_exclusion_applied =
        SELECT
          candidate.ponumber,
          candidate.itemno,
          candidate.pstype,
          candidate.poowner_email
        FROM :lt_threshold_classified AS candidate
        WHERE NOT EXISTS
        (
          SELECT 1
            FROM zfi_das_po_excl AS exclusion
           WHERE exclusion.ponumber = candidate.ponumber
             AND exclusion.itemno   = candidate.itemno
             AND exclusion.pstype   = candidate.pstype
             AND exclusion.mandt    = :i_client
        );

    END IF;

    lt_filtered =
      SELECT DISTINCT
        candidate.ponumber,
        candidate.itemno,
        candidate.pstype,
        candidate.poowner_email
      FROM :lt_exclusion_applied AS candidate;

    IF :i_item_filter <> :lc_empty_string THEN
      lt_filtered = APPLY_FILTER( :lt_filtered, :i_item_filter );
    END IF;

    IF :i_count_requested = :lc_true THEN
      SELECT COUNT(*)
        INTO e_total_count
        FROM :lt_filtered;
    ELSE
      e_total_count = 0;
    END IF;

    lt_numbered =
      SELECT
        candidate.ponumber,
        candidate.itemno,
        candidate.pstype,
        candidate.poowner_email,
        ROW_NUMBER() OVER
        (
          ORDER BY
            CASE
              WHEN :i_pstype_desc_requested = :lc_true
              THEN candidate.pstype
              ELSE :lc_empty_string
            END DESC,
            candidate.ponumber ASC,
            candidate.itemno ASC
        ) AS row_number_value
      FROM :lt_filtered AS candidate;

    et_keys =
      SELECT
        numbered.ponumber,
        numbered.itemno,
        numbered.pstype,
        numbered.poowner_email
      FROM :lt_numbered AS numbered
      WHERE numbered.row_number_value > :i_skip
        AND
        (
          :i_top_requested = :lc_empty_string
          OR numbered.row_number_value <= :i_skip + :i_top
        )
      ORDER BY numbered.row_number_value;

  ENDMETHOD.

  METHOD get_month_difference.

*   on days 1-6 the previous month is still the current period
    DATA(l_current_date) = sy-datum.
    DATA(l_current_day)  = CONV i( l_current_date+6(2) ).

    IF l_current_day BETWEEN mc_period_first_day AND mc_period_last_day.
      l_current_date = l_current_date - l_current_day.
    ENDIF.

*   Work end date already elapsed -> 100%
    IF i_work_end_date(6) < l_current_date(6).
      e_total_months   = 1.
      e_elapsed_months = 1.
      RETURN.
    ENDIF.

    DATA(l_start_bucket) = zcl_fi_das_utility=>month_bucket( i_date      = i_work_start_date
                                                             i_split_day = mc_stl_split_day ).

    DATA(l_end_bucket) = zcl_fi_das_utility=>month_bucket( i_date      = i_work_end_date
                                                           i_split_day = mc_stl_split_day ).

*   Work end date in current month -> current period follows end date
    DATA(l_current_bucket) = COND d( WHEN i_work_end_date(6) = l_current_date(6)
                                        THEN l_end_bucket
                                        ELSE zcl_fi_das_utility=>month_bucket( i_date      = l_current_date
                                                                               i_split_day = mc_period_last_day ) ).

    e_total_months = zcl_fi_das_utility=>months_between( i_from = l_start_bucket
                                                         i_to   = l_end_bucket ).

    e_elapsed_months = zcl_fi_das_utility=>months_between( i_from = l_start_bucket
                                                           i_to   = l_current_bucket ).
    IF e_elapsed_months <= 0.

      e_elapsed_months = COND #( WHEN i_work_end_date(6) = l_current_date(6)
                                    THEN 1
                                    ELSE 0 ).
    ENDIF.

    IF e_total_months <= 0.
      e_total_months = 1.
    ENDIF.

  ENDMETHOD.

  METHOD get_review_action.

    r_review_action = mc_review_action_none.

    IF i_changed_date IS INITIAL.
      RETURN.
    ENDIF.

    DATA(l_current_date) = COND d( WHEN i_current_date IS INITIAL
                                      THEN sy-datum
                                      ELSE i_current_date ).

    DATA(l_current_year)  = CONV i( l_current_date+0(4) ).
    DATA(l_current_month) = CONV i( l_current_date+4(2) ).
    DATA(l_current_day)   = CONV i( l_current_date+6(2) ).

    DATA(l_changed_year)  = CONV i( i_changed_date+0(4) ).
    DATA(l_changed_month) = CONV i( i_changed_date+4(2) ).
    DATA(l_changed_day)   = CONV i( i_changed_date+6(2) ).

    DATA(l_grace_period) = xsdbool( l_current_day BETWEEN mc_period_first_day AND mc_period_last_day ).

*    Previous year: keep Reviewed only during the grace period when the review belongs to the immediately preceding month
    IF l_changed_year < l_current_year.

      r_review_action = mc_review_action_reset.

      IF l_grace_period = abap_true.
        DATA(l_previous_month) = CONV d( l_current_date - l_current_day ).
        IF i_changed_date(6) = l_previous_month(6) AND l_changed_day > l_current_day.
          r_review_action = mc_review_action_keep.
        ENDIF.
      ENDIF.
      RETURN.
    ENDIF.

*   Future year is not expected
    IF l_changed_year > l_current_year.
      RETURN.
    ENDIF.

*   Previous month in the same year
    IF l_changed_month < l_current_month.
      r_review_action = mc_review_action_reset.
      IF l_changed_month = l_current_month - 1 AND l_grace_period = abap_true AND l_changed_day > l_current_day.
        r_review_action = mc_review_action_keep.
      ENDIF.
      RETURN.
    ENDIF.

*   Future month is not expected
    IF l_changed_month > l_current_month.
      RETURN.
    ENDIF.

*   Same month: review completed before the configured review day
    IF l_changed_day < i_review_day.

      r_review_action = COND #( WHEN l_current_day >= i_review_day
                                    THEN mc_review_action_reset
                                    ELSE mc_review_action_keep ).
      RETURN.
    ENDIF.

*   Same month: review completed after the configured review day
    IF l_changed_day > i_review_day.
      IF l_current_day > i_review_day.
        r_review_action = mc_review_action_keep.
      ENDIF.
      RETURN.
    ENDIF.

*   Same month: review completed exactly on the configured review day
    IF l_current_day >= i_review_day.
      r_review_action = mc_review_action_keep.
    ENDIF.

  ENDMETHOD.

  METHOD get_review_day.

    SELECT SINGLE config_value
      FROM zfi_das_conf_val
    WHERE config_name = @mc_config_review_day
      INTO @DATA(l_value).

    IF sy-subrc <> 0 OR l_value IS INITIAL.
      r_review_day = mc_default_review_day.
      RETURN.
    ENDIF.

    TRY.
        r_review_day = CONV i( l_value ).
      CATCH cx_sy_conversion_error.
        r_review_day = mc_default_review_day.
    ENDTRY.

    IF r_review_day NOT BETWEEN mc_min_review_day AND mc_max_review_day.

      RAISE EXCEPTION TYPE zcx_fi_das_error
       MESSAGE e016(zfi_das).

    ENDIF.

  ENDMETHOD.

                   CHANGING cs_result    = rs_result ).

                       CHANGING  cs_result = rs_result ).

                    CHANGING  cs_result = rs_result ).

    apply_closure( CHANGING cs_result = rs_result ).

    IF is_request-mode = zif_fi_das_dashboard=>mc_over.

      apply_notification( EXPORTING is_notification = VALUE #( is_context-notifications[ ponumber = is_key-ponumber
                                                                                         itemno   = is_key-itemno ] OPTIONAL )
                          CHANGING  cs_result = rs_result ).
    ENDIF.

    IF is_request-mode = zif_fi_das_dashboard=>mc_exclude.

      apply_exclusion( EXPORTING is_exclusion = VALUE #( is_context-exclusions[ ponumber = is_key-ponumber
                                                                                itemno   = is_key-itemno ] OPTIONAL )
                       CHANGING  cs_result = rs_result ).

    ENDIF.

    apply_odata_format( EXPORTING i_mode           = is_request-mode
                                  i_decimal_format = i_decimal_format
                        CHANGING  cs_result        = rs_result ).

  ENDMETHOD.

  METHOD get_rule_text.

    r_text = SWITCH string( i_rule_type
              WHEN mc_rule_gl THEN mc_rule_text_gl
              WHEN mc_rule_cost_center THEN mc_rule_text_cost_center
              WHEN mc_rule_profit_center THEN mc_rule_text_profit_center
              WHEN mc_rule_po THEN mc_rule_text_po
              WHEN mc_rule_vendor THEN mc_rule_text_vendor
              WHEN mc_rule_internal_order THEN mc_rule_text_internal_order ).

  ENDMETHOD.

  METHOD get_status.

    IF it_keys IS NOT INITIAL.

      SELECT
        ponumber,
        lineitem              AS itemno,
        status,
        accrualmethod         AS accrual_method,
        changeddate           AS changed_date,
        changedtime           AS changed_time,
        changed_on,
        reviewed_by,
        po_closure,
        totalworkcompltodate  AS wtd_amount,
        workcomplastmonthamt  AS last_month_amount
      FROM zfi_das_postatus
         FOR ALL ENTRIES IN @it_keys
      WHERE ponumber = @it_keys-ponumber AND
            lineitem = @it_keys-itemno
         INTO TABLE @rt_statuses.

      IF sy-subrc <> 0.
        CLEAR rt_statuses.
      ENDIF.
    ENDIF.

  ENDMETHOD.

  METHOD get_threshold.

    SELECT SINGLE po_threshold
      FROM zfi_das_thresh
    WHERE currency = @mc_currency_usd
      INTO @r_threshold_usd.

    IF sy-subrc <> 0.

      RAISE EXCEPTION TYPE zcx_fi_das_error
        MESSAGE e014(zfi_das).

    ENDIF.

  ENDMETHOD.

  METHOD get_users
   BY DATABASE PROCEDURE
   FOR HDB
   LANGUAGE SQLSCRIPT
   OPTIONS READ-ONLY
   USING usr21
         adrp
         adr6
         usr01
         usr02.

    DECLARE lc_empty_string     CONSTANT NVARCHAR(1)  = '';
    DECLARE lc_user_type_dialog CONSTANT NVARCHAR(1)  = 'A';
    DECLARE lc_rpa_pattern      CONSTANT NVARCHAR(5)  = '%RPA%';
    DECLARE lc_numeric_chars    CONSTANT NVARCHAR(10) = '0123456789';

    lt_requested =
      SELECT DISTINCT
        LOWER( TRIM( poowner_email ) ) AS email
      FROM :it_details
      WHERE poowner_email IS NOT NULL AND TRIM( poowner_email ) <> :lc_empty_string;

    lt_candidates =
      SELECT
        LOWER( TRIM( email.smtp_addr ) ) AS email,
        user_data.bname                  AS username,
        person.name_text                 AS full_name,
        user_defaults.dcpfm              AS decimal_format,
        user_defaults.datfm              AS datfm,
        user_data.bname                  AS usr21_bname,
        email.addrnumber                 AS adr6_addrnumber,
        email.persnumber                 AS adr6_persnumber
      FROM usr21 AS user_data

      INNER JOIN adrp AS person ON person.persnumber = user_data.persnumber

      INNER JOIN adr6 AS email ON email.persnumber = user_data.persnumber AND
                                  email.addrnumber = user_data.addrnumber

      INNER JOIN usr01 AS user_defaults ON user_defaults.bname = user_data.bname

      INNER JOIN usr02 AS user_logon ON user_logon.bname = user_data.bname

      INNER JOIN :lt_requested AS requested ON requested.email = LOWER( TRIM( email.smtp_addr ) )

      WHERE user_logon.ustyp = :lc_user_type_dialog AND
            user_logon.class NOT LIKE :lc_rpa_pattern AND
            user_logon.gltgv < CURRENT_DATE AND
            user_logon.gltgb >= CURRENT_DATE AND
            LOCATE( :lc_numeric_chars, SUBSTRING( user_data.bname, 1, 1 ) ) = 0;

    et_users =
      SELECT email, username, full_name, decimal_format, datfm
      FROM
      (
        SELECT
          email,
          username,
          full_name,
          decimal_format,
          datfm,
          ROW_NUMBER() OVER
          (
            PARTITION BY email
            ORDER BY
              usr21_bname DESC,
              adr6_addrnumber DESC,
              adr6_persnumber DESC
          ) AS row_number_value

        FROM :lt_candidates )
      WHERE row_number_value = 1;

  ENDMETHOD.

  METHOD is_full_processing_required.

*   Generic sorting or filtering requires the full result set
    r_is_full_processing_required = xsdbool( is_request-orderby IS NOT INITIAL OR
                                             is_request-filter-filter_string IS NOT INITIAL ).
  ENDMETHOD.

  METHOD remove_default_sorting.

*   Skip sorting when the requested order is a prefix of the AMDP order
*   The AMDP can additionally lead with PSTYPE DESC when requested
    DATA l_pstype_desc_requested TYPE abap_bool.

    CLEAR cs_request-pstype_desc_requested.

    DATA(lt_default_orderby) = get_default_orderby( ).

    IF cs_request-orderby IS NOT INITIAL AND
       to_upper( cs_request-orderby[ 1 ]-property ) = mc_pstype AND
                 cs_request-orderby[ 1 ]-descending = abap_true.

      INSERT VALUE #( property   = mc_pstype
                      descending = abap_true ) INTO lt_default_orderby INDEX 1.

      l_pstype_desc_requested = abap_true.

    ENDIF.

    IF lines( cs_request-orderby ) > lines( lt_default_orderby ).
      RETURN.
    ENDIF.

    LOOP AT cs_request-orderby INTO DATA(ls_orderby).

      DATA(l_index) = sy-tabix.

      ls_orderby-property = to_upper( ls_orderby-property ).

      IF ls_orderby <> lt_default_orderby[ l_index ].
        RETURN.
      ENDIF.

    ENDLOOP.

    cs_request-pstype_desc_requested = l_pstype_desc_requested.

    CLEAR cs_request-orderby.

  ENDMETHOD.

  METHOD set_full_processing.

    IF is_full_processing_required( cs_request ).
      CLEAR: cs_request-paging-skip,
             cs_request-paging-top,
             cs_request-paging-top_requested.
    ENDIF.

  ENDMETHOD.

  METHOD validate_request.

    IF is_request-mode <> zif_fi_das_dashboard=>mc_over  AND
       is_request-mode <> zif_fi_das_dashboard=>mc_under AND
       is_request-mode <> zif_fi_das_dashboard=>mc_exclude.

      RAISE EXCEPTION TYPE zcx_fi_das_error
        MESSAGE e005(zfi_das).

    ENDIF.

    IF is_request-parameter-email IS INITIAL.

      RAISE EXCEPTION TYPE zcx_fi_das_error
        MESSAGE e006(zfi_das).

    ENDIF.

    IF is_request-parameter-poowner <> abap_true AND is_request-parameter-poowner <> abap_false.

      RAISE EXCEPTION TYPE zcx_fi_das_error
        MESSAGE e007(zfi_das).

    ENDIF.

    IF is_request-parameter-pstype <> zif_fi_das_dashboard=>mc_pstype_material AND
       is_request-parameter-pstype <> zif_fi_das_dashboard=>mc_pstype_service  AND
       is_request-parameter-pstype <> zif_fi_das_dashboard=>mc_pstype_all AND
       is_request-mode <> zif_fi_das_dashboard=>mc_exclude.

      RAISE EXCEPTION TYPE zcx_fi_das_error
         MESSAGE e008(zfi_das).

    ENDIF.

    IF is_request-paging-skip < 0 OR is_request-paging-top < 0.

      RAISE EXCEPTION TYPE zcx_fi_das_error
        MESSAGE e009(zfi_das).

    ENDIF.

    IF is_request-filter-filter_select_options IS NOT INITIAL AND is_request-filter-filter_string IS INITIAL.

      RAISE EXCEPTION TYPE zcx_fi_das_error
          MESSAGE e010(zfi_das).

    ENDIF.

  ENDMETHOD.

  METHOD get_default_orderby.

    rt_orderby = VALUE #( ( property = mc_ponumber descending = abap_false )
                          ( property = mc_itemno   descending = abap_false ) ).

  ENDMETHOD.

  METHOD get_unique_keys.

    LOOP AT it_key ASSIGNING FIELD-SYMBOL(<fs_key>).
      INSERT <fs_key> INTO TABLE rt_key.
    ENDLOOP.

  ENDMETHOD.

ENDCLASS.
