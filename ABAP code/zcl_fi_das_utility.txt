CLASS zcl_fi_das_utility DEFINITION
  PUBLIC
  FINAL
  CREATE PRIVATE.

  PUBLIC SECTION.

    TYPES: tt_string          TYPE STANDARD TABLE OF string  WITH EMPTY KEY,
           t_amount_precision TYPE p LENGTH 16 DECIMALS 8.

    CONSTANTS:
      mc_dot                TYPE c LENGTH 1 VALUE '.',
      mc_comma              TYPE c LENGTH 1 VALUE ',',
      mc_percentage_pattern TYPE string VALUE '^[+-]?[0-9]+([.][0-9]+)?$',

      mc_decimal_dot        TYPE xudcpfm VALUE 'X',
      mc_decimal_space      TYPE xudcpfm VALUE 'Y',

      mc_max_decimals       TYPE i VALUE 14,

      mc_first_day_of_month TYPE c LENGTH 2 VALUE '01',
      mc_month_roll_days    TYPE i VALUE 32,

      mc_timezone_cet       TYPE ttzz-tzone VALUE 'CET',
      mc_timezone_utc       TYPE ttzz-tzone VALUE 'UTC',

      mc_zero               TYPE i VALUE 0,
      mc_padding_length     TYPE i VALUE 2,
      mc_single_digit_limit TYPE i VALUE 10,
      mc_zero_character     TYPE c LENGTH 1 VALUE '0',

      mc_boolean_true       TYPE c LENGTH 5 VALUE 'true',
      mc_boolean_false      TYPE c LENGTH 5 VALUE 'false',

      mc_min_odata_decimals TYPE i VALUE 0,
      mc_max_odata_decimals TYPE i VALUE 8,
      mc_group_size         TYPE i VALUE 3,
      mc_plus_sign          TYPE c LENGTH 1 VALUE '+',
      mc_minus_sign         TYPE c LENGTH 1 VALUE '-'.

    CLASS-METHODS normalize_email
      IMPORTING
        i_email        TYPE zif_fi_das_dashboard=>t_email
      RETURNING
        VALUE(r_email) TYPE zif_fi_das_dashboard=>t_email.

    CLASS-METHODS split_list
      IMPORTING
        i_value          TYPE string
        i_delimiter      TYPE char1 DEFAULT mc_comma
      RETURNING
        VALUE(rt_values) TYPE tt_string.

    CLASS-METHODS append_delimited
      IMPORTING
        i_current       TYPE string
        i_value         TYPE string
        i_delimiter     TYPE char1 DEFAULT ';'
        i_skip_initial  TYPE abap_bool DEFAULT abap_false
        i_first         TYPE abap_bool DEFAULT abap_false
      RETURNING
        VALUE(r_result) TYPE string.

    CLASS-METHODS trim_delimiters
      IMPORTING
        i_value        TYPE string
        i_delimiter    TYPE char1 DEFAULT ';'
      RETURNING
        VALUE(r_value) TYPE string.

    CLASS-METHODS alpha_output
      IMPORTING
        i_value        TYPE string
      RETURNING
        VALUE(r_value) TYPE string.

    CLASS-METHODS utc_to_cet_timestamp
      IMPORTING
        i_timestamp        TYPE timestampl
      RETURNING
        VALUE(r_timestamp) TYPE timestampl
      RAISING
        cx_sy_conversion_error.

    CLASS-METHODS is_current_utc_day
      IMPORTING
        i_timestamp         TYPE timestampl
      RETURNING
        VALUE(r_is_current) TYPE abap_bool.

    CLASS-METHODS bool_to_text
      IMPORTING
        i_value        TYPE char5
      RETURNING
        VALUE(r_value) TYPE char5.

    CLASS-METHODS truncate_decimal
      IMPORTING
        i_value        TYPE decfloat34
        i_decimals     TYPE i
      RETURNING
        VALUE(r_value) TYPE decfloat34
      RAISING
        zcx_fi_das_error.

    CLASS-METHODS parse_percentage
      IMPORTING
        i_value             TYPE string
      RETURNING
        VALUE(r_percentage) TYPE zcl_fi_das_dashboard=>t_percentage
      RAISING
        zcx_fi_das_error.

    CLASS-METHODS month_bucket
      IMPORTING
        i_date          TYPE d
        i_split_day     TYPE i
      RETURNING
        VALUE(r_bucket) TYPE d.

    CLASS-METHODS months_between
      IMPORTING
        i_from          TYPE d
        i_to            TYPE d
      RETURNING
        VALUE(r_months) TYPE i.

    CLASS-METHODS format_odata_amount
      IMPORTING
        i_amount                TYPE t_amount_precision
        i_decimal_format        TYPE xudcpfm
        i_decimals              TYPE i DEFAULT 2
        i_use_grouping          TYPE abap_bool DEFAULT abap_true
        i_pad_nonzero_below_ten TYPE abap_bool DEFAULT abap_true
      RETURNING
        VALUE(r_value)          TYPE string
      RAISING
        zcx_fi_das_error.

  PRIVATE SECTION.

    TYPES:
      BEGIN OF ts_number_parts,
        sign     TYPE char1,
        integer  TYPE string,
        fraction TYPE string,
      END OF ts_number_parts.

    CLASS-METHODS get_number_parts
      IMPORTING
        i_amount        TYPE t_amount_precision
        i_decimals      TYPE i
      RETURNING
        VALUE(rs_parts) TYPE ts_number_parts.

    CLASS-METHODS get_number_separators
      IMPORTING
        i_decimal_format     TYPE xudcpfm
      EXPORTING
        e_decimal_separator  TYPE char1
        e_grouping_separator TYPE char1.

    CLASS-METHODS group_integer
      IMPORTING
        i_integer            TYPE string
        i_grouping_separator TYPE char1
      RETURNING
        VALUE(r_integer)     TYPE string.

ENDCLASS.

CLASS zcl_fi_das_utility IMPLEMENTATION.

  METHOD alpha_output.

    DATA(l_value) = i_value.

    SHIFT l_value LEFT  DELETING LEADING space.
    SHIFT l_value RIGHT DELETING TRAILING space.

    IF l_value IS INITIAL.
      RETURN.
    ENDIF.

    SHIFT l_value LEFT DELETING LEADING mc_zero_character.

    r_value = COND #( WHEN l_value IS INITIAL
                         THEN mc_zero_character
                         ELSE l_value ).

  ENDMETHOD.

  METHOD append_delimited.

    DATA(l_value) = i_value.

    SHIFT l_value LEFT DELETING LEADING space.

    IF i_skip_initial = abap_true.
      SHIFT l_value RIGHT DELETING TRAILING space.

      IF l_value IS INITIAL.
        r_result = i_current.
        RETURN.
      ENDIF.
    ENDIF.

    IF i_first = abap_true OR ( i_skip_initial = abap_true AND i_current IS INITIAL ).
      r_result = l_value.
      RETURN.
    ENDIF.

    r_result = i_current && i_delimiter && l_value.

  ENDMETHOD.

  METHOD format_odata_amount.

    IF i_decimals < mc_min_odata_decimals OR i_decimals > mc_max_odata_decimals.

      RAISE EXCEPTION TYPE zcx_fi_das_error
         MESSAGE e018(zfi_das) WITH i_decimals.

    ENDIF.

    DATA(l_parts) = get_number_parts( i_amount   = i_amount
                                      i_decimals = i_decimals ).

    IF i_pad_nonzero_below_ten = abap_true AND i_amount <> mc_zero AND i_amount > - mc_single_digit_limit AND i_amount < mc_single_digit_limit.

      WHILE strlen( l_parts-integer ) < mc_padding_length.
        l_parts-integer = mc_zero_character && l_parts-integer.
      ENDWHILE.

    ENDIF.

    get_number_separators(
      EXPORTING
        i_decimal_format     = i_decimal_format
      IMPORTING
        e_decimal_separator  = DATA(l_decimal_separator)
        e_grouping_separator = DATA(l_grouping_separator) ).

    IF i_use_grouping = abap_true.

      l_parts-integer = group_integer( i_integer            = l_parts-integer
                                       i_grouping_separator = l_grouping_separator ).

    ENDIF.

    r_value = l_parts-sign && l_parts-integer.

    IF i_decimals > 0.

      r_value = r_value && l_decimal_separator && l_parts-fraction.

    ENDIF.

  ENDMETHOD.

  METHOD get_number_parts.

    DATA l_write_text TYPE c LENGTH 100.

    WRITE i_amount TO l_write_text
      DECIMALS i_decimals
      NO-GROUPING.

    DATA(l_normalized) = CONV string( l_write_text ).

    CONDENSE l_normalized NO-GAPS.

    IF l_normalized IS INITIAL.
      l_normalized = mc_zero_character.
    ENDIF.

    REPLACE ALL OCCURRENCES OF mc_comma IN l_normalized WITH mc_dot.

*   WRITE can place the minus sign after the number
    DATA(l_last_position) = strlen( l_normalized ) - 1.

    IF substring( val = l_normalized
                  off = l_last_position
                  len = 1 ) = mc_minus_sign.

      rs_parts-sign = mc_minus_sign.

      l_normalized = substring( val = l_normalized
                                off = 0
                                len = l_last_position ).
    ENDIF.

    DATA(l_character) = substring( val = l_normalized
                                   off = 0
                                   len = 1 ).

    IF l_character = mc_minus_sign.

      IF rs_parts-sign IS INITIAL.
        rs_parts-sign = mc_minus_sign.
      ENDIF.

      l_normalized = substring( val = l_normalized
                                off = 1 ).

    ELSEIF l_character = mc_plus_sign.

      l_normalized = substring( val = l_normalized
                                off = 1 ).

    ENDIF.

    FIND FIRST OCCURRENCE OF mc_dot IN l_normalized MATCH OFFSET DATA(l_decimal_position).

    IF sy-subrc = 0.

      IF l_decimal_position > mc_zero.

        rs_parts-integer = substring( val = l_normalized
                                      off = mc_zero
                                      len = l_decimal_position ).
      ENDIF.

      DATA(l_position) = l_decimal_position + 1.

      IF l_position < strlen( l_normalized ).

        rs_parts-fraction = substring( val = l_normalized
                                       off = l_position ).
      ENDIF.

    ELSE.
      rs_parts-integer = l_normalized.
    ENDIF.

    IF rs_parts-integer IS INITIAL.
      rs_parts-integer = mc_zero_character.
    ENDIF.

    DATA(l_fraction_length) = strlen( rs_parts-fraction ).

    IF l_fraction_length > i_decimals.

      rs_parts-fraction = substring( val = rs_parts-fraction
                                     off = mc_zero
                                     len = i_decimals ).

    ELSEIF l_fraction_length < i_decimals.

      rs_parts-fraction = rs_parts-fraction && repeat( val = mc_zero_character
                                                       occ = i_decimals - l_fraction_length ).

    ENDIF.

  ENDMETHOD.

  METHOD get_number_separators.

    CASE i_decimal_format.

      WHEN mc_decimal_dot.
        e_decimal_separator  = mc_dot.
        e_grouping_separator = mc_comma.

      WHEN mc_decimal_space.
        e_decimal_separator  = mc_comma.
        e_grouping_separator = space.

      WHEN space.
        e_decimal_separator  = mc_comma.
        e_grouping_separator = mc_dot.

    ENDCASE.

  ENDMETHOD.

  METHOD group_integer.

    DATA(l_length) = strlen( i_integer ).

    DATA(l_first_group) = l_length MOD mc_group_size.

    IF l_first_group = 0.
      l_first_group = mc_group_size.
    ENDIF.

    r_integer = substring( val = i_integer
                           off = 0
                           len = l_first_group ).

    DATA(l_position) = l_first_group.

    WHILE l_position < l_length.

      DATA(l_group) = substring( val = i_integer
                                 off = l_position
                                 len = mc_group_size ).

      r_integer = |{ r_integer }{ i_grouping_separator }{ l_group }|.

      l_position = l_position + mc_group_size.

    ENDWHILE.

  ENDMETHOD.

  METHOD is_current_utc_day.

    IF i_timestamp IS INITIAL.
      RETURN.
    ENDIF.

    GET TIME STAMP FIELD DATA(l_now).

    CONVERT TIME STAMP l_now
      TIME ZONE mc_timezone_utc
      INTO DATE DATA(l_current_date)
           TIME DATA(l_current_time).

    CONVERT TIME STAMP i_timestamp
      TIME ZONE mc_timezone_utc
      INTO DATE DATA(l_timestamp_date)
           TIME DATA(l_timestamp_time).

    r_is_current = xsdbool( l_timestamp_date = l_current_date ).

  ENDMETHOD.

  METHOD months_between.

    DATA(l_year_diff) = CONV i( i_to(4) ) - CONV i( i_from(4) ).

    DATA(l_month_diff) = CONV i( i_to+4(2) ) - CONV i( i_from+4(2) ).

    r_months = l_year_diff * 12 + l_month_diff.

  ENDMETHOD.

  METHOD month_bucket.

    DATA(l_first_day_of_month) = CONV d( i_date(6) && mc_first_day_of_month ).

    IF CONV i( i_date+6(2) ) > i_split_day.

      DATA(l_first_day_of_next_month) = CONV d( l_first_day_of_month + mc_month_roll_days ).
      r_bucket = l_first_day_of_next_month(6) && mc_first_day_of_month.

    ELSE.
      r_bucket = l_first_day_of_month.
    ENDIF.

  ENDMETHOD.

  METHOD normalize_email.

    r_email = to_lower( condense( val  = CONV string( i_email )
                                  from = space ) ).

  ENDMETHOD.

  METHOD parse_percentage.

    DATA(l_value) = i_value.

    CONDENSE l_value NO-GAPS.

    IF l_value IS INITIAL.
      RETURN.
    ENDIF.

    IF l_value CS mc_comma AND l_value CS mc_dot.
      RAISE EXCEPTION TYPE zcx_fi_das_error
        MESSAGE e012(zfi_das) WITH i_value.
    ENDIF.

    REPLACE ALL OCCURRENCES OF mc_comma IN l_value WITH mc_dot.

    TRY.
        FIND REGEX mc_percentage_pattern IN l_value.

        IF sy-subrc <> 0.
          RAISE EXCEPTION TYPE zcx_fi_das_error
            MESSAGE e012(zfi_das) WITH i_value.
        ENDIF.

        r_percentage = CONV decfloat34( l_value ).

      CATCH cx_sy_conversion_error INTO DATA(lx_conversion).

        RAISE EXCEPTION TYPE zcx_fi_das_error
          MESSAGE e012(zfi_das) WITH i_value
        EXPORTING previous = lx_conversion.
    ENDTRY.

  ENDMETHOD.

  METHOD split_list.

    SPLIT i_value AT i_delimiter INTO TABLE DATA(lt_parts).

    rt_values = VALUE #( FOR l_part IN lt_parts
                          LET l_trimmed = condense( l_part ) IN
                            ( l_trimmed ) ).

    DELETE rt_values WHERE table_line IS INITIAL.

  ENDMETHOD.

  METHOD trim_delimiters.

    DATA(l_value) = i_value.
    DATA(l_start) = 0.
    DATA(l_end)   = strlen( l_value ).
    DATA l_last TYPE i.

    WHILE l_start < l_end
      AND l_value+l_start(1) = i_delimiter.
      l_start = l_start + 1.
    ENDWHILE.

    WHILE l_end > l_start.
      l_last = l_end - 1.

      IF l_value+l_last(1) <> i_delimiter.
        EXIT.
      ENDIF.

      l_end = l_last.
    ENDWHILE.

    r_value = substring(
      val = l_value
      off = l_start
      len = l_end - l_start ).

  ENDMETHOD.

  METHOD truncate_decimal.

    IF i_decimals < 0 OR i_decimals > mc_max_decimals.
      RAISE EXCEPTION TYPE zcx_fi_das_error
        MESSAGE e018(zfi_das) WITH i_decimals.
    ENDIF.

    DATA(l_factor) = CONV decfloat34( 10 ) ** i_decimals.

    r_value = trunc( i_value * l_factor ) / l_factor.

  ENDMETHOD.

  METHOD utc_to_cet_timestamp.

    IF i_timestamp IS INITIAL.
      RETURN.
    ENDIF.

*   Separate whole seconds without rounding the input
    DATA(l_utc_whole) = CONV timestampl( trunc( CONV decfloat34( i_timestamp ) ) ).

    DATA(l_fraction) = CONV decfloat34( i_timestamp ) - CONV decfloat34( l_utc_whole ).

*   UTC -> local civil time using SAP CET time zone rules
    CONVERT TIME STAMP l_utc_whole TIME ZONE mc_timezone_cet
      INTO DATE DATA(l_local_date)
           TIME DATA(l_local_time).

*   Preserve the original seven fractional digits
    r_timestamp =  CONV timestampl(  |{ l_local_date DATE = RAW }{ l_local_time TIME = RAW }| ) + CONV timestampl( l_fraction ).

  ENDMETHOD.

  METHOD bool_to_text.

    r_value = COND #( WHEN i_value = abap_true
                        THEN mc_boolean_true
                        ELSE mc_boolean_false ).

  ENDMETHOD.

ENDCLASS.
