REPORT zfi_das_exclusion.

" Create this include; SAP_BASIS 750 SP0020 / HANA is required.
INCLUDE zfi_das_exclusion_def.

CONSTANTS:
  mc_message_abort TYPE symsgty VALUE 'A',
  mc_message_status TYPE symsgty VALUE 'S',
  mc_message_error TYPE symsgty VALUE 'E'.

PARAMETERS p_dry AS CHECKBOX DEFAULT abap_true.

START-OF-SELECTION.
  DATA l_exclusion TYPE REF TO lcl_exclusion.
  TRY.
      CREATE OBJECT l_exclusion.
      l_exclusion->run( i_dry = p_dry ).
    CATCH zcx_fi_das_error INTO DATA(l_error).
      DATA(l_text) = l_error->get_text( ).
      IF sy-batch = abap_true.
        MESSAGE l_text TYPE mc_message_abort.
      ELSE.
        MESSAGE l_text TYPE mc_message_status DISPLAY LIKE mc_message_error.
      ENDIF.
  ENDTRY.
