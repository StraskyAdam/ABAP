  METHOD vectors_and_priority.
    DATA(ls_item) = base_item( ).
    DATA(ls_rule) = base_rule( ).
    ls_rule-po = VALUE #( value = CONV string( ls_item-ponumber ) ).
    ls_rule-vendor = ls_item-vendor.
    DATA(lt_vectors) = mo_cut->evaluate(
      it_items = VALUE #( ( ls_item ) ( ls_item ) )
      it_rules = VALUE #( ( ls_rule ) ( ls_rule ) ) ).
    cl_abap_unit_assert=>assert_equals( act = lines( lt_vectors ) exp = 1 ).
    DATA(lt_result) = mo_cut->results( it_vectors = lt_vectors
      i_pstype = zif_fi_das_dashboard=>mc_pstype_service ).
    cl_abap_unit_assert=>assert_equals( act = lt_result[ 1 ]-ruletype exp = lcl_exclusion=>mc_gl ).
    cl_abap_unit_assert=>assert_equals(
      act = lt_result[ 1 ]-pstype exp = zif_fi_das_dashboard=>mc_pstype_service ).
    " Every CASE arm and the cross-rule priority order.
    lt_vectors = VALUE #( ( ponumber = ls_item-ponumber itemno = ls_item-itemno
                           gl = abap_true cc = abap_true iv = abap_true
                           pc = abap_true po = abap_true vd = abap_true ) ).
    DO 6 TIMES.
      lt_result = mo_cut->results( it_vectors = lt_vectors
        i_pstype = zif_fi_das_dashboard=>mc_pstype_material ).
      DATA(l_expected) = SWITCH zfi_das_po_excl-ruletype( sy-index
        WHEN 1 THEN lcl_exclusion=>mc_gl WHEN 2 THEN lcl_exclusion=>mc_cc
        WHEN 3 THEN lcl_exclusion=>mc_iv WHEN 4 THEN lcl_exclusion=>mc_pc
        WHEN 5 THEN lcl_exclusion=>mc_po ELSE lcl_exclusion=>mc_vd ).
      cl_abap_unit_assert=>assert_equals( act = lt_result[ 1 ]-ruletype exp = l_expected ).
      READ TABLE lt_vectors INTO DATA(ls_next)
        WITH KEY ponumber = ls_item-ponumber itemno = ls_item-itemno.
      CASE sy-index.
        WHEN 1. ls_next-gl = abap_false.
        WHEN 2. ls_next-cc = abap_false.
        WHEN 3. ls_next-iv = abap_false.
        WHEN 4. ls_next-pc = abap_false.
        WHEN 5. ls_next-po = abap_false.
      ENDCASE.
      CLEAR lt_vectors.
      INSERT ls_next INTO TABLE lt_vectors.
    ENDDO.
  ENDMETHOD.
